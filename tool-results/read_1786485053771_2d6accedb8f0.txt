    27→/* Escape state name                                                           */
    28→/* ========================================================================== */
    29→
    30→const char *vtx_escape_state_name(vtx_escape_state_t s)
    31→{
    32→    switch (s) {
    33→    case VTX_ESCAPE_NONE:   return "NoEscape";
    34→    case VTX_ESCAPE_ARG:    return "ArgEscape";
    35→    case VTX_ESCAPE_GLOBAL: return "GlobalEscape";
    36→    }
    37→    return "Unknown";
    38→}

    43→
    44→static inline bool is_allocation(vtx_node_opcode_t opcode)
    45→{
    46→    return opcode == VTX_OP_NewObject ||
    47→           opcode == VTX_OP_NewArray  ||
    48→           opcode == VTX_OP_Allocate;
    49→}

    51→/* ========================================================================== */
    52→/* Internal: identify allocations and build the map                            */
    53→/* ========================================================================== */
    54→
    55→static int build_allocation_map(vtx_graph_t *graph, vtx_escape_map_t *map,
    56→                                 vtx_arena_t *arena)
    57→{
    58→    vtx_node_table_t *table = &graph->node_table;
    59→    uint32_t count = table->count;
    60→
    61→    map->state_count = count;
    62→    map->states = vtx_arena_alloc(arena, count * sizeof(vtx_escape_state_t));
    63→    if (!map->states) return -1;
    64→
    65→    /* PEA-1-2: Initialize ALL states to VTX_ESCAPE_GLOBAL (conservative
    66→     * default for non-allocations). Only allocations get their actual
    67→     * state set in the finalization step (line ~768). Previously this
    68→     * was memset to 0 (NoEscape), which meant vtx_pea_get_escape()
    69→     * returned NoEscape for non-allocations — breaking the public API
    70→     * contract ("returns GlobalEscape for non-allocations") and causing
    71→     * vtx_pea_is_scalar_replaceable() to incorrectly return true for
    72→     * Constants, Parameters, Phis, etc. */
    73→    for (uint32_t s = 0; s < count; s++) {
    74→        map->states[s] = VTX_ESCAPE_GLOBAL;
    75→    }
    76→
    77→    /* Count allocations first for capacity */
    78→    uint32_t alloc_count = 0;
    79→    for (uint32_t i = 0; i < count; i++) {
    80→        vtx_node_t *node = &table->nodes[i];
    81→        if (!node->dead && is_allocation(node->opcode)) {
    82→            alloc_count++;
    83→        }
    84→    }
    85→
    86→    map->alloc_capacity = alloc_count > 0 ? alloc_count : 1;
    87→    map->alloc_ids = vtx_arena_alloc(arena, map->alloc_capacity * sizeof(vtx_nodeid_t));
    88→    if (!map->alloc_ids) return -1;
    89→
    90→    map->alloc_count = 0;
    91→    for (uint32_t i = 0; i < count; i++) {
    92→        vtx_node_t *node = &table->nodes[i];
    93→        if (!node->dead && is_allocation(node->opcode)) {
    94→            map->alloc_ids[map->alloc_count++] = node->id;
    95→        }
    96→    }
    97→
    98→    return 0;
    99→}

   117→                                  const vtx_escape_state_t *b,

   120→    return memcmp(a, b, count * sizeof(vtx_escape_state_t)) != 0;

   123→/* ========================================================================== */
   124→/* Internal: join two escape state arrays (dst = join(dst, src))               */
   125→/* Returns true if dst changed.                                                */
   126→/* ========================================================================== */
   127→
   128→static bool join_escape_states(vtx_escape_state_t *dst,
   129→                                const vtx_escape_state_t *src,
   130→                                uint32_t count)
   131→{
   132→    bool changed = false;
   133→    for (uint32_t i = 0; i < count; i++) {
   134→        vtx_escape_state_t joined = vtx_escape_join(dst[i], src[i]);
   135→        if (joined != dst[i]) {
   136→            dst[i] = joined;
   137→            changed = true;
   138→        }
   139→    }
   140→    return changed;
   141→}

   143→/* ========================================================================== */
   144→/* Internal: build per-block state arrays                                      */
   145→/* ========================================================================== */
   146→
   147→static int build_block_states(vtx_graph_t *graph, vtx_pea_analysis_t *result,
   148→                               vtx_arena_t *arena)
   149→{
   150→    uint32_t block_count = graph->block_count;
   151→    uint32_t state_count = result->escape_map.state_count;
   152→
   153→    result->block_state_count = block_count;
   154→    result->block_states = vtx_arena_alloc(arena,
   155→        block_count * sizeof(vtx_pea_block_state_t));
   156→    if (!result->block_states) return -1;
   157→
   158→    for (uint32_t i = 0; i < block_count; i++) {
   159→        vtx_pea_block_state_t *bs = &result->block_states[i];
   160→        bs->state_count = state_count;
   161→        bs->entry_changed = false;
   162→
   163→        bs->entry_state = vtx_arena_alloc(arena,
   164→            state_count * sizeof(vtx_escape_state_t));
   165→        bs->exit_state = vtx_arena_alloc(arena,
   166→            state_count * sizeof(vtx_escape_state_t));
   167→
   168→        if (!bs->entry_state || !bs->exit_state) return -1;
   169→
   170→        /* Initialize all to NoEscape */
   171→        memset(bs->entry_state, 0, state_count * sizeof(vtx_escape_state_t));
   172→        memset(bs->exit_state, 0, state_count * sizeof(vtx_escape_state_t));
   173→    }
   174→
   175→    return 0;
   176→}

   210→/* ========================================================================== */
   211→/* Internal: transfer function for a single node                               */
   212→/*                                                                           */
   213→/* Examines how the node uses values and updates escape states accordingly.   */
   214→/* ========================================================================== */
   215→
   216→/* PEA-1-1: recursively propagate escape state through Phi chains.
   217→ *
   218→ * When an escape point (Return/StoreField/Call*) encounters a Phi as
   219→ * input, the Phi itself is not an allocation, so the old code skipped
   220→ * it — the escape never flowed back to the Phi's allocation inputs.
   221→ * This function walks through a Phi's inputs (and recursively through
   222→ * nested Phis) to find all allocation inputs and propagate the escape
   223→ * state to them.
   224→ *
   225→ * `depth` limits recursion to prevent infinite loops on cyclic Phi
   226→ * graphs (shouldn't happen in SSA, but defensive). */
   227→static void propagate_escape_through_phi(vtx_node_table_t *table,
   228→                                          vtx_escape_state_t *state,
   229→                                          uint32_t state_count,
   230→                                          vtx_nodeid_t phi_id,
   231→                                          vtx_escape_state_t escape_state,
   232→                                          uint32_t depth)
   233→{
   234→    if (depth > 8) return;  /* recursion limit */
   235→    if (phi_id >= state_count) return;
   236→
   237→    vtx_node_t *phi = vtx_node_get(table, phi_id);
   238→    if (!phi || phi->dead || phi->opcode != VTX_OP_Phi) return;
   239→
   240→    for (uint32_t i = 0; i < phi->input_count; i++) {
   241→        vtx_nodeid_t input_id = phi->inputs[i];
   242→        if (input_id >= state_count) continue;
   243→
   244→        vtx_node_t *input = vtx_node_get(table, input_id);
   245→        if (!input || input->dead) continue;
   246→
   247→        if (is_allocation(input->opcode)) {
   248→            /* Found an allocation input — propagate the escape state. */
   249→            if (state[input_id] < escape_state) {
   250→                state[input_id] = escape_state;
   251→            }
   252→        } else if (input->opcode == VTX_OP_Phi) {
   253→            /* Nested Phi — recurse. */
   254→            propagate_escape_through_phi(table, state, state_count,
   255→                                          input_id, escape_state, depth + 1);
   256→        }
   257→    }
   258→}
   259→
   260→/* Helper: propagate escape to a value, handling Phi chains (PEA-1-1).
   261→ * If the value is an allocation, set its escape state directly.
   262→ * If the value is a Phi, walk through to find allocation inputs. */
   263→static void propagate_escape_to_value(vtx_node_table_t *table,
   264→                                       vtx_escape_state_t *state,
   265→                                       uint32_t state_count,
   266→                                       vtx_nodeid_t value_id,
   267→                                       vtx_escape_state_t escape_state)
   268→{
   269→    if (value_id >= state_count) return;
   270→    vtx_node_t *val = vtx_node_get(table, value_id);
   271→    if (!val || val->dead) return;
   272→
   273→    if (is_allocation(val->opcode)) {
   274→        if (state[value_id] < escape_state) {
   275→            state[value_id] = escape_state;
   276→        }
   277→    } else if (val->opcode == VTX_OP_Phi) {
   278→        /* PEA-1-1: walk through the Phi to find allocation inputs. */
   279→        propagate_escape_through_phi(table, state, state_count,
   280→                                      value_id, escape_state, 0);
   281→    }
   282→}
   283→
   284→static void transfer_node(vtx_node_t *node, vtx_node_table_t *table,
   285→                           vtx_escape_state_t *state, uint32_t state_count)
   286→{
   287→    if (node->dead) return;
   288→
   289→    switch (node->opcode) {
   290→    /* ---- Return: any returned allocation escapes globally ----
   291→     *
   292→     * PEA-1-1: if the returned value is a Phi, walk through the Phi
   293→     * to find all allocation inputs and mark them GlobalEscape. */
   294→    case VTX_OP_Return:
   295→        for (uint32_t i = 0; i < node->input_count; i++) {
   296→            vtx_nodeid_t input_id = node->inputs[i];
   297→            if (input_id < state_count) {
   298→                propagate_escape_to_value(table, state, state_count,
   299→                                           input_id, VTX_ESCAPE_GLOBAL);
   300→            }
   301→        }
   302→        break;
   303→
   304→    /* ---- StoreField: if the stored value is an allocation, it escapes
   305→     * based on the receiver's escape state ----
   306→     *
   307→     * PEA-1-4: previously the code unconditionally joined with
   308→     * VTX_ESCAPE_ARG, which bumped even NoEscape containers to
   309→     * ArgEscape — disabling scalar replacement for the common pattern
   310→     * of an object holding a private reference to another non-escaping
   311→     * object. Fix: only bump to the container's state (no unconditional
   312→     * ARG join). The container's state already captures whether the
   313→     * value can be observed externally.
   314→     *
   315→     * PEA-1-1: if the stored value is a Phi, walk through to find
   316→     * allocation inputs. */
   317→    case VTX_OP_StoreField:
   318→        if (node->input_count >= 2) {
   319→            vtx_nodeid_t receiver_id = node->inputs[node->input_count - 2];
   320→            vtx_nodeid_t value_id    = node->inputs[node->input_count - 1];
   321→
   322→            /* Determine the container's escape state. */
   323→            vtx_escape_state_t container_state = VTX_ESCAPE_GLOBAL;
   324→            if (receiver_id < state_count) {
   325→                vtx_node_t *recv_node = vtx_node_get(table, receiver_id);
   326→                if (recv_node && !recv_node->dead && is_allocation(recv_node->opcode)) {
   327→                    container_state = state[receiver_id];
   328→                }
   329→            }
   330→
   331→            /* PEA-1-4: propagate container_state to the stored value,
   332→             * WITHOUT the unconditional ARG bump. The value escapes at
   333→             * least as much as the container (if the container escapes,
   334→             * the stored value escapes with it). If the container is
   335→             * NoEscape, the value stays NoEscape too. */
   336→            propagate_escape_to_value(table, state, state_count,
   337→                                      value_id, container_state);
   338→        }
   339→        break;
   340→
   341→    /* ---- StoreIndexed: storing into an array — same as StoreField ----
   342→     * PEA-1-4: same fix — no unconditional ARG bump. */
   343→    case VTX_OP_StoreIndexed:
   344→        if (node->input_count >= 3) {
   345→            vtx_nodeid_t array_id = node->inputs[node->input_count - 3];
   346→            vtx_nodeid_t value_id = node->inputs[node->input_count - 1];
   347→
   348→            vtx_escape_state_t array_state = VTX_ESCAPE_GLOBAL;
   349→            if (array_id < state_count) {
   350→                vtx_node_t *arr_node = vtx_node_get(table, array_id);
   351→                if (arr_node && !arr_node->dead && is_allocation(arr_node->opcode)) {
   352→                    array_state = state[array_id];
   353→                }
   354→            }
   355→            propagate_escape_to_value(table, state, state_count,
   356→                                      value_id, array_state);
   357→        }
   358→        break;
   359→
   360→    /* ---- Calls: any allocation passed as argument escapes ----
   361→     *
   362→     * PEA-1-1: if an argument is a Phi, walk through to find
   363→     * allocation inputs and propagate the escape state to them. */
   364→    case VTX_OP_CallStatic:
   365→    case VTX_OP_CallVirtual:
   366→    case VTX_OP_CallInterface:
   367→    case VTX_OP_CallRuntime:
   368→        for (uint32_t i = 0; i < node->input_count; i++) {
   369→            vtx_nodeid_t input_id = node->inputs[i];
   370→            if (input_id >= state_count) continue;
   371→
   372→            /* PEA-1-1: propagate through Phi chains. CallRuntime
   373→             * is conservative (GlobalEscape); others use ArgEscape. */
   374→            vtx_escape_state_t call_escape =
   375→                (node->opcode == VTX_OP_CallRuntime)
   376→                    ? VTX_ESCAPE_GLOBAL
   377→                    : VTX_ESCAPE_ARG;
   378→            propagate_escape_to_value(table, state, state_count,
   379→                                       input_id, call_escape);
   380→        }
   381→        break;
   382→
   383→    /* ---- Monitor: using an object in a monitor makes it globally escape ---- */
   384→    case VTX_OP_DeoptGuard:
   385→        /* DeoptGuard: the guarded value is consumed but doesn't escape */
   386→        break;
   387→
   388→    /* ---- CheckCast / InstanceOf: the object doesn't escape through these
   389→       (they are pure queries on the object's type) ---- */
   390→    case VTX_OP_CheckCast:
   391→    case VTX_OP_InstanceOf:
   392→        /* No escape — these are read-only type checks */
   393→        break;
   394→
   395→    /* ---- Guard: the checked value doesn't escape ---- */
   396→    case VTX_OP_Guard:
   397→        /* Guards don't cause escape — they just check a condition */
   398→        break;
   399→
   400→    /* ---- Phi: a Phi of allocations — propagate escape states ---- */
   401→    case VTX_OP_Phi:
   402→        /* Phi nodes merge values from multiple control flow paths.
   403→         * If any input allocation has a non-NoEscape state, all allocation
   404→         * inputs of the Phi must have at least that escape state, because
   405→         * the Phi merges them into a single value that may escape.
   406→         *
   407→         * Bug 8 fix: Previously, the Phi transfer was a no-op — it just
   408→         * iterated over inputs without propagating escape states. This meant
   409→         * that if Phi(A, B) where A escapes but B doesn't, B would remain
   410→         * NoEscape even though the merged value escapes. Now we compute
   411→         * the maximum escape state across all Phi inputs and propagate it
   412→         * to every allocation input of the Phi. */
   413→        {
   414→            vtx_escape_state_t max_state = VTX_ESCAPE_NONE;
   415→            /* First pass: compute the maximum escape state across all inputs */
   416→            for (uint32_t i = 0; i < node->input_count; i++) {
   417→                vtx_nodeid_t input_id = node->inputs[i];
   418→                if (input_id < state_count) {
   419→                    vtx_node_t *input = vtx_node_get(table, input_id);
   420→                    if (input && !input->dead && is_allocation(input->opcode)) {
   421→                        max_state = vtx_escape_join(max_state, state[input_id]);
   422→                    }
   423→                }
   424→            }
   425→            /* Second pass: propagate max_state to all allocation inputs.
   426→             * If any input allocation has a lower escape state than the
   427→             * Phi's maximum, it must be elevated because the Phi's result
   428→             * (which merges all inputs) carries the highest escape state. */
   429→            for (uint32_t i = 0; i < node->input_count; i++) {
   430→                vtx_nodeid_t input_id = node->inputs[i];
   431→                if (input_id < state_count) {
   432→                    vtx_node_t *input = vtx_node_get(table, input_id);
   433→                    if (input && !input->dead && is_allocation(input->opcode)) {
   434→                        if (state[input_id] < max_state) {
   435→                            state[input_id] = max_state;
   436→                            /* changed flag is not accessible here, but the
   437→                             * caller (transfer_block -> worklist) will detect
   438→                             * the state change when comparing exit states. */
   439→                        }
   440→                    }
   441→                }
   442→            }
   443→        }
   444→        break;
   445→
   446→    default:
   447→        break;
   448→    }
   449→}

   451→/* ========================================================================== */
   452→/* Internal: pre-compute block-to-node mapping (§2.1 — fix O(N×B) scan)      */
   453→/* ========================================================================== */
   454→/*
   455→ * PEA audit: transfer_block re-scans ALL N nodes for EVERY block to find
   456→ * which nodes belong to that block. For a graph with B blocks and N nodes,
   457→ * this is O(N×B) per dataflow iteration — for T2 graphs with 5000 nodes
   458→ * and 50 blocks, that's 250K comparisons per iteration × ~10 iterations
   459→ * = 2.5M comparisons.
   460→ *
   461→ * GraalVM's PEA uses a pre-computed BlockMap (BlockBag in
   462→ * EscapeAnalysisPhase). V8's escape analysis uses a schedule
   463→ * (Schedule::rpo_order) to iterate nodes per-block.
   464→ *
   465→ * Fix: pre-compute a block→node-list mapping once before the worklist,
   466→ * then iterate only the relevant nodes in transfer_block. */
   467→typedef struct {
   468→    vtx_nodeid_t *nodes;    /* node IDs belonging to this block */
   469→    uint32_t      count;    /* number of nodes in this block */
   470→} vtx_block_node_list_t;
   471→
   472→static vtx_block_node_list_t *build_block_node_lists(vtx_graph_t *graph,
   473→                                                       vtx_arena_t *arena)
   474→{
   475→    uint32_t block_count = graph->block_count;
   476→    uint32_t node_count = graph->node_table.count;
   477→
   478→    /* Allocate per-block lists */
   479→    vtx_block_node_list_t *lists = vtx_arena_alloc(arena,
   480→        block_count * sizeof(vtx_block_node_list_t));
   481→    if (!lists) return NULL;
   482→    memset(lists, 0, block_count * sizeof(vtx_block_node_list_t));
   483→
   484→    /* First pass: count nodes per block */
   485→    for (uint32_t i = 0; i < node_count; i++) {
   486→        vtx_node_t *node = &graph->node_table.nodes[i];
   487→        if (node->dead) continue;
   488→
   489→        /* Determine which block this node belongs to */
   490→        for (uint32_t b = 0; b < block_count; b++) {
   491→            vtx_block_info_t *block = &graph->blocks[b];
   492→            bool belongs = false;
   493→
   494→            if (node->id == block->region_node ||
   495→                node->id == block->control_node ||
   496→                node->id == block->memory_node) {
   497→                belongs = true;
   498→            } else {
   499→                for (uint32_t j = 0; j < node->input_count; j++) {
   500→                    if (node->inputs[j] == block->region_node ||
   501→                        node->inputs[j] == block->control_node ||
   502→                        node->inputs[j] == block->memory_node) {
   503→                        belongs = true;
   504→                        break;
   505→                    }
   506→                }
   507→                if (!belongs && block->region_node < node_count) {
   508→                    vtx_node_t *region = &graph->node_table.nodes[block->region_node];
   509→                    if (node->bytecode_pc == region->bytecode_pc) {
   510→                        belongs = true;
   511→                    }
   512→                }
   513→            }
   514→
   515→            if (belongs) {
   516→                lists[b].count++;
   517→                break;  /* each node belongs to at most one block */
   518→            }
   519→        }
   520→    }
   521→
   522→    /* Allocate arrays */
   523→    for (uint32_t b = 0; b < block_count; b++) {
   524→        if (lists[b].count > 0) {
   525→            lists[b].nodes = vtx_arena_alloc(arena,
   526→                lists[b].count * sizeof(vtx_nodeid_t));
   527→            if (!lists[b].nodes) return NULL;
   528→            lists[b].count = 0;  /* reset for second pass */
   529→        }
   530→    }
   531→
   532→    /* Second pass: fill arrays */
   533→    for (uint32_t i = 0; i < node_count; i++) {
   534→        vtx_node_t *node = &graph->node_table.nodes[i];
   535→        if (node->dead) continue;
   536→
   537→        for (uint32_t b = 0; b < block_count; b++) {
   538→            vtx_block_info_t *block = &graph->blocks[b];
   539→            bool belongs = false;
   540→
   541→            if (node->id == block->region_node ||
   542→                node->id == block->control_node ||
   543→                node->id == block->memory_node) {
   544→                belongs = true;
   545→            } else {
   546→                for (uint32_t j = 0; j < node->input_count; j++) {
   547→                    if (node->inputs[j] == block->region_node ||
   548→                        node->inputs[j] == block->control_node ||
   549→                        node->inputs[j] == block->memory_node) {
   550→                        belongs = true;
   551→                        break;
   552→                    }
   553→                }
   554→                if (!belongs && block->region_node < node_count) {
   555→                    vtx_node_t *region = &graph->node_table.nodes[block->region_node];
   556→                    if (node->bytecode_pc == region->bytecode_pc) {
   557→                        belongs = true;
   558→                    }
   559→                }
   560→            }
   561→
   562→            if (belongs) {
   563→                lists[b].nodes[lists[b].count++] = node->id;
   564→                break;
   565→            }
   566→        }
   567→    }
   568→
   569→    return lists;
   570→}
   571→
   572→/* ========================================================================== */
   573→/* Internal: fast transfer function using pre-computed block-node lists        */
   574→/* ========================================================================== */
   575→/*
   576→ * §2.1: Uses the pre-computed block_node_lists to iterate only the nodes
   577→ * that belong to this block, instead of scanning all N nodes.
   578→ * GraalVM: EscapeAnalysisPhase uses BlockBag for per-block iteration.
   579→ * V8: EscapeAnalysis uses Schedule::rpo_order for per-block iteration.
   580→ */
   581→static void transfer_block_fast(vtx_graph_t *graph, uint32_t block_idx,
   582→                                 vtx_pea_block_state_t *bs,
   583→                                 vtx_node_table_t *table,
   584→                                 const vtx_block_node_list_t *block_lists)
   585→{
   586→    /* Start from entry state */
   587→    copy_escape_states(bs->exit_state, bs->entry_state, bs->state_count);
   588→
   589→    const vtx_block_node_list_t *list = &block_lists[block_idx];
   590→    for (uint32_t i = 0; i < list->count; i++) {
   591→        vtx_node_t *node = &table->nodes[list->nodes[i]];
   592→        if (node->dead) continue;
   593→        transfer_node(node, table, bs->exit_state, bs->state_count);
   594→    }
   595→}

   597→/* ========================================================================== */
   598→/* Internal: compute reverse postorder for worklist                            */
   599→/* ========================================================================== */
   600→
   601→static uint32_t *compute_reverse_postorder(vtx_graph_t *graph, vtx_arena_t *arena,
   602→                                             uint32_t *rpo_count)
   603→{
   604→    uint32_t n = graph->block_count;
   605→    if (n == 0) {
   606→        *rpo_count = 0;
   607→        return NULL;
   608→    }
   609→
   610→    uint32_t *rpo = vtx_arena_alloc(arena, n * sizeof(uint32_t));
   611→    if (!rpo) return NULL;
   612→
   613→    /* Simple DFS-based reverse postorder computation.
   614→     * We start from block 0 (entry block) and traverse successors. */
   615→    bool *visited = vtx_arena_alloc(arena, n * sizeof(bool));
   616→    uint32_t *stack = vtx_arena_alloc(arena, n * sizeof(uint32_t));
   617→    if (!visited || !stack) return NULL;
   618→
   619→    memset(visited, 0, n * sizeof(bool));
   620→
   621→    uint32_t rpo_idx = n; /* fill from end */
   622→    int32_t top = 0;
   623→    stack[0] = 0;
   624→    visited[0] = true;
   625→
   626→    while (top >= 0) {
   627→        uint32_t current = stack[top];
   628→        bool has_unvisited_succ = false;
   629→
   630→        vtx_block_info_t *block = &graph->blocks[current];
   631→        for (uint32_t i = 0; i < block->succ_count; i++) {
   632→            uint32_t succ = block->succ_indices[i];
   633→            if (succ < n && !visited[succ]) {
   634→                visited[succ] = true;
   635→                top++;
   636→                stack[top] = succ;
   637→                has_unvisited_succ = true;
   638→                break;
   639→            }
   640→        }
   641→
   642→        if (!has_unvisited_succ) {
   643→            rpo_idx--;
   644→            rpo[rpo_idx] = current;
   645→            top--;
   646→        }
   647→    }
   648→
   649→    *rpo_count = n - rpo_idx;
   650→    if (rpo_idx > 0) {
   651→        /* Some blocks were unreachable; shift the array */
   652→        memmove(rpo, rpo + rpo_idx, *rpo_count * sizeof(uint32_t));
   653→    }
   654→
   655→    return rpo;
   656→}

   658→/* ========================================================================== */
   659→/* Main analysis entry point                                                   */
   660→/* ========================================================================== */
   661→
   662→vtx_pea_analysis_t *vtx_pea_run(vtx_graph_t *graph, vtx_arena_t *arena)
   663→{
   664→    VTX_ASSERT(graph != NULL, "graph must not be NULL");
   665→    VTX_ASSERT(arena != NULL, "arena must not be NULL");
   666→
   667→    if (graph->block_count == 0) {
   668→        /* No blocks — degenerate graph */
   669→        return NULL;
   670→    }
   671→
   672→    /* Allocate the result */
   673→    vtx_pea_analysis_t *result = vtx_arena_alloc(arena, sizeof(vtx_pea_analysis_t));
   674→    if (!result) return NULL;
   675→    memset(result, 0, sizeof(*result));
   676→
   677→    /* Step 1: Identify all allocations */
   678→    if (build_allocation_map(graph, &result->escape_map, arena) != 0) {
   679→        return NULL;
   680→    }
   681→
   682→    /* Step 2: Build per-block state arrays */
   683→    if (build_block_states(graph, result, arena) != 0) {
   684→        return NULL;
   685→    }
   686→
   687→    vtx_node_table_t *table = &graph->node_table;
   688→    uint32_t state_count = result->escape_map.state_count;
   689→    uint32_t block_count = graph->block_count;
   690→
   691→    /* Step 3: Compute reverse postorder for efficient worklist */
   692→    uint32_t rpo_count = 0;
   693→    uint32_t *rpo = compute_reverse_postorder(graph, arena, &rpo_count);
   694→
   695→    /* §2.1: Pre-compute block-to-node lists to avoid O(N×B) scan
   696→     * in transfer_block. This is built once before the worklist loop.
   697→     * GraalVM: BlockBag in EscapeAnalysisPhase.
   698→     * V8: Schedule::rpo_order in EscapeAnalysis. */
   699→    vtx_block_node_list_t *block_lists = build_block_node_lists(graph, arena);
   700→    if (!block_lists) return NULL;
   701→
   702→    /* Step 4: Initialize worklist with all blocks in RPO */
   703→    /* We use a simple flag-based worklist: a block is on the worklist
   704→     * if its `entry_changed` flag is true. */
   705→    bool *on_worklist = vtx_arena_alloc(arena, block_count * sizeof(bool));
   706→    if (!on_worklist) return NULL;
   707→    memset(on_worklist, 0, block_count * sizeof(bool));
   708→
   709→    /* Circular buffer worklist */
   710→    uint32_t wl_capacity = block_count + 1;
   711→    uint32_t *worklist = vtx_arena_alloc(arena, wl_capacity * sizeof(uint32_t));
   712→    if (!worklist) return NULL;
   713→
   714→    uint32_t wl_head = 0, wl_tail = 0;
   715→
   716→    /* Enqueue all blocks in RPO */
   717→    for (uint32_t i = 0; i < rpo_count; i++) {
   718→        uint32_t block_idx = rpo[i];
   719→        worklist[wl_tail] = block_idx;
   720→        wl_tail = (wl_tail + 1) % wl_capacity;
   721→        on_worklist[block_idx] = true;
   722→    }
   723→
   724→    /* Step 5: Iterate to fixed point.
   725→     *
   726→     * BUGFIX (audit #4): The old hard cap of 100 iterations was too low
   727→     * for large CFGs. If the worklist hadn't converged by iteration 100,
   728→     * the escape states were left in an intermediate (wrong) state.
   729→     * Allocations that should be GlobalEscape might be marked NoEscape
   730→     * → heap corruption from scalar replacement.
   731→     *
   732→     * Fix: Increase to 10000 (sufficient for any real-world CFG), and
   733→     * if we hit the cap, conservatively mark ALL allocations as
   734→     * GlobalEscape (safe fallback — no scalar replacement). */
   735→    uint32_t iterations = 0;
   736→    const uint32_t MAX_ITERATIONS = 10000; /* safety bound — was 100 */
   737→    bool hit_cap = false;
   738→
   739→    while (wl_head != wl_tail && iterations < MAX_ITERATIONS) {
   740→        iterations++;
   741→
   742→        uint32_t block_idx = worklist[wl_head];
   743→        wl_head = (wl_head + 1) % wl_capacity;
   744→        on_worklist[block_idx] = false;
   745→
   746→        vtx_pea_block_state_t *bs = &result->block_states[block_idx];
   747→        vtx_block_info_t *block = &graph->blocks[block_idx];
   748→
   749→        /* Compute entry state as join of predecessor exit states */
   750→        vtx_escape_state_t *new_entry = vtx_arena_alloc(arena,
   751→            state_count * sizeof(vtx_escape_state_t));
   752→        if (!new_entry) return NULL;
   753→        memset(new_entry, 0, state_count * sizeof(vtx_escape_state_t));
   754→
   755→        if (block->pred_count == 0) {
   756→            /* Entry block: entry state is all NoEscape (already zeroed) */
   757→        } else {
   758→            /* Join all predecessor exit states */
   759→            for (uint32_t p = 0; p < block->pred_count; p++) {
   760→                uint32_t pred_idx = block->pred_indices[p];
   761→                if (pred_idx < block_count) {
   762→                    join_escape_states(new_entry,
   763→                                       result->block_states[pred_idx].exit_state,
   764→                                       state_count);
   765→                }
   766→            }
   767→        }
   768→
   769→        /* Check if entry state changed */
   770→        bool entry_changed = escape_states_differ(new_entry, bs->entry_state,
   771→                                                    state_count);
   772→        if (entry_changed) {
   773→            copy_escape_states(bs->entry_state, new_entry, state_count);
   774→        }
   775→
   776→        /* Apply transfer function */
   777→        vtx_escape_state_t *old_exit = vtx_arena_alloc(arena,
   778→            state_count * sizeof(vtx_escape_state_t));
   779→        if (!old_exit) return NULL;
   780→        copy_escape_states(old_exit, bs->exit_state, state_count);
   781→
   782→        /* §2.1: Use fast transfer with pre-computed block-node lists */
   783→        transfer_block_fast(graph, block_idx, bs, table, block_lists);
   784→
   785→        bool exit_changed = escape_states_differ(bs->exit_state, old_exit,
   786→                                                   state_count);
   787→
   788→        /* If exit state changed, add successors to worklist */
   789→        if (entry_changed || exit_changed) {
   790→            for (uint32_t s = 0; s < block->succ_count; s++) {
   791→                uint32_t succ_idx = block->succ_indices[s];
   792→                if (succ_idx < block_count && !on_worklist[succ_idx]) {
   793→                    worklist[wl_tail] = succ_idx;
   794→                    wl_tail = (wl_tail + 1) % wl_capacity;
   795→                    on_worklist[succ_idx] = true;
   796→                }
   797→            }
   798→        }
   799→    }
   800→
   801→    result->iterations = iterations;
   802→
   803→    /* If we hit the iteration cap, the escape states may not have
   804→     * converged. Conservatively mark ALL allocations as GlobalEscape
   805→     * to prevent incorrect scalar replacement. */
   806→    if (iterations >= MAX_ITERATIONS) {
   807→        hit_cap = true;
   808→        for (uint32_t a = 0; a < result->escape_map.alloc_count; a++) {
   809→            vtx_nodeid_t alloc_id = result->escape_map.alloc_ids[a];
   810→            for (uint32_t b = 0; b < block_count; b++) {
   811→                if (alloc_id < result->block_states[b].state_count) {
   812→                    result->block_states[b].exit_state[alloc_id] = VTX_ESCAPE_GLOBAL;
   813→                    result->block_states[b].entry_state[alloc_id] = VTX_ESCAPE_GLOBAL;
   814→                }
   815→            }
   816→        }
   817→    }
   818→
   819→    /* Step 6: Finalize — compute the global escape state for each allocation
   820→     * as the join of all block exit states */
   821→    for (uint32_t a = 0; a < result->escape_map.alloc_count; a++) {
   822→        vtx_nodeid_t alloc_id = result->escape_map.alloc_ids[a];
   823→        vtx_escape_state_t final_state = VTX_ESCAPE_NONE;
   824→        for (uint32_t b = 0; b < block_count; b++) {
   825→            if (alloc_id < result->block_states[b].state_count) {
   826→                final_state = vtx_escape_join(final_state,
   827→                    result->block_states[b].exit_state[alloc_id]);
   828→            }
   829→        }
   830→
   831→        result->escape_map.states[alloc_id] = final_state;
   832→
   833→        /* Update statistics */
   834→        result->total_allocs++;
   835→        switch (final_state) {
   836→        case VTX_ESCAPE_NONE:   result->no_escape_count++; break;
   837→        case VTX_ESCAPE_ARG:    result->arg_escape_count++; break;
   838→        case VTX_ESCAPE_GLOBAL: result->global_escape_count++; break;
   839→        }
   840→    }
   841→
   842→    return result;
   843→}

   846→/* Query helpers                                                               */
   847→/* ========================================================================== */
   848→
   849→vtx_escape_state_t vtx_pea_get_escape(const vtx_pea_analysis_t *analysis,
   850→                                       vtx_nodeid_t node_id)
   851→{
   852→    VTX_ASSERT(analysis != NULL, "analysis must not be NULL");
   853→    if (node_id < analysis->escape_map.state_count) {
   854→        return analysis->escape_map.states[node_id];
   855→    }
   856→    return VTX_ESCAPE_GLOBAL; /* conservative default */
   857→}
   858→
   859→bool vtx_pea_is_scalar_replaceable(const vtx_pea_analysis_t *analysis,
   860→                                    vtx_nodeid_t node_id)
   861→{
   862→    return vtx_pea_get_escape(analysis, node_id) == VTX_ESCAPE_NONE;
   863→}
   864→
   865→vtx_escape_state_t vtx_pea_block_entry_state(const vtx_pea_analysis_t *analysis,
   866→                                              uint32_t block_idx,
   867→                                              vtx_nodeid_t alloc_id)
   868→{
   869→    VTX_ASSERT(analysis != NULL, "analysis must not be NULL");
   870→    /* PEA-1-2/PEA-1-12: non-allocations always return GlobalEscape
   871→     * (conservative). The per-block entry_state[] is zero-initialized
   872→     * and only allocations get their state updated during dataflow —
   873→     * so reading entry_state[non_alloc_id] would return NoEscape
   874→     * (wrong). Check the global escape map first: if the global state
   875→     * is GlobalEscape, the node is either a non-allocation or an
   876→     * escaping allocation — in both cases, the per-block state is
   877→     * irrelevant (the object escapes regardless). */
   878→    if (vtx_pea_get_escape(analysis, alloc_id) == VTX_ESCAPE_GLOBAL) {
   879→        return VTX_ESCAPE_GLOBAL;
   880→    }
   881→    if (block_idx < analysis->block_state_count &&
   882→        alloc_id < analysis->block_states[block_idx].state_count) {
   883→        return analysis->block_states[block_idx].entry_state[alloc_id];
   884→    }
   885→    return VTX_ESCAPE_GLOBAL;
   886→}
   887→
   888→vtx_escape_state_t vtx_pea_block_exit_state(const vtx_pea_analysis_t *analysis,
   889→                                             uint32_t block_idx,
   890→                                             vtx_nodeid_t alloc_id)
   891→{
   892→    VTX_ASSERT(analysis != NULL, "analysis must not be NULL");
   893→    /* PEA-1-2/PEA-1-12: same as block_entry_state — non-allocations
   894→     * return GlobalEscape. */
   895→    if (vtx_pea_get_escape(analysis, alloc_id) == VTX_ESCAPE_GLOBAL) {
   896→        return VTX_ESCAPE_GLOBAL;
   897→    }
   898→    if (block_idx < analysis->block_state_count &&
   899→        alloc_id < analysis->block_states[block_idx].state_count) {
   900→        return analysis->block_states[block_idx].exit_state[alloc_id];
   901→    }
   902→    return VTX_ESCAPE_GLOBAL;
   903→}
