    25→/**
    26→ * Check if a pointer falls within a memory range [start, start+size).
    27→ */
    28→static inline bool ptr_in_range(const void *ptr, const uint8_t *start, size_t size)
    29→{
    30→    return (const uint8_t *)ptr >= start && (const uint8_t *)ptr < (start + size);
    31→}
    32→
    33→/**
    34→ * Align a pointer up to 8-byte alignment.
    35→ */
    36→static inline size_t align_up_8(size_t value)
    37→{
    38→    return (value + 7) & ~(size_t)7;
    39→}

    41→/**
    42→ * Initialize a semi-space by allocating VTX_GC_YOUNG_SIZE bytes via mmap.
    43→ * Returns 0 on success, -1 on failure.
    44→ */
    45→static int semi_space_init(vtx_semi_space_t *space, size_t size)
    46→{
    47→    space->start = (uint8_t *)mmap(NULL, size, PROT_READ | PROT_WRITE,
    48→                                    MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    49→    if (space->start == MAP_FAILED) {
    50→        space->start = NULL;
    51→        space->current = NULL;
    52→        space->size = 0;
    53→        return -1;
    54→    }
    55→    space->current = space->start;
    56→    space->size = size;
    57→    return 0;
    58→}
    59→
    60→/**
    61→ * Destroy a semi-space by munmap-ing its memory.
    62→ */
    63→static void semi_space_destroy(vtx_semi_space_t *space)
    64→{
    65→    if (space->start != NULL) {
    66→        munmap(space->start, space->size);
    67→        space->start = NULL;
    68→        space->current = NULL;
    69→        space->size = 0;
    70→    }
    71→}
    72→
    73→/**
    74→ * Reset a semi-space for reuse (e.g., to-space after a collection).
    75→ */
    76→static void semi_space_reset(vtx_semi_space_t *space)
    77→{
    78→    space->current = space->start;
    79→}
    80→
    81→/**
    82→ * Allocate from a semi-space using bump-pointer allocation.
    83→ * Returns NULL if there's not enough space.
    84→ */
    85→static void *semi_space_alloc(vtx_semi_space_t *space, size_t size)
    86→{
    87→    size = align_up_8(size);
    88→    if (space->current + size > space->start + space->size) {
    89→        return NULL; /* out of space */
    90→    }
    91→    void *ptr = space->current;
    92→    space->current += size;
    93→    return ptr;
    94→}

   100→/**
   101→ * Initialize the old generation.
   102→ */
   103→static int old_gen_init(vtx_old_gen_t *old, size_t size)
   104→{
   105→    old->start = (uint8_t *)mmap(NULL, size, PROT_READ | PROT_WRITE,
   106→                                  MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
   107→    if (old->start == MAP_FAILED) {
   108→        old->start = NULL;
   109→        old->size = 0;
   110→        old->used = 0;
   111→        old->free_list = NULL;
   112→        return -1;
   113→    }
   114→    old->size = size;
   115→    old->used = 0;
   116→
   117→    /* The entire old gen starts as one big free block */
   118→    old->free_list = (vtx_free_node_t *)old->start;
   119→    vtx_free_node_set_size(old->free_list, size);
   120→    old->free_list->next = NULL;
   121→    old->free_list->gc_mark = 0xFF; /* RT-4: Mark as free block */
   122→
   123→    return 0;
   124→}
   125→
   126→/**
   127→ * Destroy the old generation.
   128→ */
   129→static void old_gen_destroy(vtx_old_gen_t *old)
   130→{
   131→    if (old->start != NULL) {
   132→        munmap(old->start, old->size);
   133→        old->start = NULL;
   134→        old->size = 0;
   135→        old->used = 0;
   136→        old->free_list = NULL;
   137→    }
   138→}
   139→
   140→/**
   141→ * Allocate from the old generation using first-fit free list.
   142→ * Returns NULL if no suitable block is found.
   143→ */
   144→static void *old_gen_alloc(vtx_old_gen_t *old, size_t size)
   145→{
   146→    size = align_up_8(size);
   147→
   148→    /* Minimum allocation size must fit the free list header */
   149→    if (size < sizeof(vtx_free_node_t)) {
   150→        size = sizeof(vtx_free_node_t);
   151→    }
   152→
   153→    vtx_free_node_t *prev = NULL;
   154→    vtx_free_node_t *curr = old->free_list;
   155→
   156→    while (curr != NULL) {
   157→        size_t curr_size = vtx_free_node_get_size(curr);
   158→        if (curr_size >= size) {
   159→            /* Found a suitable block */
   160→            if (curr_size >= size + sizeof(vtx_free_node_t)) {
   161→                /* Split the block */
   162→                vtx_free_node_t *remainder = (vtx_free_node_t *)((uint8_t *)curr + size);
   163→                vtx_free_node_set_size(remainder, curr_size - size);
   164→                remainder->next = curr->next;
   165→
   166→                if (prev != NULL) {
   167→                    prev->next = remainder;
   168→                } else {
   169→                    old->free_list = remainder;
   170→                }
   171→            } else {
   172→                /* Use the entire block (may be slightly larger than requested) */
   173→                size = curr_size;
   174→                if (prev != NULL) {
   175→                    prev->next = curr->next;
   176→                } else {
   177→                    old->free_list = curr->next;
   178→                }
   179→            }
   180→
   181→            old->used += size;
   182→            return (void *)curr;
   183→        }
   184→        prev = curr;
   185→        curr = curr->next;
   186→    }
   187→
   188→    /* No suitable block found */
   189→    return NULL;
   190→}
   191→
   192→/**
   193→ * Add a block back to the old generation free list.
   194→ * Simplest approach: prepend to free list. In a production system,
   195→ * we would coalesce adjacent blocks.
   196→ */
   197→/**
   198→ * Remove a free node from the old-gen free list.
   199→ * Returns true if found and removed, false otherwise.
   200→ */
   201→static bool free_list_remove(vtx_old_gen_t *old, vtx_free_node_t *target)
   202→{
   203→    vtx_free_node_t *prev = NULL;
   204→    vtx_free_node_t *curr = old->free_list;
   205→    while (curr != NULL) {
   206→        if (curr == target) {
   207→            if (prev != NULL) {
   208→                prev->next = curr->next;
   209→            } else {
   210→                old->free_list = curr->next;
   211→            }
   212→            return true;
   213→        }
   214→        prev = curr;
   215→        curr = curr->next;
   216→    }
   217→    return false;
   218→}

   220→static void old_gen_free(vtx_old_gen_t *old, void *ptr, size_t size)
   221→{
   222→    size = align_up_8(size);
   223→    if (size < sizeof(vtx_free_node_t)) {
   224→        size = sizeof(vtx_free_node_t);
   225→    }
   226→
   227→    vtx_free_node_t *node = (vtx_free_node_t *)ptr;
   228→    vtx_free_node_set_size(node, size);
   229→    node->next = old->free_list;
   230→    node->gc_mark = 0xFF; /* RT-4: Mark as free block so sweep can identify it */
   231→    old->free_list = node;
   232→    old->used -= size;
   233→
   234→    /* Coalesce with adjacent free blocks — both forward and backward.
   235→     *
   236→     * BUGFIX: The original code only coalesced forward (merged a free block
   237→     * that starts immediately after the newly freed block). It never checked
   238→     * for a backward adjacent block (a free block whose end is exactly at the
   239→     * start of the newly freed block). This caused fragmentation when blocks
   240→     * were freed in reverse address order: two adjacent free blocks would
   241→     * remain separate, preventing the allocator from satisfying larger
   242→     * allocation requests that span both blocks. */
   243→
   244→    /* Forward coalescing: look for a free block that starts immediately
   245→     * after the newly freed block ends. */
   246→    bool merged = true;
   247→    while (merged) {
   248→        merged = false;
   249→        size_t node_size = vtx_free_node_get_size(node);
   250→        vtx_free_node_t *curr = old->free_list;
   251→        while (curr != NULL) {
   252→            if (curr != node &&
   253→                (uint8_t *)curr == (uint8_t *)node + node_size) {
   254→                /* curr is the forward adjacent block — merge into node */
   255→                vtx_free_node_set_size(node, node_size + vtx_free_node_get_size(curr));
   256→                free_list_remove(old, curr);
   257→                merged = true;
   258→                break;
   259→            }
   260→            curr = curr->next;
   261→        }
   262→    }
   263→
   264→    /* Backward coalescing: look for a free block whose end is exactly at
   265→     * the start of the newly freed block. */
   266→    merged = true;
   267→    while (merged) {
   268→        merged = false;
   269→        vtx_free_node_t *curr = old->free_list;
   270→        while (curr != NULL) {
   271→            if (curr != node) {
   272→                size_t curr_size = vtx_free_node_get_size(curr);
   273→                if ((uint8_t *)node == (uint8_t *)curr + curr_size) {
   274→                    /* curr is the backward adjacent block — merge node
   275→                     * into curr, expanding curr to cover both. */
   276→                    vtx_free_node_set_size(curr, curr_size + vtx_free_node_get_size(node));
   277→                    /* Remove node from the free list since it's now
   278→                     * part of curr. Set curr as the new head if node
   279→                     * was the head. */
   280→                    if (old->free_list == node) {
   281→                        old->free_list = node->next;
   282→                    } else {
   283→                        free_list_remove(old, node);
   284→                    }
   285→                    /* node is gone; curr is now the representative block */
   286→                    node = curr;
   287→                    merged = true;
   288→                    break;
   289→                }
   290→            }
   291→            curr = curr->next;
   292→        }
   293→    }
   294→}

   300→int vtx_gc_init(vtx_gc_t *gc, vtx_type_system_t *ts, vtx_gc_mode_t mode)
   301→{
   302→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   303→    VTX_ASSERT(ts != NULL, "type system must not be NULL");
   304→
   305→    memset(gc, 0, sizeof(vtx_gc_t));
   306→
   307→    /* Initialize young generation semi-spaces */
   308→    if (semi_space_init(&gc->young_from, VTX_GC_YOUNG_SIZE) != 0) {
   309→        return -1;
   310→    }
   311→    if (semi_space_init(&gc->young_to, VTX_GC_YOUNG_SIZE) != 0) {
   312→        semi_space_destroy(&gc->young_from);
   313→        return -1;
   314→    }
   315→
   316→    /* Initialize old generation */
   317→    if (old_gen_init(&gc->old_gen, VTX_GC_OLD_SIZE) != 0) {
   318→        semi_space_destroy(&gc->young_from);
   319→        semi_space_destroy(&gc->young_to);
   320→        return -1;
   321→    }
   322→
   323→    /* Initialize root stack */
   324→    gc->root_capacity = VTX_GC_ROOT_STACK_SIZE;
   325→    gc->root_stack = (vtx_root_entry_t *)malloc(
   326→        gc->root_capacity * sizeof(vtx_root_entry_t));
   327→    if (gc->root_stack == NULL) {
   328→        semi_space_destroy(&gc->young_from);
   329→        semi_space_destroy(&gc->young_to);
   330→        old_gen_destroy(&gc->old_gen);
   331→        return -1;
   332→    }
   333→    gc->root_count = 0;
   334→
   335→    /* Initialize remembered set */
   336→    gc->remembered_capacity = VTX_GC_REMEMBERED_SET_INIT;
   337→    gc->remembered_set = (vtx_remembered_entry_t *)malloc(
   338→        gc->remembered_capacity * sizeof(vtx_remembered_entry_t));
   339→    if (gc->remembered_set == NULL) {
   340→        semi_space_destroy(&gc->young_from);
   341→        semi_space_destroy(&gc->young_to);
   342→        old_gen_destroy(&gc->old_gen);
   343→        free(gc->root_stack);
   344→        return -1;
   345→    }
   346→    gc->remembered_count = 0;
   347→
   348→    /* Initialize pinned objects list */
   349→    gc->pinned_capacity = 64;
   350→    gc->pinned_objects = (vtx_heap_object_t **)malloc(
   351→        gc->pinned_capacity * sizeof(vtx_heap_object_t *));
   352→    if (gc->pinned_objects == NULL) {
   353→        semi_space_destroy(&gc->young_from);
   354→        semi_space_destroy(&gc->young_to);
   355→        old_gen_destroy(&gc->old_gen);
   356→        free(gc->root_stack);
   357→        free(gc->remembered_set);
   358→        return -1;
   359→    }
   360→    gc->pinned_count = 0;
   361→
   362→    gc->type_system = ts;
   363→    gc->collection_requested = false;
   364→    gc->collections_done = 0;
   365→    gc->mode = mode;
   366→
   367→    /* Initialize card table covering the entire heap.
   368→     * The card table covers from the minimum heap address (old gen start)
   369→     * through the end of old gen. Young-gen objects don't need card marking
   370→     * (they're collected every GC cycle), but old-gen objects do.
   371→     * We compute heap_base as the minimum of old gen and young gen starts,
   372→     * and heap_size to cover all heap regions. */
   373→    {
   374→        /* Find the minimum heap address to use as heap_base */
   375→        uint8_t *min_addr = gc->old_gen.start;
   376→        if (gc->young_from.start < min_addr) min_addr = gc->young_from.start;
   377→        if (gc->young_to.start < min_addr) min_addr = gc->young_to.start;
   378→
   379→        /* Find the maximum end address */
   380→        uint8_t *max_end = gc->old_gen.start + gc->old_gen.size;
   381→        if (gc->young_from.start + gc->young_from.size > max_end)
   382→            max_end = gc->young_from.start + gc->young_from.size;
   383→        if (gc->young_to.start + gc->young_to.size > max_end)
   384→            max_end = gc->young_to.start + gc->young_to.size;
   385→
   386→        gc->heap_base = min_addr;
   387→        gc->heap_size = (size_t)(max_end - min_addr);
   388→        gc->card_table_size = (gc->heap_size + VTX_CARD_SIZE - 1) / VTX_CARD_SIZE;
   389→        gc->card_table = (uint8_t *)calloc(gc->card_table_size, 1);
   390→        if (gc->card_table == NULL) {
   391→            /* Card table allocation failed — not fatal, we fall back
   392→             * to remembered-set-only barrier */
   393→            gc->card_table_size = 0;
   394→            gc->heap_base = NULL;
   395→            gc->heap_size = 0;
   396→        }
   397→    }
   398→
   399→    vtx_gc_set_mode(gc, mode);
   400→
   401→    return 0;
   402→}
   403→
   404→void vtx_gc_set_mode(vtx_gc_t *gc, vtx_gc_mode_t mode)
   405→{
   406→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   407→    gc->mode = mode;
   408→
   409→    switch (mode) {
   410→    case VTX_GC_GENERATIONAL:
   411→        gc->fn_write_barrier = gc->card_table != NULL
   412→            ? vtx_gc_write_barrier_card
   413→            : vtx_gc_write_barrier;
   414→        gc->fn_safepoint = vtx_gc_safepoint;
   415→        gc->fn_root_push = vtx_gc_root_push;
   416→        gc->fn_root_pop = vtx_gc_root_pop;
   417→        break;
   418→    case VTX_GC_NONE:
   419→        gc->fn_write_barrier = NULL;
   420→        gc->fn_safepoint = NULL;
   421→        gc->fn_root_push = NULL;
   422→        gc->fn_root_pop = NULL;
   423→        break;
   424→    case VTX_GC_MANUAL:
   425→        gc->fn_write_barrier = NULL;
   426→        gc->fn_safepoint = NULL;
   427→        gc->fn_root_push = vtx_gc_root_push;
   428→        gc->fn_root_pop = vtx_gc_root_pop;
   429→        gc->manual_free_list = NULL;
   430→        break;
   431→    case VTX_GC_ARENA:
   432→        gc->fn_write_barrier = NULL;
   433→        gc->fn_safepoint = NULL;
   434→        gc->fn_root_push = vtx_gc_root_push;
   435→        gc->fn_root_pop = vtx_gc_root_pop;
   436→        gc->arena_save_point = gc->young_from.current;
   437→        break;
   438→    }
   439→}

   447→void vtx_gc_manual_free(vtx_gc_t *gc, void *ptr, size_t size)
   448→{
   449→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   450→    VTX_ASSERT(gc->mode == VTX_GC_MANUAL, "manual_free only valid in manual mode");
   451→    if (gc->mode != VTX_GC_MANUAL) return;
   452→
   453→    size = align_up_8(size);
   454→    if (size < sizeof(vtx_free_node_t)) {
   455→        size = sizeof(vtx_free_node_t);
   456→    }
   457→    vtx_free_node_t *node = (vtx_free_node_t *)ptr;
   458→    vtx_free_node_set_size(node, size);
   459→    node->next = gc->manual_free_list;
   460→    node->gc_mark = 0xFF; /* RT-4: Mark as free block */
   461→    gc->manual_free_list = node;
   462→}

   478→void vtx_gc_destroy(vtx_gc_t *gc)
   479→{
   480→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   481→
   482→    semi_space_destroy(&gc->young_from);
   483→    semi_space_destroy(&gc->young_to);
   484→    old_gen_destroy(&gc->old_gen);
   485→
   486→    free(gc->root_stack);
   487→    gc->root_stack = NULL;
   488→
   489→    free(gc->remembered_set);
   490→    gc->remembered_set = NULL;
   491→
   492→    free(gc->pinned_objects);
   493→    gc->pinned_objects = NULL;
   494→
   495→    free(gc->card_table);
   496→    gc->card_table = NULL;
   497→    gc->card_table_size = 0;
   498→    gc->heap_base = NULL;
   499→    gc->heap_size = 0;
   500→
   501→    gc->root_count = 0;
   502→    gc->remembered_count = 0;
   503→    gc->pinned_count = 0;
   504→}

   510→vtx_heap_object_t *vtx_gc_alloc(vtx_gc_t *gc, size_t size, vtx_typeid_t type_id)
   511→{
   512→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   513→    VTX_ASSERT(size >= VTX_HEAP_OBJECT_HEADER_SIZE, "allocation too small for header");
   514→
   515→    size = align_up_8(size);
   516→
   517→    /* Try young gen first */
   518→    void *ptr = semi_space_alloc(&gc->young_from, size);
   519→    if (ptr == NULL) {
   520→        /* Young gen is full — request a collection */
   521→        gc->collection_requested = true;
   522→        vtx_gc_safepoint(gc);
   523→
   524→        /* Try again after collection */
   525→        ptr = semi_space_alloc(&gc->young_from, size);
   526→        if (ptr == NULL) {
   527→            /* Still no space — allocation failure */
   528→            return NULL;
   529→        }
   530→    }
   531→
   532→    vtx_heap_object_t *obj = (vtx_heap_object_t *)ptr;
   533→
   534→    /* Initialize the header */
   535→    const vtx_type_desc_t *td = vtx_type_get(gc->type_system, type_id);
   536→    uint32_t shape_id = td != NULL ? td->shape_id : VTX_SHAPE_INVALID;
   537→    uint32_t field_count = td != NULL ?
   538→        (uint32_t)((size - VTX_HEAP_OBJECT_HEADER_SIZE) / sizeof(vtx_value_t)) : 0;
   539→
   540→    vtx_heap_object_init(obj, type_id, shape_id, field_count, (uint32_t)size);
   541→
   542→    return obj;
   543→}
   544→
   545→void vtx_gc_safepoint(vtx_gc_t *gc)
   546→{
   547→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   548→
   549→    if (gc->collection_requested) {
   550→        gc->collection_requested = false;
   551→
   552→        /* If a safepoint manager is wired, request all threads to safepoint
   553→         * before collecting. This ensures JIT-compiled code is not modifying
   554→         * the heap during collection. After collection, release all threads. */
   555→        if (gc->safepoint_mgr != NULL) {
   556→            vtx_safepoint_manager_t *mgr =
   557→                (vtx_safepoint_manager_t *)gc->safepoint_mgr;
   558→            vtx_safepoint_request_all(mgr);
   559→        }
   560→
   561→        vtx_gc_collect_young(gc);
   562→
   563→        if (gc->safepoint_mgr != NULL) {
   564→            vtx_safepoint_manager_t *mgr =
   565→                (vtx_safepoint_manager_t *)gc->safepoint_mgr;
   566→            vtx_safepoint_release_all(mgr);
   567→        }
   568→    }
   569→}

   571→/* ========================================================================== */
   572→/* Root management                                                             */
   573→/* ========================================================================== */
   574→
   575→void vtx_gc_root_push(vtx_gc_t *gc, vtx_value_t value)
   576→{
   577→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   578→
   579→    if (gc->root_count >= gc->root_capacity) {
   580→        /* Grow the root stack */
   581→        uint32_t new_cap = gc->root_capacity > 0 ? gc->root_capacity * 2 : 16;
   582→        if (new_cap <= gc->root_capacity) new_cap = gc->root_capacity + 16; /* overflow guard */
   583→        vtx_root_entry_t *new_stack = (vtx_root_entry_t *)realloc(
   584→            gc->root_stack, new_cap * sizeof(vtx_root_entry_t));
   585→        if (new_stack == NULL) {
   586→            VTX_ASSERT(false, "root stack overflow — out of memory");
   587→            return;
   588→        }
   589→        gc->root_stack = new_stack;
   590→        gc->root_capacity = new_cap;
   591→    }
   592→
   593→    gc->root_stack[gc->root_count].value = value;
   594→    gc->root_count++;
   595→}
   596→
   597→vtx_value_t vtx_gc_root_pop(vtx_gc_t *gc)
   598→{
   599→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   600→    VTX_ASSERT(gc->root_count > 0, "root stack underflow");
   601→
   602→    gc->root_count--;
   603→    return gc->root_stack[gc->root_count].value;
   604→}

   606→/* ========================================================================== */
   607→/* Write barrier                                                               */
   608→/* ========================================================================== */
   609→
   610→void vtx_gc_write_barrier(vtx_gc_t *gc, vtx_heap_object_t *obj,
   611→                          uint32_t field_offset, vtx_value_t value)
   612→{
   613→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   614→    VTX_ASSERT(obj != NULL, "object must not be NULL");
   615→    (void)field_offset; /* parameter kept for API compatibility; may be used for card marking */
   616→
   617→    /* Only care about old→young references */
   618→    if (!vtx_gc_in_old(gc, obj)) {
   619→        return; /* obj is in young gen, no barrier needed */
   620→    }
   621→
   622→    if (!vtx_is_heap_ptr(value)) {
   623→        return; /* value is not a heap pointer, no barrier needed */
   624→    }
   625→
   626→    void *val_ptr = vtx_heap_ptr(value);
   627→    if (!vtx_gc_in_young(gc, val_ptr)) {
   628→        return; /* value is not in young gen, no barrier needed */
   629→    }
   630→
   631→    /* Add obj to remembered set */
   632→    if (obj->gc_remembered) {
   633→        return; /* already in remembered set */
   634→    }
   635→
   636→    /* Grow remembered set if needed */
   637→    if (gc->remembered_count >= gc->remembered_capacity) {
   638→        uint32_t new_cap = gc->remembered_capacity > 0 ? gc->remembered_capacity * 2 : 16;
   639→        if (new_cap <= gc->remembered_capacity) new_cap = gc->remembered_capacity + 16; /* overflow guard */
   640→        vtx_remembered_entry_t *new_set = (vtx_remembered_entry_t *)realloc(
   641→            gc->remembered_set, new_cap * sizeof(vtx_remembered_entry_t));
   642→        if (new_set == NULL) {
   643→            VTX_ASSERT(false, "remembered set overflow — out of memory");
   644→            return;
   645→        }
   646→        gc->remembered_set = new_set;
   647→        gc->remembered_capacity = new_cap;
   648→    }
   649→
   650→    gc->remembered_set[gc->remembered_count].obj = obj;
   651→    gc->remembered_count++;
   652→    obj->gc_remembered = 1;
   653→}
   654→
   655→void vtx_gc_write_barrier_card(vtx_gc_t *gc, vtx_heap_object_t *obj,
   656→                               uint32_t field_offset, vtx_value_t value)
   657→{
   658→    /* Card marking write barrier: ~3 instructions for the common case.
   659→     *
   660→     * Algorithm:
   661→     * 1. If value is not a young-gen pointer → return (no barrier needed)
   662→     * 2. Compute card index from object address: (obj - heap_base) >> CARD_SHIFT
   663→     * 3. Mark the card as dirty: card_table[index] = CARD_DIRTY
   664→     *
   665→     * This replaces the old 10-branch barrier with:
   666→     *   - 1 branch (is value young-gen?)
   667→     *   - 1 subtraction + shift (compute card index)
   668→     *   - 1 byte store (mark card dirty)
   669→     *
   670→     * Total: ~3 instructions on the fast path (value is young-gen pointer)
   671→     *        ~1 instruction on the super-fast path (value is not young-gen)
   672→     */
   673→    (void)field_offset;
   674→
   675→    /* Fast check: is the value a heap pointer at all? */
   676→    if (!vtx_is_heap_ptr(value)) return;
   677→
   678→    void *value_ptr = vtx_heap_ptr(value);
   679→
   680→    /* Check if value is in young generation */
   681→    if (!vtx_gc_in_young(gc, value_ptr)) return;
   682→
   683→    /* Object must be in old gen for the barrier to matter */
   684→    if (!vtx_gc_in_old(gc, obj)) return;
   685→
   686→    /* Mark the card as dirty */
   687→    if (gc->card_table != NULL && gc->heap_base != NULL) {
   688→        size_t card_index = ((size_t)((uint8_t *)obj - gc->heap_base)) >> VTX_CARD_SHIFT;
   689→        if (card_index < gc->card_table_size) {
   690→            gc->card_table[card_index] = VTX_CARD_DIRTY;
   691→        }
   692→    }
   693→
   694→    /* Also maintain the remembered set as a fallback for correctness.
   695→     * This ensures the old remembered-set scanning path still works
   696→     * if needed, and provides a complete list of old→young references
   697→     * for the remembered-set rebuild after collection. */
   698→    if (obj->gc_remembered) {
   699→        return; /* already in remembered set */
   700→    }
   701→
   702→    if (gc->remembered_count >= gc->remembered_capacity) {
   703→        uint32_t new_cap = gc->remembered_capacity > 0 ? gc->remembered_capacity * 2 : 16;
   704→        if (new_cap <= gc->remembered_capacity) new_cap = gc->remembered_capacity + 16; /* overflow guard */
   705→        vtx_remembered_entry_t *new_set = (vtx_remembered_entry_t *)realloc(
   706→            gc->remembered_set, new_cap * sizeof(vtx_remembered_entry_t));
   707→        if (new_set == NULL) {
   708→            VTX_ASSERT(false, "remembered set overflow — out of memory");
   709→            return;
   710→        }
   711→        gc->remembered_set = new_set;
   712→        gc->remembered_capacity = new_cap;
   713→    }
   714→
   715→    gc->remembered_set[gc->remembered_count].obj = obj;
   716→    gc->remembered_count++;
   717→    obj->gc_remembered = 1;
   718→}
   719→
   720→/* ========================================================================== */
   721→/* Card-table write barrier (JIT-called helper)                                */
   722→/* ========================================================================== */
   723→
   724→/**
   725→ * Card-table based write barrier for JIT-compiled code.
   726→ *
   727→ * This is a simplified card-table barrier that marks the card containing
   728→ * the field at [obj + field_offset] as dirty (0xFF). It is designed to
   729→ * be called from the vtx_helpers_write_barrier() runtime helper, which
   730→ * is the entry point for JIT-compiled code.
   731→ *
   732→ * Card table design:
   733→ *   - Heap is divided into 512-byte cards (VTX_CARD_SIZE)
   734→ *   - Each card has a 1-byte entry in the card table
   735→ *   - On write barrier: compute card index = (field_addr - heap_base) >> 9
   736→ *   - Mark card dirty: card_table[card_index] = 0xFF
   737→ *
   738→ * The 0xFF dirty marker (instead of VTX_CARD_DIRTY = 0x01) allows the
   739→ * lower bits to encode additional metadata when the card is clean, and
   740→ * matches the HotSpot JVM convention where 0xFF means "definitely dirty."
   741→ */
   742→void vtx_gc_card_mark_dirty(vtx_gc_t *gc, const void *field_addr)
   743→{
   744→    if (gc == NULL || gc->card_table == NULL || gc->heap_base == NULL) return;
   745→
   746→    uintptr_t offset = (uintptr_t)((const uint8_t *)field_addr - gc->heap_base);
   747→    if (offset >= gc->heap_size) return;
   748→
   749→    size_t card_index = offset >> VTX_CARD_SHIFT;
   750→    if (card_index < gc->card_table_size) {
   751→        gc->card_table[card_index] = 0xFF;
   752→    }
   753→}
   754→
   755→/* ========================================================================== */
   756→/* Young-gen collection (copying collector)                                    */
   757→/* ========================================================================== */
   758→
   759→/**
   760→ * Forward (copy) an object from from-space to to-space.
   761→ * Returns the new location of the object in to-space.
   762→ * If the object has already been forwarded, returns the forwarding address.
   763→ */
   764→/* Sentinel value for size field indicating the object has been forwarded.
   765→ * No valid object can have this size because it's astronomically large.
   766→ * The forwarding address is stored in the first sizeof(uintptr_t) bytes
   767→ * of the object (overwriting type_id + gc_* fields), which is safe because
   768→ * the object's data has already been copied to the new location. */
   769→#define VTX_GC_FORWARDING_SENTINEL ((uint32_t)0xDEADF00D)
   770→
   771→static vtx_heap_object_t *forward_object(vtx_gc_t *gc, vtx_heap_object_t *obj)
   772→{
   773→    VTX_ASSERT(obj != NULL, "object must not be NULL");
   774→
   775→    /* RT-1 fix: Check if this object has already been forwarded.
   776→     * OLD BUG: Used `first_word & 1` as the forwarding pointer tag, but on
   777→     * little-endian x86-64, any object with an odd type_id (e.g.
   778→     * VTX_TYPE_OBJECT=1) sets bit 0 of the first uintptr_t, causing the GC
   779→     * to misidentify LIVE objects as forwarding pointers → wild pointers,
   780→     * memory corruption, segfaults.
   781→     *
   782→     * NEW approach: Use a sentinel value in the size field
   783→     * (VTX_GC_FORWARDING_SENTINEL = 0xDEADF00D). No valid object can have
   784→     * this size. The forwarding address is stored in the first 8 bytes of
   785→     * the object (overwriting type_id + gc_* byte fields). This is safe
   786→     * because the object has already been copied to its new location. */
   787→    if (obj->size == VTX_GC_FORWARDING_SENTINEL) {
   788→        /* This is a forwarding pointer — address is in first 8 bytes */
   789→        uintptr_t fwd_addr = *(uintptr_t *)obj;
   790→        return (vtx_heap_object_t *)fwd_addr;
   791→    }
   792→
   793→    /* Check if the object is pinned — don't move it */
   794→    if (obj->gc_pinned) {
   795→        return obj;
   796→    }
   797→
   798→    /* Check if the object should be promoted to old gen */
   799→    if (obj->gc_age >= VTX_GC_PROMOTION_AGE) {
   800→        /* Promote to old gen */
   801→        void *new_ptr = old_gen_alloc(&gc->old_gen, obj->size);
   802→        if (new_ptr != NULL) {
   803→            memcpy(new_ptr, obj, obj->size);
   804→            vtx_heap_object_t *new_obj = (vtx_heap_object_t *)new_ptr;
   805→
   806→            /* Leave a forwarding pointer in the old location.
   807→             * RT-1 fix: Store the new address in the first 8 bytes
   808→             * and set size to the sentinel value. */
   809→            /* RT-006 fix: use memcpy instead of *(uintptr_t *)obj = ... to avoid
   810→             * C17 strict-aliasing UB (6.5p7). The old code wrote through a
   811→             * uintptr_t * lvalue while obj is vtx_heap_object_t *, which
   812→             * compilers may optimize away or reorder under -O2+. */
   813→            uintptr_t _fwd_ptr = (uintptr_t)new_obj;
   814→            memcpy(obj, &_fwd_ptr, sizeof(uintptr_t));
   815→            obj->size = VTX_GC_FORWARDING_SENTINEL;
   816→
   817→            return new_obj;
   818→        }
   819→        /* If old gen is full, fall through to copy to to-space */
   820→    }
   821→
   822→    /* Copy to to-space */
   823→    void *new_ptr = semi_space_alloc(&gc->young_to, obj->size);
   824→    if (new_ptr == NULL) {
   825→        /* to-space is full — this is a fatal error in a simple implementation */
   826→        VTX_ASSERT(false, "young gen to-space overflow during collection");
   827→        return obj;
   828→    }
   829→
   830→    memcpy(new_ptr, obj, obj->size);
   831→    vtx_heap_object_t *new_obj = (vtx_heap_object_t *)new_ptr;
   832→    new_obj->gc_age = obj->gc_age + 1;
   833→
   834→    /* Leave a forwarding pointer in the old location.
   835→     * RT-1 fix: Store the new address in the first 8 bytes
   836→     * and set size to the sentinel value. */
   837→    /* RT-006 fix: use memcpy instead of *(uintptr_t *)obj = ... to avoid
   838→             * C17 strict-aliasing UB (6.5p7). The old code wrote through a
   839→             * uintptr_t * lvalue while obj is vtx_heap_object_t *, which
   840→             * compilers may optimize away or reorder under -O2+. */
   841→            uintptr_t _fwd_ptr = (uintptr_t)new_obj;
   842→            memcpy(obj, &_fwd_ptr, sizeof(uintptr_t));
   843→    obj->size = VTX_GC_FORWARDING_SENTINEL;
   844→
   845→    return new_obj;
   846→}
   847→
   848→/**
   849→ * Trace a value: if it's a heap pointer in young gen, forward it
   850→ * and return the updated value.
   851→ */
   852→static vtx_value_t trace_value(vtx_gc_t *gc, vtx_value_t value)
   853→{
   854→    if (!vtx_is_heap_ptr(value)) {
   855→        return value;
   856→    }
   857→
   858→    vtx_heap_object_t *obj = (vtx_heap_object_t *)vtx_heap_ptr(value);
   859→
   860→    /* Only trace objects in the from-space */
   861→    if (!vtx_gc_in_young(gc, obj)) {
   862→        return value;
   863→    }
   864→
   865→    vtx_heap_object_t *new_obj = forward_object(gc, obj);
   866→    return vtx_make_heap_ptr(new_obj);
   867→}
   868→
   869→/**
   870→ * Scan an object's fields and trace any heap pointers.
   871→ */
   872→static void scan_object(vtx_gc_t *gc, vtx_heap_object_t *obj)
   873→{
   874→    VTX_ASSERT(obj != NULL, "object must not be NULL");
   875→
   876→    for (uint32_t i = 0; i < obj->field_count; i++) {
   877→        vtx_value_t field = obj->fields[i];
   878→        if (vtx_is_heap_ptr(field)) {
   879→            vtx_value_t new_field = trace_value(gc, field);
   880→            obj->fields[i] = new_field;
   881→
   882→            /* If the field was updated and this is an old-gen object,
   883→             * we need the write barrier */
   884→            if (vtx_gc_in_old(gc, obj) && vtx_is_heap_ptr(new_field)) {
   885→                void *new_ptr = vtx_heap_ptr(new_field);
   886→                if (vtx_gc_in_young(gc, new_ptr)) {
   887→                    vtx_gc_write_barrier(gc, obj, i, new_field);
   888→                }
   889→            }
   890→        }
   891→    }
   892→}
   893→
   894→/**
   895→ * Process the to-space scan pointer: scan all copied objects.
   896→ */
   897→static void scan_to_space(vtx_gc_t *gc)
   898→{
   899→    /* We scan from the beginning of to-space up to the current pointer.
   900→     * New objects may be added during scanning, so we keep scanning
   901→     * until we catch up to the allocation pointer. */
   902→    uint8_t *scan_ptr = gc->young_to.start;
   903→
   904→    while (scan_ptr < gc->young_to.current) {
   905→        vtx_heap_object_t *obj = (vtx_heap_object_t *)scan_ptr;
   906→        if (obj->size == 0) {
   907→            /* RT-008 fix: the old code did `break`, which stopped
   908→             * scanning the entire to-space when ANY object had size 0.
   909→             * This broke the transitive closure — live objects
   910→             * referenced only by unscanned objects were collected.
   911→             * Fix: advance by the header size and continue scanning. */
   912→            scan_ptr += sizeof(vtx_heap_object_t);
   913→            continue;
   914→        }
   915→        scan_object(gc, obj);
   916→        scan_ptr += align_up_8(obj->size);
   917→    }
   918→}
   919→
   920→/**
   921→ * Scan dirty cards in the card table for old→young references.
   922→ * For each dirty card, scan all objects in that card's region.
   923→ * This replaces scanning the entire remembered set with a targeted
   924→ * scan of only the cards that have been marked dirty by write barriers.
   925→ *
   926→ * After scanning, all dirty cards are cleared.
   927→ */
   928→static void scan_dirty_cards(vtx_gc_t *gc)
   929→{
   930→    if (gc->card_table == NULL || gc->heap_base == NULL) {
   931→        return; /* no card table available */
   932→    }
   933→
   934→    /* Iterate over all cards in the card table */
   935→    for (size_t i = 0; i < gc->card_table_size; i++) {
   936→        if (gc->card_table[i] == VTX_CARD_CLEAN) {
   937→            continue;
   938→        }
   939→
   940→        /* This card is dirty — compute the address range it covers */
   941→        uint8_t *card_start = gc->heap_base + (i << VTX_CARD_SHIFT);
   942→        uint8_t *card_end   = card_start + VTX_CARD_SIZE;
   943→
   944→        /* Only scan within old gen (young-gen objects are traced directly) */
   945→        uint8_t *old_end = gc->old_gen.start + gc->old_gen.size;
   946→        if (card_start < gc->old_gen.start) {
   947→            card_start = gc->old_gen.start;
   948→        }
   949→        if (card_end > old_end) {
   950→            card_end = old_end;
   951→        }
   952→
   953→        /* Scan objects within this card's address range */
   954→        uint8_t *ptr = card_start;
   955→        while (ptr < card_end) {
   956→            /* Check if this is a free block */
   957→            vtx_heap_object_t *obj = (vtx_heap_object_t *)ptr;
   958→            if (obj->gc_mark == 0xFF) {
   959→                /* Free block — skip it */
   960→                vtx_free_node_t *free_node = (vtx_free_node_t *)ptr;
   961→                size_t free_size = align_up_8(vtx_free_node_get_size(free_node));
   962→                if (free_size < sizeof(vtx_free_node_t)) {
   963→                    free_size = sizeof(vtx_free_node_t);
   964→                }
   965→                ptr += free_size;
   966→                continue;
   967→            }
   968→
   969→            if (obj->size == 0 || obj->size >= VTX_GC_FORWARDING_SENTINEL) {
   970→                /* Uninitialized/corrupted block — skip minimum unit */
   971→                ptr += sizeof(vtx_free_node_t);
   972→                continue;
   973→            }
   974→
   975→            /* Scan this object's fields for young-gen pointers */
   976→            scan_object(gc, obj);
   977→
   978→            ptr += align_up_8(obj->size);
   979→        }
   980→
   981→        /* Clear the card after scanning */
   982→        gc->card_table[i] = VTX_CARD_CLEAN;
   983→    }
   984→}
   985→
   986→void vtx_gc_collect_young(vtx_gc_t *gc)
   987→{
   988→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
   989→
   990→    /* Reset to-space */
   991→    semi_space_reset(&gc->young_to);
   992→
   993→    /* Phase 1: Trace roots */
   994→    for (uint32_t i = 0; i < gc->root_count; i++) {
   995→        gc->root_stack[i].value = trace_value(gc, gc->root_stack[i].value);
   996→    }
   997→
   998→    /* Phase 1b: Scan JIT-compiled frames on the native stack for GC roots.
   999→     * The JIT root scan callback walks the stack and calls vtx_gc_root_push
  1000→     * for each root it finds. Those roots are then traced in Phase 3. */
  1001→    if (gc->jit_root_scan_fn != NULL) {
  1002→        gc->jit_root_scan_fn(gc);
  1003→        /* Trace any new roots pushed by the JIT scan */
  1004→        for (uint32_t i = 0; i < gc->root_count; i++) {
  1005→            gc->root_stack[i].value = trace_value(gc, gc->root_stack[i].value);
  1006→        }
  1007→    }
  1008→
  1009→    /* Phase 2: Trace old→young references.
  1010→     * If card table is available, scan dirty cards (more efficient — only
  1011→     * scans cards that were actually written to). Otherwise, fall back
  1012→     * to scanning the entire remembered set. */
  1013→    if (gc->card_table != NULL) {
  1014→        scan_dirty_cards(gc);
  1015→    } else {
  1016→        /* Fallback: scan remembered set */
  1017→        for (uint32_t i = 0; i < gc->remembered_count; i++) {
  1018→            vtx_heap_object_t *old_obj = gc->remembered_set[i].obj;
  1019→            scan_object(gc, old_obj);
  1020→        }
  1021→    }
  1022→
  1023→    /* Phase 3: Scan to-space (processes objects copied in phases 1 & 2) */
  1024→    scan_to_space(gc);
  1025→
  1026→    /* Phase 4: Handle pinned objects — they stayed in from-space.
  1027→     * RT-3 fix: After scanning pinned objects, we must continue scanning
  1028→     * to-space because pinned-object scanning may have forwarded additional
  1029→     * young-gen objects that weren't reached in Phase 3. These newly-forwarded
  1030→     * objects need their fields traced too (the transitive closure must be
  1031→     * complete). We keep alternating between scanning pinned objects and
  1032→     * scanning to-space until no new objects are forwarded. */
  1033→    bool pinned_changed = true;
  1034→    while (pinned_changed) {
  1035→        pinned_changed = false;
  1036→        for (uint32_t i = 0; i < gc->pinned_count; i++) {
  1037→            vtx_heap_object_t *pinned = gc->pinned_objects[i];
  1038→            /* Pinned objects stay in from-space, but their fields may need updating */
  1039→            uint8_t old_field_count = pinned->field_count; /* save before scan may change refs */
  1040→            (void)old_field_count;
  1041→            scan_object(gc, pinned);
  1042→        }
  1043→        /* Scan to-space again — pinned-object scanning may have forwarded
  1044→         * new objects that need their fields traced */
  1045→        uint8_t *scan_before = gc->young_to.current;
  1046→        scan_to_space(gc);
  1047→        if (gc->young_to.current != scan_before) {
  1048→            pinned_changed = true; /* New objects were forwarded, re-scan pinned */
  1049→        }
  1050→    }
  1051→
  1052→    /* Phase 5: Swap semi-spaces.
  1053→     * RT-2 fix: Pinned objects remain in the OLD from-space (young_from
  1054→     * before swap). After swap, young_from becomes the NEW from-space
  1055→     * (containing the live objects), and young_to becomes the OLD from-space
  1056→     * (to be used as to-space next collection). The pinned objects are in
  1057→     * the OLD from-space, which after swap is young_to. We must NOT reset
  1058→     * young_to on the next collection until all pinned objects have been
  1059→     * copied out or unpinned.
  1060→     *
  1061→     * For now, the simplest correct fix is to copy pinned objects into
  1062→     * the new from-space (young_to before swap = young_from after swap)
  1063→     * before swapping, so they survive the swap. */
  1064→    for (uint32_t i = 0; i < gc->pinned_count; i++) {
  1065→        vtx_heap_object_t *pinned = gc->pinned_objects[i];
  1066→        /* C3 BUGFIX: Skip old-gen pinned objects — they must NOT be
  1067→         * copied into young semi-space. Copying an old-gen object
  1068→         * into young gen corrupts old-gen heap metadata and breaks
  1069→         * card-table / mark bitmap. */
  1070→        if (vtx_gc_in_old(gc, pinned)) continue;
  1071→        /* Pinned objects are still in from-space. Copy them to to-space
  1072→         * so they survive the swap. We can't move them (they're pinned),
  1073→         * so we copy them and leave a forwarding pointer. But wait —
  1074→         * other references to the pinned object (from root stack, etc.)
  1075→         * still point to the OLD location. We need to update those too.
  1076→         *
  1077→         * The simplest approach: since pinned objects are rare, just copy
  1078→         * them to to-space and update all references. The forwarding pointer
  1079→         * left in the old location will redirect any stale references.
  1080→         *
  1081→         * BUG #6 KNOWN LIMITATION: Old-gen objects that hold references
  1082→         * to pinned objects are NOT updated here. The forwarding pointer
  1083→         * in the old location will redirect root-stack references on the
  1084→         * next scan, but old-gen → pinned-young references that were
  1085→         * recorded in the remembered set may now point to the old
  1086→         * (forwarded) location. We add the new copy to the remembered
  1087→         * set below so it will be rescanned on the next collection. */
  1088→        void *new_ptr = semi_space_alloc(&gc->young_to, pinned->size);
  1089→        if (new_ptr != NULL) {
  1090→            memcpy(new_ptr, pinned, pinned->size);
  1091→            vtx_heap_object_t *new_obj = (vtx_heap_object_t *)new_ptr;
  1092→            new_obj->gc_pinned = 0; /* No longer needs to be pinned in new location */
  1093→            /* Leave forwarding pointer in old location */
  1094→            /* RT-006 fix: memcpy to avoid strict-aliasing UB. */
  1095→            uintptr_t _fwd_ptr2 = (uintptr_t)new_obj;
  1096→            memcpy(pinned, &_fwd_ptr2, sizeof(uintptr_t));
  1097→            pinned->size = VTX_GC_FORWARDING_SENTINEL;
  1098→
  1099→            /* RT-009 fix: the old code added the NEW copy (in young_to)
  1100→             * to the remembered set. The remembered set is for OLD→young
  1101→             * references, but new_obj is in young gen — adding it pollutes
  1102→             * the set with young-gen objects, causing incorrect scanning on
  1103→             * the next collection. The new copy doesn't have any old→young
  1104→             * references yet (it was just allocated); it will be added to
  1105→             * the remembered set naturally if/when an old-gen object later
  1106→             * writes to it (via the write barrier). Skip the remembered-set
  1107→             * add here. */
  1108→        }
  1109→    }
  1110→
  1111→    vtx_semi_space_t tmp = gc->young_from;
  1112→    gc->young_from = gc->young_to;
  1113→    gc->young_to = tmp;
  1114→
  1115→    /* RT-6 fix: Rebuild the remembered set instead of clearing it.
  1116→     * OLD BUG: The remembered set was unconditionally cleared, hoping write
  1117→     * barriers would rebuild it. But write barriers only fire on NEW writes.
  1118→     * Old-gen objects that still reference into the new young generation
  1119→     * would not have their entries recreated, causing the next collection
  1120→     * to miss those references and prematurely collect live objects.
  1121→     *
  1122→     * NEW approach: After the swap, scan all old-gen objects and rebuild
  1123→     * the remembered set with any old→young references found. */
  1124→    for (uint32_t i = 0; i < gc->remembered_count; i++) {
  1125→        vtx_heap_object_t *old_obj = gc->remembered_set[i].obj;
  1126→        if (old_obj != NULL) {
  1127→            old_obj->gc_remembered = 0;
  1128→        }
  1129→    }
  1130→    gc->remembered_count = 0;
  1131→
  1132→    /* Rebuild remembered set by scanning old-gen objects for young references */
  1133→    if (gc->old_gen.start != NULL && gc->old_gen.size > 0) {
  1134→        uint8_t *ptr = gc->old_gen.start;
  1135→        /* B11 fix: Scan the entire old-gen region [start, start+size) rather
  1136→         * than [start, start+used). Using `used` misses old-gen objects that
  1137→         * live past the current `used` watermark (e.g., free list blocks or
  1138→         * pinned objects near the end), dropping their old→young references
  1139→         * from the remembered set and leading to premature collection. */
  1140→        uint8_t *end = gc->old_gen.start + gc->old_gen.size;
  1141→        while (ptr < end) {
  1142→            vtx_heap_object_t *obj = (vtx_heap_object_t *)ptr;
  1143→            if (obj->gc_mark == 0xFF) {
  1144→                /* Free block — skip it */
  1145→                vtx_free_node_t *free_node = (vtx_free_node_t *)ptr;
  1146→                size_t free_size = align_up_8(vtx_free_node_get_size(free_node));
  1147→                if (free_size < sizeof(vtx_free_node_t)) {
  1148→                    free_size = sizeof(vtx_free_node_t);
  1149→                }
  1150→                ptr += free_size;
  1151→                continue;
  1152→            }
  1153→            if (obj->size == 0 || obj->size == VTX_GC_FORWARDING_SENTINEL) {
  1154→                ptr += sizeof(vtx_free_node_t);
  1155→                continue;
  1156→            }
  1157→            if (obj->size >= VTX_GC_FORWARDING_SENTINEL) {
  1158→                /* Skip corrupted/invalid entries */
  1159→                break;
  1160→            }
  1161→            size_t obj_size = align_up_8(obj->size);
  1162→            /* Check if this old-gen object has any references into young gen */
  1163→            for (uint32_t f = 0; f < obj->field_count; f++) {
  1164→                vtx_value_t field = obj->fields[f];
  1165→                if (vtx_is_heap_ptr(field)) {
  1166→                    void *field_ptr = vtx_heap_ptr(field);
  1167→                    if (vtx_gc_in_young(gc, field_ptr)) {
  1168→                        /* Add to remembered set.
  1169→                         * BUGFIX: Grow the remembered set if full instead of
  1170→                         * silently dropping entries. Dropping entries causes
  1171→                         * old→young references to be missed, leading to
  1172→                         * premature collection of live young-gen objects. */
  1173→                        if (gc->remembered_count >= gc->remembered_capacity) {
  1174→                            uint32_t new_cap = gc->remembered_capacity > 0 ? gc->remembered_capacity * 2 : 16;
  1175→                            if (new_cap <= gc->remembered_capacity) new_cap = gc->remembered_capacity + 16; /* overflow guard */
  1176→                            vtx_remembered_entry_t *new_set = (vtx_remembered_entry_t *)realloc(
  1177→                                gc->remembered_set, new_cap * sizeof(vtx_remembered_entry_t));
  1178→                            if (new_set == NULL) {
  1179→                                /* Cannot grow — this is a fatal condition.
  1180→                                 * We must not silently drop, so we assert. */
  1181→                                VTX_ASSERT(false, "remembered set overflow during rebuild");
  1182→                                break;
  1183→                            }
  1184→                            gc->remembered_set = new_set;
  1185→                            gc->remembered_capacity = new_cap;
  1186→                        }
  1187→                        gc->remembered_set[gc->remembered_count].obj = obj;
  1188→                        gc->remembered_count++;
  1189→                        obj->gc_remembered = 1;
  1190→                        break; /* Only add each object once */
  1191→                    }
  1192→                }
  1193→            }
  1194→            ptr += obj_size;
  1195→        }
  1196→    }
  1197→
  1198→    gc->collections_done++;
  1199→}
  1200→
  1201→/* ========================================================================== */
  1202→/* Old-gen collection (mark-sweep)                                             */
  1203→/* ========================================================================== */
  1204→
  1205→/**
  1206→ * Mark an object and recursively trace its fields.
  1207→ * RT-5 fix: Trace through ALL reachable objects, not just old-gen.
  1208→ * Young-gen objects can hold references to old-gen objects, and we must
  1209→ * follow those references during old-gen collection to avoid sweeping
  1210→ * live old-gen objects.
  1211→ *
  1212→ * BUGFIX: The old code only traced ONE level through young-gen objects.
  1213→ * If there was a chain Root → YoungObj1 → YoungObj2 → OldObj2, OldObj2
  1214→ * was never marked. Now we recursively trace through young-gen objects
  1215→ * to find ALL transitively reachable old-gen objects. We use the gc_mark
  1216→ * bit on young-gen objects as a temporary visited flag during old-gen
  1217→ * collection to prevent infinite loops on cycles.
  1218→ */
  1219→/* RT-010 fix: mark_object was recursive — a deep object graph (e.g.,
  1220→ * a linked list of 100K elements) would overflow the C stack. Convert
  1221→ * to iterative using an explicit mark stack. The stack is grown on
  1222→ * demand; entries are pushed when we encounter an unmarked object,
  1223→ * and popped when we finish tracing its fields. */
  1224→static void mark_object(vtx_gc_t *gc, vtx_heap_object_t *root)
  1225→{
  1226→    VTX_ASSERT(root != NULL, "object must not be NULL");
  1227→
  1228→    if (root->gc_mark) return; /* already marked */
  1229→    root->gc_mark = 1;
  1230→
  1231→    /* Explicit mark stack — grown on demand. */
  1232→    vtx_heap_object_t **stack = NULL;
  1233→    uint32_t stack_size = 0;
  1234→    uint32_t stack_cap = 0;
  1235→
  1236→    /* Push the root. */
  1237→    stack_cap = 64;
  1238→    stack = (vtx_heap_object_t **)malloc(stack_cap * sizeof(vtx_heap_object_t *));
  1239→    if (!stack) {
  1240→        /* OOM — can't trace. The root is already marked, so we lose
  1241→         * transitive reachability but at least the root survives. */
  1242→        return;
  1243→    }
  1244→    stack[stack_size++] = root;
  1245→
  1246→    while (stack_size > 0) {
  1247→        vtx_heap_object_t *obj = stack[--stack_size];
  1248→
  1249→        /* Trace fields — follow ALL references, not just old-gen. */
  1250→        for (uint32_t i = 0; i < obj->field_count; i++) {
  1251→            vtx_value_t field = obj->fields[i];
  1252→            if (vtx_is_heap_ptr(field)) {
  1253→                vtx_heap_object_t *field_obj = (vtx_heap_object_t *)vtx_heap_ptr(field);
  1254→                if (field_obj->gc_mark) continue; /* already visited */
  1255→
  1256→                /* Mark and push. Both old-gen and young-gen objects are
  1257→                 * traced: young-gen objects are marked with gc_mark=1
  1258→                 * as a visited flag (cleared after the mark phase). */
  1259→                field_obj->gc_mark = 1;
  1260→
  1261→                /* Grow stack if needed. */
  1262→                if (stack_size >= stack_cap) {
  1263→                    uint32_t new_cap = stack_cap * 2;
  1264→                    vtx_heap_object_t **new_stack = (vtx_heap_object_t **)
  1265→                        realloc(stack, new_cap * sizeof(vtx_heap_object_t *));
  1266→                    if (!new_stack) {
  1267→                        /* OOM — field_obj is marked but its children
  1268→                         * won't be traced. Acceptable: they survive if
  1269→                         * reachable from another path. */
  1270→                        continue;
  1271→                    }
  1272→                    stack = new_stack;
  1273→                    stack_cap = new_cap;
  1274→                }
  1275→                stack[stack_size++] = field_obj;
  1276→            }
  1277→        }
  1278→    }
  1279→
  1280→    free(stack);
  1281→}
  1282→
  1283→/**
  1284→ * Mark phase: trace from all roots.
  1285→ * RT-5 fix: Also trace through young-gen objects that may reference old-gen.
  1286→ * OLD BUG: mark_phase only traced roots that are directly in old gen.
  1287→ * If Root → YoungObj → OldObj, the OldObj was never marked and got swept.
  1288→ */
  1289→static void mark_phase(vtx_gc_t *gc)
  1290→{
  1291→    /* Mark from root stack — trace ALL reachable objects, not just old-gen */
  1292→    for (uint32_t i = 0; i < gc->root_count; i++) {
  1293→        vtx_value_t value = gc->root_stack[i].value;
  1294→        if (vtx_is_heap_ptr(value)) {
  1295→            vtx_heap_object_t *obj = (vtx_heap_object_t *)vtx_heap_ptr(value);
  1296→            mark_object(gc, obj);
  1297→        }
  1298→    }
  1299→}
  1300→
  1301→/**
  1302→ * Sweep phase: walk through old gen memory, free unmarked objects,
  1303→ * clear marks on marked objects.
  1304→ * RT-4 fix: Use gc_mark == 0xFF sentinel to distinguish free blocks
  1305→ * from live objects. Free blocks are skipped (already on the free list).
  1306→ */
  1307→static void sweep_phase(vtx_gc_t *gc)
  1308→{
  1309→    uint8_t *ptr = gc->old_gen.start;
  1310→    uint8_t *end = gc->old_gen.start + gc->old_gen.size;
  1311→
  1312→    while (ptr < end) {
  1313→        vtx_heap_object_t *obj = (vtx_heap_object_t *)ptr;
  1314→
  1315→        /* RT-4 fix: Check if this is a free block (gc_mark == 0xFF) */
  1316→        if (obj->gc_mark == 0xFF) {
  1317→            /* This is a free block — skip it */
  1318→            vtx_free_node_t *free_node = (vtx_free_node_t *)ptr;
  1319→            size_t free_size = align_up_8(vtx_free_node_get_size(free_node));
  1320→            if (free_size < sizeof(vtx_free_node_t)) {
  1321→                free_size = sizeof(vtx_free_node_t);
  1322→            }
  1323→            ptr += free_size;
  1324→            continue;
  1325→        }
  1326→
  1327→        if (obj->size == 0 || obj->size >= VTX_GC_FORWARDING_SENTINEL) {
  1328→            /* Uninitialized/corrupted block — skip minimum unit */
  1329→            ptr += sizeof(vtx_free_node_t);
  1330→            continue;
  1331→        }
  1332→
  1333→        size_t obj_size = align_up_8(obj->size);
  1334→
  1335→        if (obj->gc_mark) {
  1336→            /* Object is alive — clear the mark */
  1337→            obj->gc_mark = 0;
  1338→        } else {
  1339→            /* Object is dead — free it */
  1340→            old_gen_free(&gc->old_gen, ptr, obj_size);
  1341→        }
  1342→
  1343→        ptr += obj_size;
  1344→    }
  1345→}
  1346→
  1347→void vtx_gc_collect_old(vtx_gc_t *gc)
  1348→{
  1349→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
  1350→
  1351→    /* Mark phase */
  1352→    mark_phase(gc);
  1353→
  1354→    /* Sweep phase */
  1355→    sweep_phase(gc);
  1356→
  1357→    /* BUGFIX: Clear gc_mark on young-gen objects that were temporarily
  1358→     * marked during old-gen mark phase (used as visited flags to prevent
  1359→     * infinite loops). Without this, young-gen objects would retain
  1360→     * gc_mark=1, causing the next young-gen collection to skip them
  1361→     * during mark phase (thinking they're already marked) and potentially
  1362→     * collect live objects. */
  1363→    if (gc->young_from.start != NULL) {
  1364→        uint8_t *ptr = gc->young_from.start;
  1365→        uint8_t *end = gc->young_from.current;
  1366→        while (ptr < end) {
  1367→            vtx_heap_object_t *obj = (vtx_heap_object_t *)ptr;
  1368→            if (obj->size == 0) {
  1369→                ptr += align_up_8(VTX_HEAP_OBJECT_HEADER_SIZE);
  1370→                continue;
  1371→            }
  1372→            obj->gc_mark = 0;
  1373→            ptr += align_up_8(obj->size);
  1374→        }
  1375→    }
  1376→}

  1379→/* Pinning                                                                     */
  1380→/* ========================================================================== */
  1381→
  1382→void vtx_gc_pin(vtx_gc_t *gc, vtx_heap_object_t *obj)
  1383→{
  1384→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
  1385→    VTX_ASSERT(obj != NULL, "object must not be NULL");
  1386→
  1387→    if (obj->gc_pinned) {
  1388→        return; /* already pinned */
  1389→    }
  1390→
  1391→    obj->gc_pinned = 1;
  1392→
  1393→    /* Add to pinned objects list */
  1394→    if (gc->pinned_count >= gc->pinned_capacity) {
  1395→        uint32_t new_cap = gc->pinned_capacity > 0 ? gc->pinned_capacity * 2 : 16;
  1396→        if (new_cap <= gc->pinned_capacity) new_cap = gc->pinned_capacity + 16; /* overflow guard */
  1397→        vtx_heap_object_t **new_list = (vtx_heap_object_t **)realloc(
  1398→            gc->pinned_objects, new_cap * sizeof(vtx_heap_object_t *));
  1399→        if (new_list == NULL) {
  1400→            VTX_ASSERT(false, "pinned objects list overflow — out of memory");
  1401→            return;
  1402→        }
  1403→        gc->pinned_objects = new_list;
  1404→        gc->pinned_capacity = new_cap;
  1405→    }
  1406→
  1407→    gc->pinned_objects[gc->pinned_count] = obj;
  1408→    gc->pinned_count++;
  1409→}
  1410→
  1411→void vtx_gc_unpin(vtx_gc_t *gc, vtx_heap_object_t *obj)
  1412→{
  1413→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
  1414→    VTX_ASSERT(obj != NULL, "object must not be NULL");
  1415→
  1416→    if (!obj->gc_pinned) {
  1417→        return;
  1418→    }
  1419→
  1420→    obj->gc_pinned = 0;
  1421→
  1422→    /* Remove from pinned objects list */
  1423→    for (uint32_t i = 0; i < gc->pinned_count; i++) {
  1424→        if (gc->pinned_objects[i] == obj) {
  1425→            /* Swap with the last element for O(1) removal */
  1426→            gc->pinned_objects[i] = gc->pinned_objects[gc->pinned_count - 1];
  1427→            gc->pinned_count--;
  1428→            return;
  1429→        }
  1430→    }
  1431→}

  1433→bool vtx_gc_is_pinned(vtx_gc_t *gc, vtx_heap_object_t *obj)
  1434→{
  1435→    (void)gc;
  1436→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
  1437→    VTX_ASSERT(obj != NULL, "object must not be NULL");
  1438→    return obj->gc_pinned != 0;
  1439→}

  1444→
  1445→bool vtx_gc_in_young(const vtx_gc_t *gc, const void *ptr)
  1446→{
  1447→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
  1448→    /* RT-015 fix: the old code checked BOTH semi-spaces on every call.
  1449→     * This is the write-barrier hot path — for old-gen objects (the
  1450→     * common case), both checks fail, doubling the cost. The active
  1451→     * allocation space is young_from; young_to is only populated
  1452→     * during a stop-the-world collection. After collection, the spaces
  1453→     * are swapped and young_to is empty. So checking only young_from
  1454→     * is correct for non-collection-time calls (which is when the
  1455→     * write barrier fires). */
  1456→    return ptr_in_range(ptr, gc->young_from.start, gc->young_from.size);
  1457→}

  1460→{
  1461→    VTX_ASSERT(gc != NULL, "GC must not be NULL");
  1462→    return ptr_in_range(ptr, gc->old_gen.start, gc->old_gen.size);
  1463→}
