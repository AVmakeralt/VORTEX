   115→   115→/* Internal: join two escape state arrays (dst = join(dst, src))               */

   117→   117→/* ========================================================================== */
   118→   118→
   119→   119→static bool join_escape_states(vtx_escape_state_t *dst,
   120→   120→                                const vtx_escape_state_t *src,
   121→   121→                                uint32_t count)
   122→   122→{

   124→   124→    for (uint32_t i = 0; i < count; i++) {

   127→   127→            dst[i] = joined;
   128→   128→            changed = true;
   129→   129→        }
   130→   130→    }
   131→   131→    return changed;

   201→   201→/* ========================================================================== */
   202→   202→/* Internal: transfer function for a single node                               */
   203→   203→/*                                                                           */
   204→   204→/* Examines how the node uses values and updates escape states accordingly.   */
   205→   205→/* ========================================================================== */
   206→   206→
   207→   207→static void transfer_node(vtx_node_t *node, vtx_node_table_t *table,
   208→   208→                           vtx_escape_state_t *state, uint32_t state_count)
   209→   209→{
   210→   210→    if (node->dead) return;
   211→   211→
   212→   212→    switch (node->opcode) {
   213→   213→    /* ---- Return: any returned allocation escapes globally ---- */
   214→   214→    case VTX_OP_Return:
   215→   215→        for (uint32_t i = 0; i < node->input_count; i++) {
   216→   216→            vtx_nodeid_t input_id = node->inputs[i];
   217→   217→            if (input_id < state_count) {
   218→   218→                /* The returned value itself is a data value. If it is an
   219→   219→                 * allocation, it escapes globally. If it references an
   220→   220→                 * allocation indirectly, we need to check the input node. */
   221→   221→                vtx_node_t *input = vtx_node_get(table, input_id);
   222→   222→                if (input && !input->dead && is_allocation(input->opcode)) {
   223→   223→                    state[input_id] = vtx_escape_join(state[input_id],
   224→   224→                                                       VTX_ESCAPE_GLOBAL);
   225→   225→                }
   226→   226→            }
   227→   227→        }
   228→   228→        break;
   229→   229→
   230→   230→    /* ---- StoreField: if the stored value is an allocation, it escapes
   231→   231→       based on the receiver's escape state ---- */
   232→   232→    case VTX_OP_StoreField:
   233→   233→        if (node->input_count >= 2) {
   234→   234→            /* Input layout: [control?, memory?, receiver, value] */
   235→   235→            /* Find the receiver and value inputs — they are the last two
   236→   236→             * data inputs. We search for them by position. */
   237→   237→            vtx_nodeid_t receiver_id = node->inputs[node->input_count - 2];
   238→   238→            vtx_nodeid_t value_id    = node->inputs[node->input_count - 1];
   239→   239→
   240→   240→            /* If the stored value is an allocation, it escapes at least
   241→   241→             * as much as the container object */
   242→   242→            if (value_id < state_count) {
   243→   243→                vtx_node_t *val_node = vtx_node_get(table, value_id);
   244→   244→                if (val_node && !val_node->dead && is_allocation(val_node->opcode)) {
   245→   245→                    vtx_escape_state_t container_state = VTX_ESCAPE_GLOBAL;
   246→   246→                    if (receiver_id < state_count) {
   247→   247→                        vtx_node_t *recv_node = vtx_node_get(table, receiver_id);
   248→   248→                        if (recv_node && !recv_node->dead && is_allocation(recv_node->opcode)) {
   249→   249→                            container_state = state[receiver_id];
   250→   250→                        }
   251→   251→                    }
   252→   252→                    /* Stored into a container that escapes → the stored value
   253→   253→                     * escapes at least as much as the container */
   254→   254→                    state[value_id] = vtx_escape_join(state[value_id],
   255→   255→                                                       vtx_escape_join(container_state, VTX_ESCAPE_ARG));
   256→   256→                }
   257→   257→            }
   258→   258→        }
   259→   259→        break;
   260→   260→
   261→   261→    /* ---- StoreIndexed: storing into an array — the value escapes
   262→   262→       at least as much as the array ---- */
   263→   263→    case VTX_OP_StoreIndexed:
   264→   264→        if (node->input_count >= 3) {
   265→   265→            vtx_nodeid_t array_id = node->inputs[node->input_count - 3];
   266→   266→            vtx_nodeid_t value_id = node->inputs[node->input_count - 1];
   267→   267→
   268→   268→            if (value_id < state_count) {
   269→   269→                vtx_node_t *val_node = vtx_node_get(table, value_id);
   270→   270→                if (val_node && !val_node->dead && is_allocation(val_node->opcode)) {
   271→   271→                    vtx_escape_state_t array_state = VTX_ESCAPE_GLOBAL;
   272→   272→                    if (array_id < state_count) {
   273→   273→                        vtx_node_t *arr_node = vtx_node_get(table, array_id);
   274→   274→                        if (arr_node && !arr_node->dead && is_allocation(arr_node->opcode)) {
   275→   275→                            array_state = state[array_id];
   276→   276→                        }
   277→   277→                    }
   278→   278→                    state[value_id] = vtx_escape_join(state[value_id],
   279→   279→                                                       vtx_escape_join(array_state, VTX_ESCAPE_ARG));
   280→   280→                }
   281→   281→            }
   282→   282→        }
   283→   283→        break;
   284→   284→
   285→   285→    /* ---- Calls: any allocation passed as argument escapes ---- */
   286→   286→    case VTX_OP_CallStatic:
   287→   287→    case VTX_OP_CallVirtual:
   288→   288→    case VTX_OP_CallInterface:
   289→   289→    case VTX_OP_CallRuntime:
   290→   290→        /* All allocation arguments to unknown functions escape.
   291→   291→         * For CallVirtual/CallInterface, the receiver also escapes. */
   292→   292→        for (uint32_t i = 0; i < node->input_count; i++) {
   293→   293→            vtx_nodeid_t input_id = node->inputs[i];
   294→   294→            if (input_id < state_count) {
   295→   295→                vtx_node_t *input = vtx_node_get(table, input_id);
   296→   296→                if (input && !input->dead && is_allocation(input->opcode)) {
   297→   297→                    /* Arguments to unknown functions escape at least through args.
   298→   298→                     * For CallRuntime (which may store globally), be conservative. */
   299→   299→                    if (node->opcode == VTX_OP_CallRuntime) {
   300→   300→                        state[input_id] = vtx_escape_join(state[input_id],
   301→   301→                                                           VTX_ESCAPE_GLOBAL);
   302→   302→                    } else {
   303→   303→                        state[input_id] = vtx_escape_join(state[input_id],
   304→   304→                                                           VTX_ESCAPE_ARG);
   305→   305→                    }
   306→   306→                }
   307→   307→            }
   308→   308→        }
   309→   309→
   310→   310→        /* For virtual/interface calls, the receiver may dispatch to an
   311→   311→         * unknown method that stores the object globally. Conservative:
   312→   312→         * ArgEscape (could be made more precise with callee analysis). */
   313→   313→        if (node->opcode == VTX_OP_CallVirtual ||
   314→   314→            node->opcode == VTX_OP_CallInterface) {
   315→   315→            /* The receiver is typically the first data input after control/memory.
   316→   316→             * We check all inputs for allocation nodes — already handled above. */
   317→   317→        }
   318→   318→        break;
   319→   319→
   320→   320→    /* ---- Monitor: using an object in a monitor makes it globally escape ---- */
   321→   321→    case VTX_OP_DeoptGuard:
   322→   322→        /* DeoptGuard: the guarded value is consumed but doesn't escape */
   323→   323→        break;
   324→   324→
   325→   325→    /* ---- CheckCast / InstanceOf: the object doesn't escape through these
   326→   326→       (they are pure queries on the object's type) ---- */
   327→   327→    case VTX_OP_CheckCast:
   328→   328→    case VTX_OP_InstanceOf:
   329→   329→        /* No escape — these are read-only type checks */
   330→   330→        break;
   331→   331→
   332→   332→    /* ---- Guard: the checked value doesn't escape ---- */
   333→   333→    case VTX_OP_Guard:
   334→   334→        /* Guards don't cause escape — they just check a condition */
   335→   335→        break;
   336→   336→
   337→   337→    /* ---- Phi: a Phi of allocations — propagate escape states ---- */
   338→   338→    case VTX_OP_Phi:
   339→   339→        /* Phi nodes merge values from multiple control flow paths.
   340→   340→         * If any input allocation has a non-NoEscape state, all allocation
   341→   341→         * inputs of the Phi must have at least that escape state, because
   342→   342→         * the Phi merges them into a single value that may escape.
   343→   343→         *
   344→   344→         * Bug 8 fix: Previously, the Phi transfer was a no-op — it just
   345→   345→         * iterated over inputs without propagating escape states. This meant
   346→   346→         * that if Phi(A, B) where A escapes but B doesn't, B would remain
   347→   347→         * NoEscape even though the merged value escapes. Now we compute
   348→   348→         * the maximum escape state across all Phi inputs and propagate it
   349→   349→         * to every allocation input of the Phi. */
   350→   350→        {
   351→   351→            vtx_escape_state_t max_state = VTX_ESCAPE_NONE;
   352→   352→            /* First pass: compute the maximum escape state across all inputs */
   353→   353→            for (uint32_t i = 0; i < node->input_count; i++) {
   354→   354→                vtx_nodeid_t input_id = node->inputs[i];
   355→   355→                if (input_id < state_count) {
   356→   356→                    vtx_node_t *input = vtx_node_get(table, input_id);
   357→   357→                    if (input && !input->dead && is_allocation(input->opcode)) {
   358→   358→                        max_state = vtx_escape_join(max_state, state[input_id]);
   359→   359→                    }
   360→   360→                }
   361→   361→            }
   362→   362→            /* Second pass: propagate max_state to all allocation inputs.
   363→   363→             * If any input allocation has a lower escape state than the
   364→   364→             * Phi's maximum, it must be elevated because the Phi's result
   365→   365→             * (which merges all inputs) carries the highest escape state. */
   366→   366→            for (uint32_t i = 0; i < node->input_count; i++) {
   367→   367→                vtx_nodeid_t input_id = node->inputs[i];
   368→   368→                if (input_id < state_count) {
   369→   369→                    vtx_node_t *input = vtx_node_get(table, input_id);
   370→   370→                    if (input && !input->dead && is_allocation(input->opcode)) {
   371→   371→                        if (state[input_id] < max_state) {
   372→   372→                            state[input_id] = max_state;
   373→   373→                            /* changed flag is not accessible here, but the
   374→   374→                             * caller (transfer_block -> worklist) will detect
   375→   375→                             * the state change when comparing exit states. */
   376→   376→                        }
   377→   377→                    }
   378→   378→                }
   379→   379→            }
   380→   380→        }
   381→   381→        break;
   382→   382→
   383→   383→    default:
   384→   384→        break;
   385→   385→    }
   386→   386→}

   388→   388→/* ========================================================================== */
   389→   389→/* Internal: pre-compute block-to-node mapping (§2.1 — fix O(N×B) scan)      */
   390→   390→/* ========================================================================== */
   391→   391→/*
   392→   392→ * PEA audit: transfer_block re-scans ALL N nodes for EVERY block to find
   393→   393→ * which nodes belong to that block. For a graph with B blocks and N nodes,
   394→   394→ * this is O(N×B) per dataflow iteration — for T2 graphs with 5000 nodes
   395→   395→ * and 50 blocks, that's 250K comparisons per iteration × ~10 iterations
   396→   396→ * = 2.5M comparisons.
   397→   397→ *
   398→   398→ * GraalVM's PEA uses a pre-computed BlockMap (BlockBag in
   399→   399→ * EscapeAnalysisPhase). V8's escape analysis uses a schedule
   400→   400→ * (Schedule::rpo_order) to iterate nodes per-block.
   401→   401→ *
   402→   402→ * Fix: pre-compute a block→node-list mapping once before the worklist,
   403→   403→ * then iterate only the relevant nodes in transfer_block. */
   404→   404→typedef struct {
   405→   405→    vtx_nodeid_t *nodes;    /* node IDs belonging to this block */
   406→   406→    uint32_t      count;    /* number of nodes in this block */
   407→   407→} vtx_block_node_list_t;
   408→   408→
   409→   409→static vtx_block_node_list_t *build_block_node_lists(vtx_graph_t *graph,
   410→   410→                                                       vtx_arena_t *arena)
   411→   411→{
   412→   412→    uint32_t block_count = graph->block_count;
   413→   413→    uint32_t node_count = graph->node_table.count;
   414→   414→
   415→   415→    /* Allocate per-block lists */
   416→   416→    vtx_block_node_list_t *lists = vtx_arena_alloc(arena,
   417→   417→        block_count * sizeof(vtx_block_node_list_t));
   418→   418→    if (!lists) return NULL;
   419→   419→    memset(lists, 0, block_count * sizeof(vtx_block_node_list_t));
   420→   420→
   421→   421→    /* First pass: count nodes per block */
   422→   422→    for (uint32_t i = 0; i < node_count; i++) {
   423→   423→        vtx_node_t *node = &graph->node_table.nodes[i];
   424→   424→        if (node->dead) continue;
   425→   425→
   426→   426→        /* Determine which block this node belongs to */
   427→   427→        for (uint32_t b = 0; b < block_count; b++) {
   428→   428→            vtx_block_info_t *block = &graph->blocks[b];
   429→   429→            bool belongs = false;
   430→   430→
   431→   431→            if (node->id == block->region_node ||
   432→   432→                node->id == block->control_node ||
   433→   433→                node->id == block->memory_node) {
   434→   434→                belongs = true;
   435→   435→            } else {
   436→   436→                for (uint32_t j = 0; j < node->input_count; j++) {
   437→   437→                    if (node->inputs[j] == block->region_node ||
   438→   438→                        node->inputs[j] == block->control_node ||
   439→   439→                        node->inputs[j] == block->memory_node) {
   440→   440→                        belongs = true;
   441→   441→                        break;
   442→   442→                    }
   443→   443→                }
   444→   444→                if (!belongs && block->region_node < node_count) {
   445→   445→                    vtx_node_t *region = &graph->node_table.nodes[block->region_node];
   446→   446→                    if (node->bytecode_pc == region->bytecode_pc) {
   447→   447→                        belongs = true;
   448→   448→                    }
   449→   449→                }
   450→   450→            }
   451→   451→
   452→   452→            if (belongs) {
   453→   453→                lists[b].count++;
   454→   454→                break;  /* each node belongs to at most one block */
   455→   455→            }
   456→   456→        }
   457→   457→    }
   458→   458→
   459→   459→    /* Allocate arrays */
   460→   460→    for (uint32_t b = 0; b < block_count; b++) {
   461→   461→        if (lists[b].count > 0) {
   462→   462→            lists[b].nodes = vtx_arena_alloc(arena,
   463→   463→                lists[b].count * sizeof(vtx_nodeid_t));
   464→   464→            if (!lists[b].nodes) return NULL;
   465→   465→            lists[b].count = 0;  /* reset for second pass */
   466→   466→        }
   467→   467→    }
   468→   468→
   469→   469→    /* Second pass: fill arrays */
   470→   470→    for (uint32_t i = 0; i < node_count; i++) {
   471→   471→        vtx_node_t *node = &graph->node_table.nodes[i];
   472→   472→        if (node->dead) continue;
   473→   473→
   474→   474→        for (uint32_t b = 0; b < block_count; b++) {
   475→   475→            vtx_block_info_t *block = &graph->blocks[b];
   476→   476→            bool belongs = false;
   477→   477→
   478→   478→            if (node->id == block->region_node ||
   479→   479→                node->id == block->control_node ||
   480→   480→                node->id == block->memory_node) {
   481→   481→                belongs = true;
   482→   482→            } else {
   483→   483→                for (uint32_t j = 0; j < node->input_count; j++) {
   484→   484→                    if (node->inputs[j] == block->region_node ||
   485→   485→                        node->inputs[j] == block->control_node ||
   486→   486→                        node->inputs[j] == block->memory_node) {
   487→   487→                        belongs = true;
   488→   488→                        break;
   489→   489→                    }
   490→   490→                }
   491→   491→                if (!belongs && block->region_node < node_count) {
   492→   492→                    vtx_node_t *region = &graph->node_table.nodes[block->region_node];
   493→   493→                    if (node->bytecode_pc == region->bytecode_pc) {
   494→   494→                        belongs = true;
   495→   495→                    }
   496→   496→                }
   497→   497→            }
   498→   498→
   499→   499→            if (belongs) {
   500→   500→                lists[b].nodes[lists[b].count++] = node->id;
   501→   501→                break;
   502→   502→            }
   503→   503→        }
   504→   504→    }
   505→   505→
   506→   506→    return lists;
   507→   507→}
   508→   508→
   509→   509→/* ========================================================================== */
   510→   510→/* Internal: fast transfer function using pre-computed block-node lists        */
   511→   511→/* ========================================================================== */
   512→   512→/*
   513→   513→ * §2.1: Uses the pre-computed block_node_lists to iterate only the nodes
   514→   514→ * that belong to this block, instead of scanning all N nodes.
   515→   515→ * GraalVM: EscapeAnalysisPhase uses BlockBag for per-block iteration.
   516→   516→ * V8: EscapeAnalysis uses Schedule::rpo_order for per-block iteration.
   517→   517→ */
   518→   518→static void transfer_block_fast(vtx_graph_t *graph, uint32_t block_idx,
   519→   519→                                 vtx_pea_block_state_t *bs,
   520→   520→                                 vtx_node_table_t *table,
   521→   521→                                 const vtx_block_node_list_t *block_lists)
   522→   522→{
   523→   523→    /* Start from entry state */
   524→   524→    copy_escape_states(bs->exit_state, bs->entry_state, bs->state_count);
   525→   525→
   526→   526→    const vtx_block_node_list_t *list = &block_lists[block_idx];
   527→   527→    for (uint32_t i = 0; i < list->count; i++) {
   528→   528→        vtx_node_t *node = &table->nodes[list->nodes[i]];
   529→   529→        if (node->dead) continue;
   530→   530→        transfer_node(node, table, bs->exit_state, bs->state_count);
   531→   531→    }
   532→   532→}
   533→   533→
   534→   534→/* ========================================================================== */
   535→   535→/* Internal: compute reverse postorder for worklist                            */
   536→   536→/* ========================================================================== */
   537→   537→
   538→   538→static uint32_t *compute_reverse_postorder(vtx_graph_t *graph, vtx_arena_t *arena,
   539→   539→                                             uint32_t *rpo_count)
   540→   540→{
   541→   541→    uint32_t n = graph->block_count;
   542→   542→    if (n == 0) {
   543→   543→        *rpo_count = 0;
   544→   544→        return NULL;
   545→   545→    }
   546→   546→
   547→   547→    uint32_t *rpo = vtx_arena_alloc(arena, n * sizeof(uint32_t));
   548→   548→    if (!rpo) return NULL;
   549→   549→
   550→   550→    /* Simple DFS-based reverse postorder computation.
   551→   551→     * We start from block 0 (entry block) and traverse successors. */
   552→   552→    bool *visited = vtx_arena_alloc(arena, n * sizeof(bool));
   553→   553→    uint32_t *stack = vtx_arena_alloc(arena, n * sizeof(uint32_t));
   554→   554→    if (!visited || !stack) return NULL;
   555→   555→
   556→   556→    memset(visited, 0, n * sizeof(bool));
   557→   557→
   558→   558→    uint32_t rpo_idx = n; /* fill from end */
   559→   559→    int32_t top = 0;
   560→   560→    stack[0] = 0;
   561→   561→    visited[0] = true;
   562→   562→
   563→   563→    while (top >= 0) {
   564→   564→        uint32_t current = stack[top];
   565→   565→        bool has_unvisited_succ = false;
   566→   566→
   567→   567→        vtx_block_info_t *block = &graph->blocks[current];
   568→   568→        for (uint32_t i = 0; i < block->succ_count; i++) {
   569→   569→            uint32_t succ = block->succ_indices[i];
   570→   570→            if (succ < n && !visited[succ]) {
   571→   571→                visited[succ] = true;
   572→   572→                top++;
   573→   573→                stack[top] = succ;
   574→   574→                has_unvisited_succ = true;
   575→   575→                break;
   576→   576→            }
   577→   577→        }
   578→   578→
   579→   579→        if (!has_unvisited_succ) {
   580→   580→            rpo_idx--;
   581→   581→            rpo[rpo_idx] = current;
   582→   582→            top--;
   583→   583→        }
   584→   584→    }
   585→   585→
   586→   586→    *rpo_count = n - rpo_idx;
   587→   587→    if (rpo_idx > 0) {
   588→   588→        /* Some blocks were unreachable; shift the array */
   589→   589→        memmove(rpo, rpo + rpo_idx, *rpo_count * sizeof(uint32_t));
   590→   590→    }
   591→   591→
   592→   592→    return rpo;
   593→   593→}

   595→   595→/* ========================================================================== */
   596→   596→/* Main analysis entry point                                                   */
   597→   597→/* ========================================================================== */
   598→   598→
   599→   599→vtx_pea_analysis_t *vtx_pea_run(vtx_graph_t *graph, vtx_arena_t *arena)
   600→   600→{
   601→   601→    VTX_ASSERT(graph != NULL, "graph must not be NULL");
   602→   602→    VTX_ASSERT(arena != NULL, "arena must not be NULL");
   603→   603→
   604→   604→    if (graph->block_count == 0) {
   605→   605→        /* No blocks — degenerate graph */
   606→   606→        return NULL;
   607→   607→    }
   608→   608→
   609→   609→    /* Allocate the result */
   610→   610→    vtx_pea_analysis_t *result = vtx_arena_alloc(arena, sizeof(vtx_pea_analysis_t));
   611→   611→    if (!result) return NULL;
   612→   612→    memset(result, 0, sizeof(*result));
   613→   613→
   614→   614→    /* Step 1: Identify all allocations */
   615→   615→    if (build_allocation_map(graph, &result->escape_map, arena) != 0) {
   616→   616→        return NULL;
   617→   617→    }
   618→   618→
   619→   619→    /* Step 2: Build per-block state arrays */
   620→   620→    if (build_block_states(graph, result, arena) != 0) {
   621→   621→        return NULL;
   622→   622→    }
   623→   623→
   624→   624→    vtx_node_table_t *table = &graph->node_table;
   625→   625→    uint32_t state_count = result->escape_map.state_count;
   626→   626→    uint32_t block_count = graph->block_count;
   627→   627→
   628→   628→    /* Step 3: Compute reverse postorder for efficient worklist */
   629→   629→    uint32_t rpo_count = 0;
   630→   630→    uint32_t *rpo = compute_reverse_postorder(graph, arena, &rpo_count);
   631→   631→
   632→   632→    /* §2.1: Pre-compute block-to-node lists to avoid O(N×B) scan
   633→   633→     * in transfer_block. This is built once before the worklist loop.
   634→   634→     * GraalVM: BlockBag in EscapeAnalysisPhase.
   635→   635→     * V8: Schedule::rpo_order in EscapeAnalysis. */
   636→   636→    vtx_block_node_list_t *block_lists = build_block_node_lists(graph, arena);
   637→   637→    if (!block_lists) return NULL;
   638→   638→
   639→   639→    /* Step 4: Initialize worklist with all blocks in RPO */
   640→   640→    /* We use a simple flag-based worklist: a block is on the worklist
   641→   641→     * if its `entry_changed` flag is true. */
   642→   642→    bool *on_worklist = vtx_arena_alloc(arena, block_count * sizeof(bool));
   643→   643→    if (!on_worklist) return NULL;
   644→   644→    memset(on_worklist, 0, block_count * sizeof(bool));
   645→   645→
   646→   646→    /* Circular buffer worklist */
   647→   647→    uint32_t wl_capacity = block_count + 1;
   648→   648→    uint32_t *worklist = vtx_arena_alloc(arena, wl_capacity * sizeof(uint32_t));
   649→   649→    if (!worklist) return NULL;
   650→   650→
   651→   651→    uint32_t wl_head = 0, wl_tail = 0;
   652→   652→
   653→   653→    /* Enqueue all blocks in RPO */
   654→   654→    for (uint32_t i = 0; i < rpo_count; i++) {
   655→   655→        uint32_t block_idx = rpo[i];
   656→   656→        worklist[wl_tail] = block_idx;
   657→   657→        wl_tail = (wl_tail + 1) % wl_capacity;
   658→   658→        on_worklist[block_idx] = true;
   659→   659→    }
   660→   660→
   661→   661→    /* Step 5: Iterate to fixed point.
   662→   662→     *
   663→   663→     * BUGFIX (audit #4): The old hard cap of 100 iterations was too low
   664→   664→     * for large CFGs. If the worklist hadn't converged by iteration 100,
   665→   665→     * the escape states were left in an intermediate (wrong) state.
   666→   666→     * Allocations that should be GlobalEscape might be marked NoEscape
   667→   667→     * → heap corruption from scalar replacement.
   668→   668→     *
   669→   669→     * Fix: Increase to 10000 (sufficient for any real-world CFG), and
   670→   670→     * if we hit the cap, conservatively mark ALL allocations as
   671→   671→     * GlobalEscape (safe fallback — no scalar replacement). */
   672→   672→    uint32_t iterations = 0;
   673→   673→    const uint32_t MAX_ITERATIONS = 10000; /* safety bound — was 100 */
   674→   674→    bool hit_cap = false;
   675→   675→
   676→   676→    while (wl_head != wl_tail && iterations < MAX_ITERATIONS) {
   677→   677→        iterations++;
   678→   678→
   679→   679→        uint32_t block_idx = worklist[wl_head];
   680→   680→        wl_head = (wl_head + 1) % wl_capacity;
   681→   681→        on_worklist[block_idx] = false;
   682→   682→
   683→   683→        vtx_pea_block_state_t *bs = &result->block_states[block_idx];
   684→   684→        vtx_block_info_t *block = &graph->blocks[block_idx];
   685→   685→
   686→   686→        /* Compute entry state as join of predecessor exit states */
   687→   687→        vtx_escape_state_t *new_entry = vtx_arena_alloc(arena,
   688→   688→            state_count * sizeof(vtx_escape_state_t));
   689→   689→        if (!new_entry) return NULL;
   690→   690→        memset(new_entry, 0, state_count * sizeof(vtx_escape_state_t));
   691→   691→
   692→   692→        if (block->pred_count == 0) {
   693→   693→            /* Entry block: entry state is all NoEscape (already zeroed) */
   694→   694→        } else {
   695→   695→            /* Join all predecessor exit states */
   696→   696→            for (uint32_t p = 0; p < block->pred_count; p++) {
   697→   697→                uint32_t pred_idx = block->pred_indices[p];
   698→   698→                if (pred_idx < block_count) {
   699→   699→                    join_escape_states(new_entry,
   700→   700→                                       result->block_states[pred_idx].exit_state,
   701→   701→                                       state_count);
   702→   702→                }
   703→   703→            }
   704→   704→        }
   705→   705→
   706→   706→        /* Check if entry state changed */
   707→   707→        bool entry_changed = escape_states_differ(new_entry, bs->entry_state,
   708→   708→                                                    state_count);
   709→   709→        if (entry_changed) {
   710→   710→            copy_escape_states(bs->entry_state, new_entry, state_count);
   711→   711→        }
   712→   712→
   713→   713→        /* Apply transfer function */
   714→   714→        vtx_escape_state_t *old_exit = vtx_arena_alloc(arena,
   715→   715→            state_count * sizeof(vtx_escape_state_t));
   716→   716→        if (!old_exit) return NULL;
   717→   717→        copy_escape_states(old_exit, bs->exit_state, state_count);
   718→   718→
   719→   719→        /* §2.1: Use fast transfer with pre-computed block-node lists */
   720→   720→        transfer_block_fast(graph, block_idx, bs, table, block_lists);
   721→   721→
   722→   722→        bool exit_changed = escape_states_differ(bs->exit_state, old_exit,
   723→   723→                                                   state_count);
   724→   724→
   725→   725→        /* If exit state changed, add successors to worklist */
   726→   726→        if (entry_changed || exit_changed) {
   727→   727→            for (uint32_t s = 0; s < block->succ_count; s++) {
   728→   728→                uint32_t succ_idx = block->succ_indices[s];
   729→   729→                if (succ_idx < block_count && !on_worklist[succ_idx]) {
   730→   730→                    worklist[wl_tail] = succ_idx;
   731→   731→                    wl_tail = (wl_tail + 1) % wl_capacity;
   732→   732→                    on_worklist[succ_idx] = true;
   733→   733→                }
   734→   734→            }
   735→   735→        }
   736→   736→    }
   737→   737→
   738→   738→    result->iterations = iterations;
   739→   739→
   740→   740→    /* If we hit the iteration cap, the escape states may not have
   741→   741→     * converged. Conservatively mark ALL allocations as GlobalEscape
   742→   742→     * to prevent incorrect scalar replacement. */
   743→   743→    if (iterations >= MAX_ITERATIONS) {
   744→   744→        hit_cap = true;
   745→   745→        for (uint32_t a = 0; a < result->escape_map.alloc_count; a++) {
   746→   746→            vtx_nodeid_t alloc_id = result->escape_map.alloc_ids[a];
   747→   747→            for (uint32_t b = 0; b < block_count; b++) {
   748→   748→                if (alloc_id < result->block_states[b].state_count) {
   749→   749→                    result->block_states[b].exit_state[alloc_id] = VTX_ESCAPE_GLOBAL;
   750→   750→                    result->block_states[b].entry_state[alloc_id] = VTX_ESCAPE_GLOBAL;
   751→   751→                }
   752→   752→            }
   753→   753→        }
   754→   754→    }
   755→   755→
   756→   756→    /* Step 6: Finalize — compute the global escape state for each allocation
   757→   757→     * as the join of all block exit states */
   758→   758→    for (uint32_t a = 0; a < result->escape_map.alloc_count; a++) {
   759→   759→        vtx_nodeid_t alloc_id = result->escape_map.alloc_ids[a];
   760→   760→        vtx_escape_state_t final_state = VTX_ESCAPE_NONE;
   761→   761→        for (uint32_t b = 0; b < block_count; b++) {
   762→   762→            if (alloc_id < result->block_states[b].state_count) {
   763→   763→                final_state = vtx_escape_join(final_state,
   764→   764→                    result->block_states[b].exit_state[alloc_id]);
   765→   765→            }
   766→   766→        }
   767→   767→
   768→   768→        result->escape_map.states[alloc_id] = final_state;
   769→   769→
   770→   770→        /* Update statistics */
   771→   771→        result->total_allocs++;
   772→   772→        switch (final_state) {
   773→   773→        case VTX_ESCAPE_NONE:   result->no_escape_count++; break;
   774→   774→        case VTX_ESCAPE_ARG:    result->arg_escape_count++; break;
   775→   775→        case VTX_ESCAPE_GLOBAL: result->global_escape_count++; break;
   776→   776→        }
   777→   777→    }
   778→   778→
   779→   779→    return result;
   780→   780→}
