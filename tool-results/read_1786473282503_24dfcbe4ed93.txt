    25→/* ========================================================================== */
    26→/* Internal: check if opcode is an allocation                                  */
    27→/* ========================================================================== */
    28→
    29→static inline bool is_allocation(vtx_node_opcode_t opcode)
    30→{
    31→    return opcode == VTX_OP_NewObject ||
    32→           opcode == VTX_OP_NewArray  ||
    33→           opcode == VTX_OP_Allocate;
    34→}

    36→/* ========================================================================== */
    37→/* Internal: grow virtual obj array                                            */
    38→/* ========================================================================== */
    39→
    40→static vtx_virtual_obj_t *grow_virtual_objs(vtx_virtual_result_t *result,
    41→                                              vtx_arena_t *arena)
    42→{
    43→    if (result->virtual_obj_count >= result->virtual_obj_capacity) {
    44→        uint32_t new_cap = result->virtual_obj_capacity == 0 ?
    45→                           16 : result->virtual_obj_capacity * 2;
    46→        vtx_virtual_obj_t *new_objs = vtx_arena_alloc(arena,
    47→            new_cap * sizeof(vtx_virtual_obj_t));
    48→        if (!new_objs) return NULL;
    49→        if (result->virtual_objs && result->virtual_obj_count > 0) {
    50→            memcpy(new_objs, result->virtual_objs,
    51→                   result->virtual_obj_count * sizeof(vtx_virtual_obj_t));
    52→        }
    53→        result->virtual_objs = new_objs;
    54→        result->virtual_obj_capacity = new_cap;
    55→    }
    56→    return &result->virtual_objs[result->virtual_obj_count];
    57→}

    59→/* ========================================================================== */
    60→/* Internal: find or create virtual obj entry                                  */
    61→/* ========================================================================== */
    62→
    63→static vtx_virtual_obj_t *find_or_create_virtual_obj(
    64→    vtx_virtual_result_t *result, vtx_nodeid_t alloc_id,
    65→    uint32_t type_id, vtx_arena_t *arena)
    66→{
    67→    /* Check if already exists */
    68→    for (uint32_t i = 0; i < result->virtual_obj_count; i++) {
    69→        if (result->virtual_objs[i].alloc_id == alloc_id) {
    70→            return &result->virtual_objs[i];
    71→        }
    72→    }
    73→
    74→    /* Create new entry.
    75→     * B17 fix: Do NOT set state=VTX_VIRTUAL_YES here. The caller is
    76→     * responsible for setting the state once the object is fully
    77→     * initialized. For allocations (classify_allocations), state is
    78→     * set to YES immediately after this call. For Phi nodes
    79→     * (resolve_virtual_phis), state is set to YES only AFTER the
    80→     * per-field Phi nodes have been created and the field_values
    81→     * array has been populated.
    82→     *
    83→     * The previous code unconditionally set state=YES on creation,
    84→     * which caused resolve_virtual_phis's `if (obj->state == YES)
    85→     * continue;` check to skip every newly-created Phi — the field
    86→     * resolution logic was dead code, and Phis of virtual objects
    87→     * were left with empty field_values arrays, producing wrong
    88→     * materialization later. */
    89→    vtx_virtual_obj_t *obj = grow_virtual_objs(result, arena);
    90→    if (!obj) return NULL;
    91→
    92→    memset(obj, 0, sizeof(*obj));
    93→    obj->alloc_id = alloc_id;
    94→    obj->type_id  = type_id;
    95→    obj->state    = VTX_VIRTUAL_UNKNOWN;  /* caller sets to YES when ready */
    96→
    97→    /* Allocate field arrays with initial capacity */
    98→    obj->field_capacity = 8;
    99→    obj->field_offsets = vtx_arena_alloc(arena,
   100→        obj->field_capacity * sizeof(uint32_t));
   101→    obj->field_values  = vtx_arena_alloc(arena,
   102→        obj->field_capacity * sizeof(vtx_nodeid_t));
   103→    if (!obj->field_offsets || !obj->field_values) return NULL;
   104→
   105→    result->virtual_obj_count++;
   106→    return obj;
   107→}

   109→/* ========================================================================== */
   110→/* Internal: set field value in virtual object                                 */
   111→/* ========================================================================== */
   112→
   113→static int virtual_obj_set_field(vtx_virtual_obj_t *obj,
   114→                                  uint32_t field_offset,
   115→                                  vtx_nodeid_t value_id,
   116→                                  vtx_arena_t *arena)
   117→{
   118→    /* Check if field already exists */
   119→    for (uint32_t i = 0; i < obj->field_count; i++) {
   120→        if (obj->field_offsets[i] == field_offset) {
   121→            obj->field_values[i] = value_id;
   122→            return 0;
   123→        }
   124→    }
   125→
   126→    /* Add new field */
   127→    if (obj->field_count >= obj->field_capacity) {
   128→        uint32_t new_cap = obj->field_capacity * 2;
   129→        uint32_t *new_offsets = vtx_arena_alloc(arena,
   130→            new_cap * sizeof(uint32_t));
   131→        vtx_nodeid_t *new_values = vtx_arena_alloc(arena,
   132→            new_cap * sizeof(vtx_nodeid_t));
   133→        if (!new_offsets || !new_values) return -1;
   134→        if (obj->field_offsets && obj->field_count > 0) {
   135→            memcpy(new_offsets, obj->field_offsets,
   136→                   obj->field_count * sizeof(uint32_t));
   137→            memcpy(new_values, obj->field_values,
   138→                   obj->field_count * sizeof(vtx_nodeid_t));
   139→        }
   140→        obj->field_offsets  = new_offsets;
   141→        obj->field_values   = new_values;
   142→        obj->field_capacity = new_cap;
   143→    }
   144→
   145→    obj->field_offsets[obj->field_count] = field_offset;
   146→    obj->field_values[obj->field_count]  = value_id;
   147→    obj->field_count++;
   148→    return 0;
   149→}

   151→/* ========================================================================== */
   152→/* Internal: classify allocations as virtual or not                            */
   153→/* ========================================================================== */
   154→
   155→static int classify_allocations(vtx_graph_t *graph,
   156→                                 const vtx_pea_analysis_t *analysis,
   157→                                 vtx_virtual_result_t *result,
   158→                                 vtx_arena_t *arena)
   159→{
   160→    vtx_node_table_t *table = &graph->node_table;
   161→    uint32_t state_count = table->count;
   162→
   163→    result->state_count = state_count;
   164→    result->virtual_states = vtx_arena_alloc(arena,
   165→        state_count * sizeof(vtx_virtual_state_t));
   166→    if (!result->virtual_states) return -1;
   167→
   168→    /* Initialize all to UNKNOWN */
   169→    memset(result->virtual_states, 0, state_count * sizeof(vtx_virtual_state_t));
   170→
   171→    /* Classify each allocation */
   172→    for (uint32_t a = 0; a < analysis->escape_map.alloc_count; a++) {
   173→        vtx_nodeid_t alloc_id = analysis->escape_map.alloc_ids[a];
   174→        vtx_escape_state_t escape = vtx_pea_get_escape(analysis, alloc_id);
   175→
   176→        result->total_allocs++;
   177→
   178→        if (escape == VTX_ESCAPE_NONE) {
   179→            /* Non-escaping allocation → virtual */
   180→            result->virtual_states[alloc_id] = VTX_VIRTUAL_YES;
   181→            result->virtual_count++;
   182→
   183→            vtx_node_t *alloc_node = vtx_node_get(table, alloc_id);
   184→            uint32_t type_id = alloc_node ? alloc_node->type_id : 0;
   185→
   186→            vtx_virtual_obj_t *obj = find_or_create_virtual_obj(
   187→                result, alloc_id, type_id, arena);
   188→            if (!obj) return -1;
   189→            /* B17 fix: find_or_create_virtual_obj no longer sets
   190→             * state=YES — set it here for confirmed-virtual allocations. */
   191→            obj->state = VTX_VIRTUAL_YES;
   192→        } else {
   193→            /* Escaping allocation → not virtual */
   194→            result->virtual_states[alloc_id] = VTX_VIRTUAL_NO;
   195→            result->non_virtual_count++;
   196→        }
   197→    }
   198→
   199→    return 0;
   200→}

   202→/* ========================================================================== */
   203→/* Internal: resolve Phi nodes for virtual objects                             */
   204→/* ========================================================================== */
   205→
   206→/**
   207→ * Process Phi nodes that reference virtual objects. A Phi of two virtual
   208→ * objects is virtual if both inputs are virtual with the same type.
   209→ * Otherwise, the inputs must be materialized (non-virtual).
   210→ *
   211→ * Returns the number of Phis resolved as virtual.
   212→ */
   213→static uint32_t resolve_virtual_phis(vtx_graph_t *graph,
   214→                                      vtx_virtual_result_t *result,
   215→                                      vtx_arena_t *arena)
   216→{
   217→    vtx_node_table_t *table = &graph->node_table;
   218→    uint32_t resolved = 0;
   219→
   220→    /* Iterate until no more changes (Phi resolution may cascade) */
   221→    bool changed = true;
   222→    while (changed) {
   223→        changed = false;
   224→
   225→        for (uint32_t i = 0; i < table->count; i++) {
   226→            vtx_node_t *node = &table->nodes[i];
   227→            if (node->dead || node->opcode != VTX_OP_Phi) continue;
   228→
   229→            /* Check if all data inputs are virtual with the same type.
   230→             * Fix C20: The old code started at inp=0, but input[0] of a
   231→             * Phi is the Region/LoopBegin control input — not a data input.
   232→             * This immediately set all_virtual=false because Region is not
   233→             * an allocation. Result: no Phi was ever classified as virtual.
   234→             * Fix: skip control/memory inputs. */
   235→            bool all_virtual = true;
   236→            uint32_t common_type = 0;
   237→            bool first_input = true;
   238→
   239→            for (uint32_t inp = 0; inp < node->input_count; inp++) {
   240→                vtx_nodeid_t input_id = node->inputs[inp];
   241→                if (input_id >= result->state_count) {
   242→                    continue; /* skip out-of-range inputs */
   243→                }
   244→
   245→                vtx_node_t *input_node = vtx_node_get(table, input_id);
   246→                if (!input_node || input_node->dead) {
   247→                    continue;
   248→                }
   249→
   250→                /* Skip control inputs (Region, LoopBegin, Proj) */
   251→                if (vtx_nf_has(input_node->flags, VTX_NF_CONTROL)) {
   252→                    continue;
   253→                }
   254→                /* Skip memory inputs */
   255→                if (vtx_nf_has(input_node->flags, VTX_NF_MEMORY)) {
   256→                    continue;
   257→                }
   258→
   259→                /* Check if this input is a virtual allocation */
   260→                if (!is_allocation(input_node->opcode)) {
   261→                    /* Non-allocation input (e.g., null constant).
   262→                     * If it's a null, the Phi can still be virtual. */
   263→                    if (input_node->opcode == VTX_OP_Constant &&
   264→                        input_node->constval.kind == VTX_TYPE_Ptr &&
   265→                        input_node->constval.as.ptr_val == NULL) {
   266→                        /* null input — ok for virtual Phi */
   267→                        continue;
   268→                    }
   269→                    all_virtual = false;
   270→                    break;
   271→                }
   272→
   273→                if (result->virtual_states[input_id] != VTX_VIRTUAL_YES) {
   274→                    all_virtual = false;
   275→                    break;
   276→                }
   277→
   278→                /* Check type compatibility */
   279→                if (first_input) {
   280→                    common_type = input_node->type_id;
   281→                    first_input = false;
   282→                } else if (input_node->type_id != common_type) {
   283→                    /* Different types — cannot be virtual */
   284→                    all_virtual = false;
   285→                    break;
   286→                }
   287→            }
   288→
   289→            if (all_virtual && node->input_count > 0) {
   290→                /* The Phi is virtual. The Phi itself represents a virtual
   291→                 * object that merges the field values of its inputs.
   292→                 * For each field, create a new Phi that merges the
   293→                 * corresponding field values. */
   294→                vtx_virtual_obj_t *obj = find_or_create_virtual_obj(
   295→                    result, node->id, common_type, arena);
   296→                if (!obj) break;
   297→
   298→                if (obj->state == VTX_VIRTUAL_YES) {
   299→                    /* Already processed */
   300→                    continue;
   301→                }
   302→
   303→                /* Collect fields from the first virtual input */
   304→                vtx_virtual_obj_t *first_vobj = NULL;
   305→                for (uint32_t inp = 0; inp < node->input_count; inp++) {
   306→                    vtx_nodeid_t input_id = node->inputs[inp];
   307→                    if (input_id < result->state_count &&
   308→                        result->virtual_states[input_id] == VTX_VIRTUAL_YES) {
   309→                        for (uint32_t v = 0; v < result->virtual_obj_count; v++) {
   310→                            if (result->virtual_objs[v].alloc_id == input_id) {
   311→                                first_vobj = &result->virtual_objs[v];
   312→                                break;
   313→                            }
   314→                        }
   315→                        if (first_vobj) break;
   316→                    }
   317→                }
   318→
   319→                if (!first_vobj) continue;
   320→
   321→                /* PEA-2-2: the per-field Phi creation loop below calls
   322→                 * vtx_node_create(table, VTX_OP_Phi) and vtx_node_create(
   323→                 * table, VTX_OP_Constant), both of which may realloc
   324→                 * table->nodes. The `node` pointer captured at the top of
   325→                 * the outer loop (line 226) would become dangling; reads
   326→                 * of node->inputs[inp] inside the inner loop and the read
   327→                 * of node->id after the field loop would dereference
   328→                 * freed memory.
   329→                 *
   330→                 * Save the invariant data we need to read across the
   331→                 * creates (NodeID and input_count) so we can re-fetch
   332→                 * `node` after each vtx_node_create call. The input_count
   333→                 * does not change as a side effect of vtx_node_create
   334→                 * (creates add a NEW node with input_count=0), so caching
   335→                 * it is safe. */
   336→                vtx_nodeid_t node_id = node->id;
   337→                uint32_t input_count = node->input_count;
   338→
   339→                /* For each field, create a Phi that merges the values
   340→                 * from all virtual inputs */
   341→                for (uint32_t f = 0; f < first_vobj->field_count; f++) {
   342→                    uint32_t field_offset = first_vobj->field_offsets[f];
   343→
   344→                    /* Create a new Phi node for this field */
   345→                    vtx_nodeid_t field_phi_id = vtx_node_create(table, VTX_OP_Phi);
   346→                    if (field_phi_id == VTX_NODEID_INVALID) break;
   347→
   348→                    vtx_node_t *field_phi = vtx_node_get(table, field_phi_id);
   349→                    field_phi->flags = VTX_NF_PINNED;
   350→
   351→                    /* PEA-2-2: vtx_node_create above may have realloc'd
   352→                     * table->nodes. Re-fetch `node` so the inner loop
   353→                     * reads node->inputs[inp] from the live array. */
   354→                    node = vtx_node_get(table, node_id);
   355→                    if (!node) break;
   356→
   357→                    /* Add inputs from each virtual object's field value */
   358→                    for (uint32_t inp = 0; inp < input_count; inp++) {
   359→                        vtx_nodeid_t input_id = node->inputs[inp];
   360→                        vtx_nodeid_t field_value = VTX_NODEID_INVALID;
   361→
   362→                        if (input_id < result->state_count &&
   363→                            result->virtual_states[input_id] == VTX_VIRTUAL_YES) {
   364→                            /* Find the field value in this virtual object */
   365→                            for (uint32_t v = 0; v < result->virtual_obj_count; v++) {
   366→                                if (result->virtual_objs[v].alloc_id == input_id) {
   367→                                    for (uint32_t ff = 0; ff < result->virtual_objs[v].field_count; ff++) {
   368→                                        if (result->virtual_objs[v].field_offsets[ff] == field_offset) {
   369→                                            field_value = result->virtual_objs[v].field_values[ff];
   370→                                            break;
   371→                                        }
   372→                                    }
   373→                                    break;
   374→                                }
   375→                            }
   376→                        }
   377→
   378→                        /* If we couldn't find the field value, check if
   379→                         * this input is a control node (Region/LoopBegin).
   380→                         * Control inputs must pass through to the field Phi
   381→                         * directly — the field Phi needs the same control
   382→                         * inputs as the original Phi to be structurally
   383→                         * valid and schedulable.
   384→                         *
   385→                         * PEA-2-3: previously ALL non-virtual inputs
   386→                         * (including Region/LoopBegin) got a NULL constant,
   387→                         * producing a structurally invalid Phi with
   388→                         * [null_const, val_0, val_1, ...] instead of
   389→                         * [val_0, val_1, ..., Region]. The scheduler
   390→                         * couldn't pin the Phi to the merge point and
   391→                         * the verifier would reject the graph. */
   392→                        if (field_value == VTX_NODEID_INVALID) {
   393→                            vtx_node_t *input_node = vtx_node_get(table, input_id);
   394→                            if (input_node &&
   395→                                (input_node->opcode == VTX_OP_Region ||
   396→                                 input_node->opcode == VTX_OP_LoopBegin)) {
   397→                                /* Pass the control input through directly. */
   398→                                field_value = input_id;
   399→                            }
   400→                        }
   401→
   402→                        /* If still no field value (non-virtual data value
   403→                         * that's not a control node), use a null constant. */
   404→                        if (field_value == VTX_NODEID_INVALID) {
   405→                            vtx_nodeid_t null_id = vtx_node_create(table, VTX_OP_Constant);
   406→                            if (null_id != VTX_NODEID_INVALID) {
   407→                                vtx_node_t *null_node = vtx_node_get(table, null_id);
   408→                                null_node->type = VTX_TYPE_Ptr;
   409→                                null_node->constval = vtx_constval_ptr(NULL);
   410→                                field_value = null_id;
   411→                            }
   412→                        }
   413→
   414→                        /* PEA-2-2: vtx_node_create above (for the null
   415→                         * constant) may realloc table->nodes again.
   416→                         * Re-fetch `node` so the next iteration reads
   417→                         * node->inputs[inp] from the live array. */
   418→                        node = vtx_node_get(table, node_id);
   419→
   420→                        vtx_node_add_input(table, field_phi_id, field_value);
   421→                    }
   422→
   423→                    /* Record the field in the virtual object */
   424→                    virtual_obj_set_field(obj, field_offset, field_phi_id, arena);
   425→                }
   426→
   427→                /* PEA-2-2: re-fetch `node` after the per-field create
   428→                 * loop — vtx_node_create calls inside the loop may have
   429→                 * realloc'd table->nodes, leaving `node` dangling. */
   430→                node = vtx_node_get(table, node_id);
   431→                obj->state = VTX_VIRTUAL_YES;
   432→                result->virtual_states[node_id] = VTX_VIRTUAL_YES;
   433→                result->virtual_count++;
   434→                resolved++;
   435→                changed = true;
   436→            } else {
   437→                /* Phi is not virtual — mark it as non-virtual */
   438→                if (node->id < result->state_count &&
   439→                    result->virtual_states[node->id] == VTX_VIRTUAL_UNKNOWN) {
   440→                    result->virtual_states[node->id] = VTX_VIRTUAL_NO;
   441→                }
   442→            }
   443→        }
   444→    }
   445→
   446→    return resolved;
   447→}

   449→/* ========================================================================== */
   450→/* Internal: rewrite field accesses for virtual objects                        */
   451→/* ========================================================================== */
   452→
   453→static uint32_t rewrite_virtual_field_accesses(vtx_graph_t *graph,
   454→                                                 vtx_virtual_result_t *result,
   455→                                                 vtx_arena_t *arena)
   456→{
   457→    vtx_node_table_t *table = &graph->node_table;
   458→    uint32_t rewritten = 0;
   459→
   460→    for (uint32_t i = 0; i < table->count; i++) {
   461→        vtx_node_t *node = &table->nodes[i];
   462→        if (node->dead) continue;
   463→
   464→        if (node->opcode != VTX_OP_LoadField &&
   465→            node->opcode != VTX_OP_StoreField) continue;
   466→
   467→        /* Find the receiver input */
   468→        vtx_nodeid_t receiver_id = VTX_NODEID_INVALID;
   469→        if (node->opcode == VTX_OP_LoadField) {
   470→            if (node->input_count >= 1) {
   471→                receiver_id = node->inputs[node->input_count - 1];
   472→            }
   473→        } else {
   474→            /* StoreField: receiver is second-to-last input */
   475→            if (node->input_count >= 2) {
   476→                receiver_id = node->inputs[node->input_count - 2];
   477→            }
   478→        }
   479→
   480→        if (receiver_id == VTX_NODEID_INVALID) continue;
   481→        if (receiver_id >= result->state_count) continue;
   482→        if (result->virtual_states[receiver_id] != VTX_VIRTUAL_YES) continue;
   483→
   484→        /* The receiver is a virtual object — rewrite the field access */
   485→        vtx_virtual_obj_t *vobj = NULL;
   486→        for (uint32_t v = 0; v < result->virtual_obj_count; v++) {
   487→            if (result->virtual_objs[v].alloc_id == receiver_id) {
   488→                vobj = &result->virtual_objs[v];
   489→                break;
   490→            }
   491→        }
   492→
   493→        if (!vobj) continue;
   494→
   495→        if (node->opcode == VTX_OP_LoadField) {
   496→            /* Replace LoadField with the scalar local value */
   497→            vtx_nodeid_t field_value = VTX_NODEID_INVALID;
   498→            for (uint32_t f = 0; f < vobj->field_count; f++) {
   499→                if (vobj->field_offsets[f] == node->field_offset) {
   500→                    field_value = vobj->field_values[f];
   501→                    break;
   502→                }
   503→            }
   504→
   505→            if (field_value != VTX_NODEID_INVALID) {
   506→                /* Replace all uses of this LoadField with the field value */
   507→                for (uint32_t u = 0; u < table->count; u++) {
   508→                    vtx_node_t *user = &table->nodes[u];
   509→                    if (user->dead) continue;
   510→                    for (uint32_t inp = 0; inp < user->input_count; inp++) {
   511→                        if (user->inputs[inp] == node->id) {
   512→                            vtx_node_replace_input(table, user->id, inp,
   513→                                                    field_value);
   514→                        }
   515→                    }
   516→                }
   517→                node->dead = true;
   518→                rewritten++;
   519→            }
   520→        } else {
   521→            /* StoreField: update the virtual object's field value */
   522→            vtx_nodeid_t value_id = node->inputs[node->input_count - 1];
   523→
   524→            /* Update the field value in the virtual object */
   525→            bool found = false;
   526→            for (uint32_t f = 0; f < vobj->field_count; f++) {
   527→                if (vobj->field_offsets[f] == node->field_offset) {
   528→                    vobj->field_values[f] = value_id;
   529→                    found = true;
   530→                    break;
   531→                }
   532→            }
   533→
   534→            if (!found) {
   535→                /* D4 fix: New field — use arena allocation instead of realloc.
   536→                 * The field arrays were originally allocated from the arena,
   537→                 * so using realloc on arena memory is undefined behavior.
   538→                 * We allocate new arrays from the arena and copy the data. */
   539→                uint32_t new_cap = vobj->field_capacity * 2;
   540→                uint32_t *new_offsets = vtx_arena_alloc(arena,
   541→                    new_cap * sizeof(uint32_t));
   542→                vtx_nodeid_t *new_values = vtx_arena_alloc(arena,
   543→                    new_cap * sizeof(vtx_nodeid_t));
   544→                if (!new_offsets || !new_values) {
   545→                    /* Can't add the field — skip to avoid buffer overflow */
   546→                    node->dead = true;
   547→                    rewritten++;
   548→                    continue;
   549→                }
   550→                if (vobj->field_offsets && vobj->field_count > 0) {
   551→                    memcpy(new_offsets, vobj->field_offsets,
   552→                           vobj->field_count * sizeof(uint32_t));
   553→                    memcpy(new_values, vobj->field_values,
   554→                           vobj->field_count * sizeof(vtx_nodeid_t));
   555→                }
   556→                vobj->field_offsets = new_offsets;
   557→                vobj->field_values = new_values;
   558→                vobj->field_capacity = new_cap;
   559→                vobj->field_offsets[vobj->field_count] = node->field_offset;
   560→                vobj->field_values[vobj->field_count]  = value_id;
   561→                vobj->field_count++;
   562→            }
   563→
   564→            node->dead = true;
   565→            rewritten++;
   566→        }
   567→    }
   568→
   569→    return rewritten;
   570→}

   572→/* ========================================================================== */
   573→/* Internal: eliminate virtual allocation nodes                                */
   574→/* ========================================================================== */
   575→
   576→static uint32_t eliminate_virtual_allocations(vtx_graph_t *graph,
   577→                                                vtx_virtual_result_t *result)
   578→{
   579→    vtx_node_table_t *table = &graph->node_table;
   580→    uint32_t eliminated = 0;
   581→
   582→    for (uint32_t i = 0; i < table->count; i++) {
   583→        vtx_node_t *node = &table->nodes[i];
   584→        if (node->dead) continue;
   585→        if (!is_allocation(node->opcode)) continue;
   586→
   587→        if (node->id < result->state_count &&
   588→            result->virtual_states[node->id] == VTX_VIRTUAL_YES) {
   589→            /* Check if the allocation has any remaining non-dead uses.
   590→             * All field accesses have been rewritten, but there might be
   591→             * remaining uses (e.g., identity comparisons, monitor ops). */
   592→            bool has_uses = false;
   593→            for (uint32_t u = 0; u < table->count; u++) {
   594→                vtx_node_t *user = &table->nodes[u];
   595→                if (user->dead) continue;
   596→                if (user->id == node->id) continue;
   597→                for (uint32_t inp = 0; inp < user->input_count; inp++) {
   598→                    if (user->inputs[inp] == node->id) {
   599→                        has_uses = true;
   600→                        break;
   601→                    }
   602→                }
   603→                if (has_uses) break;
   604→            }
   605→
   606→            if (!has_uses) {
   607→                node->dead = true;
   608→                eliminated++;
   609→            }
   610→        }
   611→    }
   612→
   613→    return eliminated;
   614→}

   616→/* ========================================================================== */
   617→/* Main entry point                                                            */
   618→/* ========================================================================== */
   619→
   620→vtx_virtual_result_t *vtx_virtual_run(vtx_graph_t *graph,
   621→                                       const vtx_pea_analysis_t *analysis,
   622→                                       vtx_arena_t *arena)
   623→{
   624→    VTX_ASSERT(graph != NULL, "graph must not be NULL");
   625→    VTX_ASSERT(analysis != NULL, "analysis must not be NULL");
   626→    VTX_ASSERT(arena != NULL, "arena must not be NULL");
   627→
   628→    /* Allocate result */
   629→    vtx_virtual_result_t *result = vtx_arena_alloc(arena,
   630→        sizeof(vtx_virtual_result_t));
   631→    if (!result) return NULL;
   632→    memset(result, 0, sizeof(*result));
   633→
   634→    /* Step 1: Classify allocations as virtual or not */
   635→    if (classify_allocations(graph, analysis, result, arena) != 0) {
   636→        return NULL;
   637→    }
   638→
   639→    /* Step 2: Collect initial field values from StoreField nodes */
   640→    vtx_node_table_t *table = &graph->node_table;
   641→
   642→    for (uint32_t i = 0; i < table->count; i++) {
   643→        vtx_node_t *node = &table->nodes[i];
   644→        if (node->dead || node->opcode != VTX_OP_StoreField) continue;
   645→        if (node->input_count < 2) continue;
   646→
   647→        vtx_nodeid_t receiver_id = node->inputs[node->input_count - 2];
   648→        vtx_nodeid_t value_id    = node->inputs[node->input_count - 1];
   649→
   650→        if (receiver_id >= result->state_count) continue;
   651→        if (result->virtual_states[receiver_id] != VTX_VIRTUAL_YES) continue;
   652→
   653→        /* Find the virtual object */
   654→        vtx_virtual_obj_t *vobj = NULL;
   655→        for (uint32_t v = 0; v < result->virtual_obj_count; v++) {
   656→            if (result->virtual_objs[v].alloc_id == receiver_id) {
   657→                vobj = &result->virtual_objs[v];
   658→                break;
   659→            }
   660→        }
   661→
   662→        if (!vobj) continue;
   663→
   664→        /* Update the field value */
   665→        virtual_obj_set_field(vobj, node->field_offset, value_id, arena);
   666→    }
   667→
   668→    /* Step 3: Resolve virtual Phis */
   669→    result->phis_resolved = resolve_virtual_phis(graph, result, arena);
   670→
   671→    /* Step 4: Rewrite field accesses */
   672→    result->field_accesses_rewritten = rewrite_virtual_field_accesses(
   673→        graph, result, arena);
   674→
   675→    /* Step 5: Eliminate virtual allocation nodes */
   676→    eliminate_virtual_allocations(graph, result);
   677→
   678→    return result;
   679→}

   681→/* ========================================================================== */
   682→/* Query helpers                                                               */
   683→/* ========================================================================== */
   684→
   685→bool vtx_virtual_is_virtual(const vtx_virtual_result_t *result,
   686→                             vtx_nodeid_t alloc_id)
   687→{
   688→    VTX_ASSERT(result != NULL, "result must not be NULL");
   689→    if (alloc_id < result->state_count) {
   690→        return result->virtual_states[alloc_id] == VTX_VIRTUAL_YES;
   691→    }
   692→    return false;
   693→}
   694→
   695→vtx_nodeid_t vtx_virtual_get_field(const vtx_virtual_result_t *result,
   696→                                    vtx_nodeid_t alloc_id,
   697→                                    uint32_t field_offset)
   698→{
   699→    VTX_ASSERT(result != NULL, "result must not be NULL");
   700→    for (uint32_t i = 0; i < result->virtual_obj_count; i++) {
   701→        if (result->virtual_objs[i].alloc_id == alloc_id) {
   702→            const vtx_virtual_obj_t *obj = &result->virtual_objs[i];
   703→            for (uint32_t f = 0; f < obj->field_count; f++) {
   704→                if (obj->field_offsets[f] == field_offset) {
   705→                    return obj->field_values[f];
   706→                }
   707→            }
   708→            break;
   709→        }
   710→    }
   711→    return VTX_NODEID_INVALID;
   712→}
   713→
   714→const vtx_virtual_obj_t *vtx_virtual_get_obj(const vtx_virtual_result_t *result,
   715→                                               vtx_nodeid_t alloc_id)
   716→{
   717→    VTX_ASSERT(result != NULL, "result must not be NULL");
   718→    for (uint32_t i = 0; i < result->virtual_obj_count; i++) {
   719→        if (result->virtual_objs[i].alloc_id == alloc_id) {
   720→            return &result->virtual_objs[i];
   721→        }
   722→    }
   723→    return NULL;
   724→}
