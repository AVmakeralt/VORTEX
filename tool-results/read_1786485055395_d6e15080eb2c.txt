    55→    55→static int build_allocation_map(vtx_graph_t *graph, vtx_escape_map_t *map,
    56→    56→                                 vtx_arena_t *arena)
    57→    57→{
    58→    58→    vtx_node_table_t *table = &graph->node_table;
    59→    59→    uint32_t count = table->count;
    60→    60→
    61→    61→    map->state_count = count;
    62→    62→    map->states = vtx_arena_alloc(arena, count * sizeof(vtx_escape_state_t));
    63→    63→    if (!map->states) return -1;
    64→    64→
    65→    65→    /* PEA-1-2: Initialize ALL states to VTX_ESCAPE_GLOBAL (conservative
    66→    66→     * default for non-allocations). Only allocations get their actual
    67→    67→     * state set in the finalization step (line ~768). Previously this
    68→    68→     * was memset to 0 (NoEscape), which meant vtx_pea_get_escape()
    69→    69→     * returned NoEscape for non-allocations — breaking the public API
    70→    70→     * contract ("returns GlobalEscape for non-allocations") and causing
    71→    71→     * vtx_pea_is_scalar_replaceable() to incorrectly return true for
    72→    72→     * Constants, Parameters, Phis, etc. */
    73→    73→    for (uint32_t s = 0; s < count; s++) {
    74→    74→        map->states[s] = VTX_ESCAPE_GLOBAL;
    75→    75→    }
    76→    76→
    77→    77→    /* Count allocations first for capacity */
    78→    78→    uint32_t alloc_count = 0;
    79→    79→    for (uint32_t i = 0; i < count; i++) {
    80→    80→        vtx_node_t *node = &table->nodes[i];
    81→    81→        if (!node->dead && is_allocation(node->opcode)) {
    82→    82→            alloc_count++;
    83→    83→        }
    84→    84→    }
    85→    85→
    86→    86→    map->alloc_capacity = alloc_count > 0 ? alloc_count : 1;
    87→    87→    map->alloc_ids = vtx_arena_alloc(arena, map->alloc_capacity * sizeof(vtx_nodeid_t));
    88→    88→    if (!map->alloc_ids) return -1;
    89→    89→
    90→    90→    map->alloc_count = 0;
    91→    91→    for (uint32_t i = 0; i < count; i++) {
    92→    92→        vtx_node_t *node = &table->nodes[i];
    93→    93→        if (!node->dead && is_allocation(node->opcode)) {
    94→    94→            map->alloc_ids[map->alloc_count++] = node->id;
    95→    95→        }
    96→    96→    }
    97→    97→
    98→    98→    return 0;
    99→    99→}

   211→   211→/* Internal: transfer function for a single node                               */
   212→   212→/*                                                                           */
   213→   213→/* Examines how the node uses values and updates escape states accordingly.   */
   214→   214→/* ========================================================================== */
   215→   215→
   216→   216→/* PEA-1-1: recursively propagate escape state through Phi chains.
   217→   217→ *
   218→   218→ * When an escape point (Return/StoreField/Call*) encounters a Phi as
   219→   219→ * input, the Phi itself is not an allocation, so the old code skipped
   220→   220→ * it — the escape never flowed back to the Phi's allocation inputs.
   221→   221→ * This function walks through a Phi's inputs (and recursively through
   222→   222→ * nested Phis) to find all allocation inputs and propagate the escape
   223→   223→ * state to them.
   224→   224→ *
   225→   225→ * `depth` limits recursion to prevent infinite loops on cyclic Phi
   226→   226→ * graphs (shouldn't happen in SSA, but defensive). */
   227→   227→static void propagate_escape_through_phi(vtx_node_table_t *table,
   228→   228→                                          vtx_escape_state_t *state,
   229→   229→                                          uint32_t state_count,
   230→   230→                                          vtx_nodeid_t phi_id,
   231→   231→                                          vtx_escape_state_t escape_state,
   232→   232→                                          uint32_t depth)
   233→   233→{
   234→   234→    if (depth > 8) return;  /* recursion limit */
   235→   235→    if (phi_id >= state_count) return;
   236→   236→
   237→   237→    vtx_node_t *phi = vtx_node_get(table, phi_id);
   238→   238→    if (!phi || phi->dead || phi->opcode != VTX_OP_Phi) return;
   239→   239→
   240→   240→    for (uint32_t i = 0; i < phi->input_count; i++) {
   241→   241→        vtx_nodeid_t input_id = phi->inputs[i];
   242→   242→        if (input_id >= state_count) continue;
   243→   243→
   244→   244→        vtx_node_t *input = vtx_node_get(table, input_id);
   245→   245→        if (!input || input->dead) continue;
   246→   246→
   247→   247→        if (is_allocation(input->opcode)) {
   248→   248→            /* Found an allocation input — propagate the escape state. */
   249→   249→            if (state[input_id] < escape_state) {
   250→   250→                state[input_id] = escape_state;
   251→   251→            }
   252→   252→        } else if (input->opcode == VTX_OP_Phi) {
   253→   253→            /* Nested Phi — recurse. */
   254→   254→            propagate_escape_through_phi(table, state, state_count,
   255→   255→                                          input_id, escape_state, depth + 1);
   256→   256→        }
   257→   257→    }
   258→   258→}
   259→   259→
   260→   260→/* Helper: propagate escape to a value, handling Phi chains (PEA-1-1).
   261→   261→ * If the value is an allocation, set its escape state directly.
   262→   262→ * If the value is a Phi, walk through to find allocation inputs. */
   263→   263→static void propagate_escape_to_value(vtx_node_table_t *table,
   264→   264→                                       vtx_escape_state_t *state,
   265→   265→                                       uint32_t state_count,
   266→   266→                                       vtx_nodeid_t value_id,
   267→   267→                                       vtx_escape_state_t escape_state)
   268→   268→{
   269→   269→    if (value_id >= state_count) return;
   270→   270→    vtx_node_t *val = vtx_node_get(table, value_id);
   271→   271→    if (!val || val->dead) return;
   272→   272→
   273→   273→    if (is_allocation(val->opcode)) {
   274→   274→        if (state[value_id] < escape_state) {
   275→   275→            state[value_id] = escape_state;
   276→   276→        }
   277→   277→    } else if (val->opcode == VTX_OP_Phi) {
   278→   278→        /* PEA-1-1: walk through the Phi to find allocation inputs. */
   279→   279→        propagate_escape_through_phi(table, state, state_count,
   280→   280→                                      value_id, escape_state, 0);
   281→   281→    }
   282→   282→}

   284→   284→static void transfer_node(vtx_node_t *node, vtx_node_table_t *table,
   285→   285→                           vtx_escape_state_t *state, uint32_t state_count)
   286→   286→{
   287→   287→    if (node->dead) return;
   288→   288→
   289→   289→    switch (node->opcode) {
   290→   290→    /* ---- Return: any returned allocation escapes globally ----
   291→   291→     *
   292→   292→     * PEA-1-1: if the returned value is a Phi, walk through the Phi
   293→   293→     * to find all allocation inputs and mark them GlobalEscape. */
   294→   294→    case VTX_OP_Return:
   295→   295→        for (uint32_t i = 0; i < node->input_count; i++) {
   296→   296→            vtx_nodeid_t input_id = node->inputs[i];
   297→   297→            if (input_id < state_count) {
   298→   298→                propagate_escape_to_value(table, state, state_count,
   299→   299→                                           input_id, VTX_ESCAPE_GLOBAL);
   300→   300→            }
   301→   301→        }
   302→   302→        break;
   303→   303→
   304→   304→    /* ---- StoreField: if the stored value is an allocation, it escapes
   305→   305→     * based on the receiver's escape state ----
   306→   306→     *
   307→   307→     * PEA-1-4: previously the code unconditionally joined with
   308→   308→     * VTX_ESCAPE_ARG, which bumped even NoEscape containers to
   309→   309→     * ArgEscape — disabling scalar replacement for the common pattern
   310→   310→     * of an object holding a private reference to another non-escaping
   311→   311→     * object. Fix: only bump to the container's state (no unconditional
   312→   312→     * ARG join). The container's state already captures whether the
   313→   313→     * value can be observed externally.
   314→   314→     *
   315→   315→     * PEA-1-1: if the stored value is a Phi, walk through to find
   316→   316→     * allocation inputs. */
   317→   317→    case VTX_OP_StoreField:
   318→   318→        if (node->input_count >= 2) {
   319→   319→            vtx_nodeid_t receiver_id = node->inputs[node->input_count - 2];
   320→   320→            vtx_nodeid_t value_id    = node->inputs[node->input_count - 1];
   321→   321→
   322→   322→            /* Determine the container's escape state. */
   323→   323→            vtx_escape_state_t container_state = VTX_ESCAPE_GLOBAL;
   324→   324→            if (receiver_id < state_count) {
   325→   325→                vtx_node_t *recv_node = vtx_node_get(table, receiver_id);
   326→   326→                if (recv_node && !recv_node->dead && is_allocation(recv_node->opcode)) {
   327→   327→                    container_state = state[receiver_id];
   328→   328→                }
   329→   329→            }
   330→   330→
   331→   331→            /* PEA-1-4: propagate container_state to the stored value,
   332→   332→             * WITHOUT the unconditional ARG bump. The value escapes at
   333→   333→             * least as much as the container (if the container escapes,
   334→   334→             * the stored value escapes with it). If the container is
   335→   335→             * NoEscape, the value stays NoEscape too. */
   336→   336→            propagate_escape_to_value(table, state, state_count,
   337→   337→                                      value_id, container_state);
   338→   338→        }
   339→   339→        break;
   340→   340→
   341→   341→    /* ---- StoreIndexed: storing into an array — same as StoreField ----
   342→   342→     * PEA-1-4: same fix — no unconditional ARG bump. */
   343→   343→    case VTX_OP_StoreIndexed:
   344→   344→        if (node->input_count >= 3) {
   345→   345→            vtx_nodeid_t array_id = node->inputs[node->input_count - 3];
   346→   346→            vtx_nodeid_t value_id = node->inputs[node->input_count - 1];
   347→   347→
   348→   348→            vtx_escape_state_t array_state = VTX_ESCAPE_GLOBAL;
   349→   349→            if (array_id < state_count) {
   350→   350→                vtx_node_t *arr_node = vtx_node_get(table, array_id);
   351→   351→                if (arr_node && !arr_node->dead && is_allocation(arr_node->opcode)) {
   352→   352→                    array_state = state[array_id];
   353→   353→                }
   354→   354→            }
   355→   355→            propagate_escape_to_value(table, state, state_count,
   356→   356→                                      value_id, array_state);
   357→   357→        }
   358→   358→        break;
   359→   359→
   360→   360→    /* ---- Calls: any allocation passed as argument escapes ----
   361→   361→     *
   362→   362→     * PEA-1-1: if an argument is a Phi, walk through to find
   363→   363→     * allocation inputs and propagate the escape state to them. */
   364→   364→    case VTX_OP_CallStatic:
   365→   365→    case VTX_OP_CallVirtual:
   366→   366→    case VTX_OP_CallInterface:
   367→   367→    case VTX_OP_CallRuntime:
   368→   368→        for (uint32_t i = 0; i < node->input_count; i++) {
   369→   369→            vtx_nodeid_t input_id = node->inputs[i];
   370→   370→            if (input_id >= state_count) continue;
   371→   371→
   372→   372→            /* PEA-1-1: propagate through Phi chains. CallRuntime
   373→   373→             * is conservative (GlobalEscape); others use ArgEscape. */
   374→   374→            vtx_escape_state_t call_escape =
   375→   375→                (node->opcode == VTX_OP_CallRuntime)
   376→   376→                    ? VTX_ESCAPE_GLOBAL
   377→   377→                    : VTX_ESCAPE_ARG;
   378→   378→            propagate_escape_to_value(table, state, state_count,
   379→   379→                                       input_id, call_escape);
   380→   380→        }
   381→   381→        break;
   382→   382→
   383→   383→    /* ---- Monitor: using an object in a monitor makes it globally escape ---- */
   384→   384→    case VTX_OP_DeoptGuard:
   385→   385→        /* DeoptGuard: the guarded value is consumed but doesn't escape */
   386→   386→        break;
   387→   387→
   388→   388→    /* ---- CheckCast / InstanceOf: the object doesn't escape through these
   389→   389→       (they are pure queries on the object's type) ---- */
   390→   390→    case VTX_OP_CheckCast:
   391→   391→    case VTX_OP_InstanceOf:
   392→   392→        /* No escape — these are read-only type checks */
   393→   393→        break;
   394→   394→
   395→   395→    /* ---- Guard: the checked value doesn't escape ---- */
   396→   396→    case VTX_OP_Guard:
   397→   397→        /* Guards don't cause escape — they just check a condition */
   398→   398→        break;
   399→   399→
   400→   400→    /* ---- Phi: a Phi of allocations — propagate escape states ---- */
   401→   401→    case VTX_OP_Phi:
   402→   402→        /* Phi nodes merge values from multiple control flow paths.
   403→   403→         * If any input allocation has a non-NoEscape state, all allocation
   404→   404→         * inputs of the Phi must have at least that escape state, because
   405→   405→         * the Phi merges them into a single value that may escape.
   406→   406→         *
   407→   407→         * Bug 8 fix: Previously, the Phi transfer was a no-op — it just
   408→   408→         * iterated over inputs without propagating escape states. This meant
   409→   409→         * that if Phi(A, B) where A escapes but B doesn't, B would remain
   410→   410→         * NoEscape even though the merged value escapes. Now we compute
   411→   411→         * the maximum escape state across all Phi inputs and propagate it
   412→   412→         * to every allocation input of the Phi. */
   413→   413→        {
   414→   414→            vtx_escape_state_t max_state = VTX_ESCAPE_NONE;
   415→   415→            /* First pass: compute the maximum escape state across all inputs */
   416→   416→            for (uint32_t i = 0; i < node->input_count; i++) {
   417→   417→                vtx_nodeid_t input_id = node->inputs[i];
   418→   418→                if (input_id < state_count) {
   419→   419→                    vtx_node_t *input = vtx_node_get(table, input_id);
   420→   420→                    if (input && !input->dead && is_allocation(input->opcode)) {
   421→   421→                        max_state = vtx_escape_join(max_state, state[input_id]);
   422→   422→                    }
   423→   423→                }
   424→   424→            }
   425→   425→            /* Second pass: propagate max_state to all allocation inputs.
   426→   426→             * If any input allocation has a lower escape state than the
   427→   427→             * Phi's maximum, it must be elevated because the Phi's result
   428→   428→             * (which merges all inputs) carries the highest escape state. */
   429→   429→            for (uint32_t i = 0; i < node->input_count; i++) {
   430→   430→                vtx_nodeid_t input_id = node->inputs[i];
   431→   431→                if (input_id < state_count) {
   432→   432→                    vtx_node_t *input = vtx_node_get(table, input_id);
   433→   433→                    if (input && !input->dead && is_allocation(input->opcode)) {
   434→   434→                        if (state[input_id] < max_state) {
   435→   435→                            state[input_id] = max_state;
   436→   436→                            /* changed flag is not accessible here, but the
   437→   437→                             * caller (transfer_block -> worklist) will detect
   438→   438→                             * the state change when comparing exit states. */
   439→   439→                        }
   440→   440→                    }
   441→   441→                }
   442→   442→            }
   443→   443→        }
   444→   444→        break;
   445→   445→
   446→   446→    default:
   447→   447→        break;
   448→   448→    }
   449→   449→}
   450→   450→
   451→   451→/* ========================================================================== */
   452→   452→/* Internal: pre-compute block-to-node mapping (§2.1 — fix O(N×B) scan)      */
   453→   453→/* ========================================================================== */
   454→   454→/*
   455→   455→ * PEA audit: transfer_block re-scans ALL N nodes for EVERY block to find
   456→   456→ * which nodes belong to that block. For a graph with B blocks and N nodes,
   457→   457→ * this is O(N×B) per dataflow iteration — for T2 graphs with 5000 nodes
   458→   458→ * and 50 blocks, that's 250K comparisons per iteration × ~10 iterations
   459→   459→ * = 2.5M comparisons.
   460→   460→ *
   461→   461→ * GraalVM's PEA uses a pre-computed BlockMap (BlockBag in
   462→   462→ * EscapeAnalysisPhase). V8's escape analysis uses a schedule
   463→   463→ * (Schedule::rpo_order) to iterate nodes per-block.
   464→   464→ *
   465→   465→ * Fix: pre-compute a block→node-list mapping once before the worklist,
   466→   466→ * then iterate only the relevant nodes in transfer_block. */
   467→   467→typedef struct {
   468→   468→    vtx_nodeid_t *nodes;    /* node IDs belonging to this block */
   469→   469→    uint32_t      count;    /* number of nodes in this block */
   470→   470→} vtx_block_node_list_t;
   471→   471→
   472→   472→static vtx_block_node_list_t *build_block_node_lists(vtx_graph_t *graph,
   473→   473→                                                       vtx_arena_t *arena)
   474→   474→{
   475→   475→    uint32_t block_count = graph->block_count;
   476→   476→    uint32_t node_count = graph->node_table.count;
   477→   477→
   478→   478→    /* Allocate per-block lists */
   479→   479→    vtx_block_node_list_t *lists = vtx_arena_alloc(arena,
   480→   480→        block_count * sizeof(vtx_block_node_list_t));
   481→   481→    if (!lists) return NULL;
   482→   482→    memset(lists, 0, block_count * sizeof(vtx_block_node_list_t));
   483→   483→
   484→   484→    /* First pass: count nodes per block */
   485→   485→    for (uint32_t i = 0; i < node_count; i++) {
   486→   486→        vtx_node_t *node = &graph->node_table.nodes[i];
   487→   487→        if (node->dead) continue;
   488→   488→
   489→   489→        /* Determine which block this node belongs to */
   490→   490→        for (uint32_t b = 0; b < block_count; b++) {
   491→   491→            vtx_block_info_t *block = &graph->blocks[b];
   492→   492→            bool belongs = false;
   493→   493→
   494→   494→            if (node->id == block->region_node ||
   495→   495→                node->id == block->control_node ||
   496→   496→                node->id == block->memory_node) {
   497→   497→                belongs = true;
   498→   498→            } else {
   499→   499→                for (uint32_t j = 0; j < node->input_count; j++) {
   500→   500→                    if (node->inputs[j] == block->region_node ||
   501→   501→                        node->inputs[j] == block->control_node ||
   502→   502→                        node->inputs[j] == block->memory_node) {
   503→   503→                        belongs = true;
   504→   504→                        break;
   505→   505→                    }
   506→   506→                }
   507→   507→                if (!belongs && block->region_node < node_count) {
   508→   508→                    vtx_node_t *region = &graph->node_table.nodes[block->region_node];
   509→   509→                    if (node->bytecode_pc == region->bytecode_pc) {
   510→   510→                        belongs = true;
   511→   511→                    }
   512→   512→                }
   513→   513→            }
   514→   514→
   515→   515→            if (belongs) {
   516→   516→                lists[b].count++;
   517→   517→                break;  /* each node belongs to at most one block */
   518→   518→            }
   519→   519→        }
   520→   520→    }
   521→   521→
   522→   522→    /* Allocate arrays */
   523→   523→    for (uint32_t b = 0; b < block_count; b++) {
   524→   524→        if (lists[b].count > 0) {
   525→   525→            lists[b].nodes = vtx_arena_alloc(arena,
   526→   526→                lists[b].count * sizeof(vtx_nodeid_t));
   527→   527→            if (!lists[b].nodes) return NULL;
   528→   528→            lists[b].count = 0;  /* reset for second pass */
   529→   529→        }
   530→   530→    }
   531→   531→
   532→   532→    /* Second pass: fill arrays */
   533→   533→    for (uint32_t i = 0; i < node_count; i++) {
   534→   534→        vtx_node_t *node = &graph->node_table.nodes[i];
   535→   535→        if (node->dead) continue;
   536→   536→
   537→   537→        for (uint32_t b = 0; b < block_count; b++) {
   538→   538→            vtx_block_info_t *block = &graph->blocks[b];
   539→   539→            bool belongs = false;
   540→   540→
   541→   541→            if (node->id == block->region_node ||
   542→   542→                node->id == block->control_node ||
   543→   543→                node->id == block->memory_node) {
   544→   544→                belongs = true;
   545→   545→            } else {
   546→   546→                for (uint32_t j = 0; j < node->input_count; j++) {
   547→   547→                    if (node->inputs[j] == block->region_node ||
   548→   548→                        node->inputs[j] == block->control_node ||
   549→   549→                        node->inputs[j] == block->memory_node) {
   550→   550→                        belongs = true;
   551→   551→                        break;
   552→   552→                    }
   553→   553→                }
   554→   554→                if (!belongs && block->region_node < node_count) {
   555→   555→                    vtx_node_t *region = &graph->node_table.nodes[block->region_node];
   556→   556→                    if (node->bytecode_pc == region->bytecode_pc) {
   557→   557→                        belongs = true;
   558→   558→                    }
   559→   559→                }
   560→   560→            }
   561→   561→
   562→   562→            if (belongs) {
   563→   563→                lists[b].nodes[lists[b].count++] = node->id;
   564→   564→                break;
   565→   565→            }
   566→   566→        }
   567→   567→    }
   568→   568→
   569→   569→    return lists;
   570→   570→}
   571→   571→
   572→   572→/* ========================================================================== */
   573→   573→/* Internal: fast transfer function using pre-computed block-node lists        */
   574→   574→/* ========================================================================== */
   575→   575→/*
   576→   576→ * §2.1: Uses the pre-computed block_node_lists to iterate only the nodes
   577→   577→ * that belong to this block, instead of scanning all N nodes.
   578→   578→ * GraalVM: EscapeAnalysisPhase uses BlockBag for per-block iteration.
   579→   579→ * V8: EscapeAnalysis uses Schedule::rpo_order for per-block iteration.
   580→   580→ */
   581→   581→static void transfer_block_fast(vtx_graph_t *graph, uint32_t block_idx,
   582→   582→                                 vtx_pea_block_state_t *bs,
   583→   583→                                 vtx_node_table_t *table,
   584→   584→                                 const vtx_block_node_list_t *block_lists)
   585→   585→{
   586→   586→    /* Start from entry state */
   587→   587→    copy_escape_states(bs->exit_state, bs->entry_state, bs->state_count);
   588→   588→
   589→   589→    const vtx_block_node_list_t *list = &block_lists[block_idx];
   590→   590→    for (uint32_t i = 0; i < list->count; i++) {
   591→   591→        vtx_node_t *node = &table->nodes[list->nodes[i]];
   592→   592→        if (node->dead) continue;
   593→   593→        transfer_node(node, table, bs->exit_state, bs->state_count);
   594→   594→    }
   595→   595→}
   596→   596→
   597→   597→/* ========================================================================== */
   598→   598→/* Internal: compute reverse postorder for worklist                            */
   599→   599→/* ========================================================================== */
   600→   600→
   601→   601→static uint32_t *compute_reverse_postorder(vtx_graph_t *graph, vtx_arena_t *arena,
   602→   602→                                             uint32_t *rpo_count)
   603→   603→{
   604→   604→    uint32_t n = graph->block_count;
   605→   605→    if (n == 0) {
   606→   606→        *rpo_count = 0;
   607→   607→        return NULL;
   608→   608→    }
   609→   609→
   610→   610→    uint32_t *rpo = vtx_arena_alloc(arena, n * sizeof(uint32_t));
   611→   611→    if (!rpo) return NULL;
   612→   612→
   613→   613→    /* Simple DFS-based reverse postorder computation.
   614→   614→     * We start from block 0 (entry block) and traverse successors. */
   615→   615→    bool *visited = vtx_arena_alloc(arena, n * sizeof(bool));
   616→   616→    uint32_t *stack = vtx_arena_alloc(arena, n * sizeof(uint32_t));
   617→   617→    if (!visited || !stack) return NULL;
   618→   618→
   619→   619→    memset(visited, 0, n * sizeof(bool));
   620→   620→
   621→   621→    uint32_t rpo_idx = n; /* fill from end */
   622→   622→    int32_t top = 0;
   623→   623→    stack[0] = 0;
   624→   624→    visited[0] = true;
   625→   625→
   626→   626→    while (top >= 0) {
   627→   627→        uint32_t current = stack[top];
   628→   628→        bool has_unvisited_succ = false;
   629→   629→
   630→   630→        vtx_block_info_t *block = &graph->blocks[current];
   631→   631→        for (uint32_t i = 0; i < block->succ_count; i++) {
   632→   632→            uint32_t succ = block->succ_indices[i];
   633→   633→            if (succ < n && !visited[succ]) {
   634→   634→                visited[succ] = true;
   635→   635→                top++;
   636→   636→                stack[top] = succ;
   637→   637→                has_unvisited_succ = true;
   638→   638→                break;
   639→   639→            }
   640→   640→        }
   641→   641→
   642→   642→        if (!has_unvisited_succ) {
   643→   643→            rpo_idx--;
   644→   644→            rpo[rpo_idx] = current;
   645→   645→            top--;
   646→   646→        }
   647→   647→    }
   648→   648→
   649→   649→    *rpo_count = n - rpo_idx;
   650→   650→    if (rpo_idx > 0) {
   651→   651→        /* Some blocks were unreachable; shift the array */
   652→   652→        memmove(rpo, rpo + rpo_idx, *rpo_count * sizeof(uint32_t));
   653→   653→    }
   654→   654→
   655→   655→    return rpo;
   656→   656→}

   658→   658→/* ========================================================================== */
   659→   659→/* Main analysis entry point                                                   */
   660→   660→/* ========================================================================== */
   661→   661→
   662→   662→vtx_pea_analysis_t *vtx_pea_run(vtx_graph_t *graph, vtx_arena_t *arena)
   663→   663→{
   664→   664→    VTX_ASSERT(graph != NULL, "graph must not be NULL");
   665→   665→    VTX_ASSERT(arena != NULL, "arena must not be NULL");
   666→   666→
   667→   667→    if (graph->block_count == 0) {
   668→   668→        /* No blocks — degenerate graph */
   669→   669→        return NULL;
   670→   670→    }
   671→   671→
   672→   672→    /* Allocate the result */
   673→   673→    vtx_pea_analysis_t *result = vtx_arena_alloc(arena, sizeof(vtx_pea_analysis_t));
   674→   674→    if (!result) return NULL;
   675→   675→    memset(result, 0, sizeof(*result));
   676→   676→
   677→   677→    /* Step 1: Identify all allocations */
   678→   678→    if (build_allocation_map(graph, &result->escape_map, arena) != 0) {
   679→   679→        return NULL;
   680→   680→    }
   681→   681→
   682→   682→    /* Step 2: Build per-block state arrays */
   683→   683→    if (build_block_states(graph, result, arena) != 0) {
   684→   684→        return NULL;
   685→   685→    }
   686→   686→
   687→   687→    vtx_node_table_t *table = &graph->node_table;
   688→   688→    uint32_t state_count = result->escape_map.state_count;
   689→   689→    uint32_t block_count = graph->block_count;
   690→   690→
   691→   691→    /* Step 3: Compute reverse postorder for efficient worklist */
   692→   692→    uint32_t rpo_count = 0;
   693→   693→    uint32_t *rpo = compute_reverse_postorder(graph, arena, &rpo_count);
   694→   694→
   695→   695→    /* §2.1: Pre-compute block-to-node lists to avoid O(N×B) scan
   696→   696→     * in transfer_block. This is built once before the worklist loop.
   697→   697→     * GraalVM: BlockBag in EscapeAnalysisPhase.
   698→   698→     * V8: Schedule::rpo_order in EscapeAnalysis. */
   699→   699→    vtx_block_node_list_t *block_lists = build_block_node_lists(graph, arena);
   700→   700→    if (!block_lists) return NULL;
   701→   701→
   702→   702→    /* Step 4: Initialize worklist with all blocks in RPO */
   703→   703→    /* We use a simple flag-based worklist: a block is on the worklist
   704→   704→     * if its `entry_changed` flag is true. */
   705→   705→    bool *on_worklist = vtx_arena_alloc(arena, block_count * sizeof(bool));
   706→   706→    if (!on_worklist) return NULL;
   707→   707→    memset(on_worklist, 0, block_count * sizeof(bool));
   708→   708→
   709→   709→    /* Circular buffer worklist */
   710→   710→    uint32_t wl_capacity = block_count + 1;
   711→   711→    uint32_t *worklist = vtx_arena_alloc(arena, wl_capacity * sizeof(uint32_t));
   712→   712→    if (!worklist) return NULL;
   713→   713→
   714→   714→    uint32_t wl_head = 0, wl_tail = 0;
   715→   715→
   716→   716→    /* Enqueue all blocks in RPO */
   717→   717→    for (uint32_t i = 0; i < rpo_count; i++) {
   718→   718→        uint32_t block_idx = rpo[i];
   719→   719→        worklist[wl_tail] = block_idx;
   720→   720→        wl_tail = (wl_tail + 1) % wl_capacity;
   721→   721→        on_worklist[block_idx] = true;
   722→   722→    }
   723→   723→
   724→   724→    /* Step 5: Iterate to fixed point.
   725→   725→     *
   726→   726→     * BUGFIX (audit #4): The old hard cap of 100 iterations was too low
   727→   727→     * for large CFGs. If the worklist hadn't converged by iteration 100,
   728→   728→     * the escape states were left in an intermediate (wrong) state.
   729→   729→     * Allocations that should be GlobalEscape might be marked NoEscape
   730→   730→     * → heap corruption from scalar replacement.
   731→   731→     *
   732→   732→     * Fix: Increase to 10000 (sufficient for any real-world CFG), and
   733→   733→     * if we hit the cap, conservatively mark ALL allocations as
   734→   734→     * GlobalEscape (safe fallback — no scalar replacement). */
   735→   735→    uint32_t iterations = 0;
   736→   736→    const uint32_t MAX_ITERATIONS = 10000; /* safety bound — was 100 */
   737→   737→    bool hit_cap = false;
   738→   738→
   739→   739→    while (wl_head != wl_tail && iterations < MAX_ITERATIONS) {
   740→   740→        iterations++;
   741→   741→
   742→   742→        uint32_t block_idx = worklist[wl_head];
   743→   743→        wl_head = (wl_head + 1) % wl_capacity;
   744→   744→        on_worklist[block_idx] = false;
   745→   745→
   746→   746→        vtx_pea_block_state_t *bs = &result->block_states[block_idx];
   747→   747→        vtx_block_info_t *block = &graph->blocks[block_idx];
   748→   748→
   749→   749→        /* Compute entry state as join of predecessor exit states */
   750→   750→        vtx_escape_state_t *new_entry = vtx_arena_alloc(arena,
   751→   751→            state_count * sizeof(vtx_escape_state_t));
   752→   752→        if (!new_entry) return NULL;
   753→   753→        memset(new_entry, 0, state_count * sizeof(vtx_escape_state_t));
   754→   754→
   755→   755→        if (block->pred_count == 0) {
   756→   756→            /* Entry block: entry state is all NoEscape (already zeroed) */
   757→   757→        } else {
   758→   758→            /* Join all predecessor exit states */
   759→   759→            for (uint32_t p = 0; p < block->pred_count; p++) {
   760→   760→                uint32_t pred_idx = block->pred_indices[p];
   761→   761→                if (pred_idx < block_count) {
   762→   762→                    join_escape_states(new_entry,
   763→   763→                                       result->block_states[pred_idx].exit_state,
   764→   764→                                       state_count);
   765→   765→                }
   766→   766→            }
   767→   767→        }
   768→   768→
   769→   769→        /* Check if entry state changed */
   770→   770→        bool entry_changed = escape_states_differ(new_entry, bs->entry_state,
   771→   771→                                                    state_count);
   772→   772→        if (entry_changed) {
   773→   773→            copy_escape_states(bs->entry_state, new_entry, state_count);
   774→   774→        }
   775→   775→
   776→   776→        /* Apply transfer function */
   777→   777→        vtx_escape_state_t *old_exit = vtx_arena_alloc(arena,
   778→   778→            state_count * sizeof(vtx_escape_state_t));
   779→   779→        if (!old_exit) return NULL;
   780→   780→        copy_escape_states(old_exit, bs->exit_state, state_count);
   781→   781→
   782→   782→        /* §2.1: Use fast transfer with pre-computed block-node lists */
   783→   783→        transfer_block_fast(graph, block_idx, bs, table, block_lists);
   784→   784→
   785→   785→        bool exit_changed = escape_states_differ(bs->exit_state, old_exit,
   786→   786→                                                   state_count);
   787→   787→
   788→   788→        /* If exit state changed, add successors to worklist */
   789→   789→        if (entry_changed || exit_changed) {
   790→   790→            for (uint32_t s = 0; s < block->succ_count; s++) {
   791→   791→                uint32_t succ_idx = block->succ_indices[s];
   792→   792→                if (succ_idx < block_count && !on_worklist[succ_idx]) {
   793→   793→                    worklist[wl_tail] = succ_idx;
   794→   794→                    wl_tail = (wl_tail + 1) % wl_capacity;
   795→   795→                    on_worklist[succ_idx] = true;
   796→   796→                }
   797→   797→            }
   798→   798→        }
   799→   799→    }
   800→   800→
   801→   801→    result->iterations = iterations;
   802→   802→
   803→   803→    /* If we hit the iteration cap, the escape states may not have
   804→   804→     * converged. Conservatively mark ALL allocations as GlobalEscape
   805→   805→     * to prevent incorrect scalar replacement. */
   806→   806→    if (iterations >= MAX_ITERATIONS) {
   807→   807→        hit_cap = true;
   808→   808→        for (uint32_t a = 0; a < result->escape_map.alloc_count; a++) {
   809→   809→            vtx_nodeid_t alloc_id = result->escape_map.alloc_ids[a];
   810→   810→            for (uint32_t b = 0; b < block_count; b++) {
   811→   811→                if (alloc_id < result->block_states[b].state_count) {
   812→   812→                    result->block_states[b].exit_state[alloc_id] = VTX_ESCAPE_GLOBAL;
   813→   813→                    result->block_states[b].entry_state[alloc_id] = VTX_ESCAPE_GLOBAL;
   814→   814→                }
   815→   815→            }
   816→   816→        }
   817→   817→    }
   818→   818→
   819→   819→    /* Step 6: Finalize — compute the global escape state for each allocation
   820→   820→     * as the join of all block exit states */
   821→   821→    for (uint32_t a = 0; a < result->escape_map.alloc_count; a++) {
   822→   822→        vtx_nodeid_t alloc_id = result->escape_map.alloc_ids[a];
   823→   823→        vtx_escape_state_t final_state = VTX_ESCAPE_NONE;
   824→   824→        for (uint32_t b = 0; b < block_count; b++) {
   825→   825→            if (alloc_id < result->block_states[b].state_count) {
   826→   826→                final_state = vtx_escape_join(final_state,
   827→   827→                    result->block_states[b].exit_state[alloc_id]);
   828→   828→            }
   829→   829→        }
   830→   830→
   831→   831→        result->escape_map.states[alloc_id] = final_state;
   832→   832→
   833→   833→        /* Update statistics */
   834→   834→        result->total_allocs++;
   835→   835→        switch (final_state) {
   836→   836→        case VTX_ESCAPE_NONE:   result->no_escape_count++; break;
   837→   837→        case VTX_ESCAPE_ARG:    result->arg_escape_count++; break;
   838→   838→        case VTX_ESCAPE_GLOBAL: result->global_escape_count++; break;
   839→   839→        }
   840→   840→    }
   841→   841→
   842→   842→    return result;
   843→   843→}

   850→   850→                                       vtx_nodeid_t node_id)

   864→   864→
   865→   865→vtx_escape_state_t vtx_pea_block_entry_state(const vtx_pea_analysis_t *analysis,
   866→   866→                                              uint32_t block_idx,
   867→   867→                                              vtx_nodeid_t alloc_id)
   868→   868→{
   869→   869→    VTX_ASSERT(analysis != NULL, "analysis must not be NULL");
