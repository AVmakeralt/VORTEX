    15→    15→ * VORTEX Speculative PEA + Lock Elision — Implementation
    16→    16→ *
    17→    17→ * Implements two profile-guided speculative optimizations:
    18→    18→ *
    19→    19→ *   1. Speculative PEA: Speculate NoEscape for allocations where PEA
    20→    20→ *      proved ArgEscape/GlobalEscape but profiling suggests they usually
    21→    21→ *      don't escape. Add guards; deopt on failure.
    22→    22→ *
    23→    23→ *   2. Speculative Lock Elision: Elide monitor enter/exit for
    24→    24→ *      synchronized blocks with zero/near-zero contention. Add guards;
    25→    25→ *      deopt on contention.
    26→    26→ *
    27→    27→ * Both follow the same pattern:
    28→    28→ *   - Profile → decide → speculate → guard → adapt
    29→    29→ *   - Failed speculations weaken the guard (FastCheck → FullCheck → DeoptAlways)
    30→    30→ *   - Successful speculations remain at FastCheck or strengthen to PredicatedCheck
    31→    31→ */

    67→    67→/* Internal: get the no-escape rate threshold for a given proven state          */
    68→    68→/* and speculation level                                                        */
    69→    69→/* ========================================================================== */
    70→    70→
    71→    71→/**
    72→    72→ * Returns the minimum no_escape_rate required to speculate that an
    73→    73→ * allocation with the given proven escape state doesn't escape, at
    74→    74→ * the given speculation level.
    75→    75→ *
    76→    76→ * Returns INFINITY if the proven state is not eligible for speculation
    77→    77→ * at this level (e.g., GlobalEscape at CONSERVATIVE level).
    78→    78→ */
    79→    79→static double get_no_escape_threshold(vtx_spec_pea_level_t level,
    80→    80→                                       vtx_escape_state_t proven_state)
    81→    81→{
    82→    82→    switch (level) {
    83→    83→    case VTX_SPEC_PEA_NONE:
    84→    84→        /* No speculation at all */
    85→    85→        return INFINITY;
    86→    86→
    87→    87→    case VTX_SPEC_PEA_CONSERVATIVE:
    88→    88→        /* Only speculate on ArgEscape */
    89→    89→        if (proven_state == VTX_ESCAPE_ARG) {
    90→    90→            return VTX_SPEC_RATE_CONSERVATIVE_ARG;  /* 0.99 */
    91→    91→        }
    92→    92→        return INFINITY;  /* GlobalEscape not eligible */
    93→    93→
    94→    94→    case VTX_SPEC_PEA_MODERATE:
    95→    95→        /* Speculate on ArgEscape + some GlobalEscape */
    96→    96→        if (proven_state == VTX_ESCAPE_ARG) {
    97→    97→            return VTX_SPEC_RATE_MODERATE_ARG;       /* 0.95 */
    98→    98→        }
    99→    99→        if (proven_state == VTX_ESCAPE_GLOBAL) {
   100→   100→            return VTX_SPEC_RATE_MODERATE_GLOBAL;    /* 0.98 */
   101→   101→        }
   102→   102→        return INFINITY;
   103→   103→
   104→   104→    case VTX_SPEC_PEA_AGGRESSIVE:
   105→   105→        /* Speculate on any escaping object */
   106→   106→        if (proven_state == VTX_ESCAPE_ARG) {
   107→   107→            return VTX_SPEC_RATE_AGGRESSIVE_ARG;     /* 0.80 */
   108→   108→        }
   109→   109→        if (proven_state == VTX_ESCAPE_GLOBAL) {
   110→   110→            return VTX_SPEC_RATE_AGGRESSIVE_GLOBAL;  /* 0.90 */
   111→   111→        }
   112→   112→        return INFINITY;
   113→   113→    }
   114→   114→
   115→   115→    return INFINITY;
   116→   116→}

   118→   118→/* ========================================================================== */
   119→   119→/* Internal: look up profile data for an allocation site                        */
   120→   120→/* ========================================================================== */
   121→   121→
   122→   122→/**
   123→   123→ * Look up value profile data for an allocation site identified by
   124→   124→ * its bytecode_pc. The allocation node's bytecode_pc field is used
   125→   125→ * to index into the value profile table.
   126→   126→ *
   127→   127→ * Returns the no_escape_rate and total_observations from the profile.
   128→   128→ * If no profile data is available, returns false.
   129→   129→ */
   130→   130→static bool lookup_alloc_profile(const vtx_value_profile_table_t *value_profiles,
   131→   131→                                  uint32_t bytecode_pc,
   132→   132→                                  uint64_t *out_total_obs,
   133→   133→                                  uint64_t *out_no_escape_obs,
   134→   134→                                  double *out_no_escape_rate)
   135→   135→{
   136→   136→    if (value_profiles == NULL) return false;
   137→   137→
   138→   138→    const vtx_value_profile_t *profile =
   139→   139→        vtx_value_profile_lookup(value_profiles, bytecode_pc);
   140→   140→    if (profile == NULL) return false;
   141→   141→
   142→   142→    if (profile->total_observations < VTX_SPEC_PEA_MIN_OBSERVATIONS) {
   143→   143→        return false;  /* not enough data */
   144→   144→    }
   145→   145→
   146→   146→    *out_total_obs = profile->total_observations;
   147→   147→
   148→   148→    /* The value profile tracks the most frequent value. For escape
   149→   149→     * analysis, the "no escape" observations are stored as the top
   150→   150→     * value count. The value itself encodes whether the object escaped:
   151→   151→     *   0 = didn't escape, 1 = escaped.
   152→   152→     * We use the top value's count as no_escape_obs if the top value
   153→   153→     * is 0 (no escape), otherwise use other_count. */
   154→   154→    uint64_t top_value;
   155→   155→    double top_freq;
   156→   156→    if (vtx_value_profile_top(profile, &top_value, &top_freq)) {
   157→   157→        if (top_value == 0) {
   158→   158→            /* Top value is "didn't escape" */
   159→   159→            *out_no_escape_obs = profile->entries[0].count;
   160→   160→        } else {
   161→   161→            /* Top value is "escaped" — the no-escape count is in the
   162→   162→             * other bucket or second entry */
   163→   163→            if (profile->entries[1].count > 0 && profile->entries[1].value == 0) {
   164→   164→                *out_no_escape_obs = profile->entries[1].count;
   165→   165→            } else {
   166→   166→                *out_no_escape_obs = profile->other_count;
   167→   167→            }
   168→   168→        }
   169→   169→    } else {
   170→   170→        return false;
   171→   171→    }
   172→   172→
   173→   173→    *out_no_escape_rate = (double)*out_no_escape_obs / (double)*out_total_obs;
   174→   174→    return true;
   175→   175→}

   226→   226→
   227→   227→vtx_spec_pea_analysis_t *vtx_spec_pea_run(vtx_graph_t *graph,
   228→   228→                                            vtx_arena_t *arena,
   229→   229→                                            vtx_spec_pea_level_t level,
   230→   230→                                            const vtx_value_profile_table_t *value_profiles)
   231→   231→{
   232→   232→    VTX_ASSERT(graph != NULL, "graph must not be NULL");
   233→   233→    VTX_ASSERT(arena != NULL, "arena must not be NULL");
   234→   234→
   235→   235→    /* Step 1: Run standard PEA to get proven escape states */
   236→   236→    vtx_pea_analysis_t *base = vtx_pea_run(graph, arena);
   237→   237→    if (base == NULL) return NULL;
   238→   238→
   239→   239→    /* Allocate the speculative analysis result */
   240→   240→    vtx_spec_pea_analysis_t *result =
   241→   241→        (vtx_spec_pea_analysis_t *)vtx_arena_alloc(arena, sizeof(vtx_spec_pea_analysis_t));
   242→   242→    if (!result) return NULL;
   243→   243→    memset(result, 0, sizeof(*result));
   244→   244→
   245→   245→    result->base_analysis = *base;  /* copy the base result */
   246→   246→    result->level = level;
   247→   247→
   248→   248→    /* Allocate the speculative decisions array */
   249→   249→    result->spec_capacity = VTX_SPEC_PEA_INITIAL_CAPACITY;
   250→   250→    result->spec_decisions = (vtx_spec_pea_decision_t *)malloc(
   251→   251→        result->spec_capacity * sizeof(vtx_spec_pea_decision_t));
   252→   252→    if (!result->spec_decisions) return NULL;
   253→   253→    result->spec_count = 0;
   254→   254→
   255→   255→    /* Step 2: Build the speculative escape map.
   256→   256→     * Start with a copy of the base escape map; we'll override entries
   257→   257→     * for speculative allocations. */
   258→   258→    uint32_t state_count = base->escape_map.state_count;
   259→   259→    result->speculative_escape_map.state_count = state_count;
   260→   260→
   261→   261→    result->speculative_escape_map.states =
   262→   262→        (vtx_escape_state_t *)malloc(state_count * sizeof(vtx_escape_state_t));
   263→   263→    if (!result->speculative_escape_map.states) {
   264→   264→        free(result->spec_decisions);
   265→   265→        return NULL;
   266→   266→    }
   267→   267→
   268→   268→    /* Copy the base states as the starting point */
   269→   269→    memcpy(result->speculative_escape_map.states,
   270→   270→           base->escape_map.states,
   271→   271→           state_count * sizeof(vtx_escape_state_t));
   272→   272→
   273→   273→    /* Copy the allocation IDs array */
   274→   274→    result->speculative_escape_map.alloc_count = base->escape_map.alloc_count;
   275→   275→    result->speculative_escape_map.alloc_capacity = base->escape_map.alloc_capacity;
   276→   276→
   277→   277→    result->speculative_escape_map.alloc_ids =
   278→   278→        (vtx_nodeid_t *)malloc(base->escape_map.alloc_capacity * sizeof(vtx_nodeid_t));
   279→   279→    if (!result->speculative_escape_map.alloc_ids) {
   280→   280→        free(result->speculative_escape_map.states);
   281→   281→        free(result->spec_decisions);
   282→   282→        return NULL;
   283→   283→    }
   284→   284→
   285→   285→    memcpy(result->speculative_escape_map.alloc_ids,
   286→   286→           base->escape_map.alloc_ids,
   287→   287→           base->escape_map.alloc_count * sizeof(vtx_nodeid_t));
   288→   288→
   289→   289→    /* Step 3: For each allocation with proven_state > NoEscape,
   290→   290→     * check if profiling supports speculation. */
   291→   291→    vtx_node_table_t *table = &graph->node_table;
   292→   292→
   293→   293→    for (uint32_t a = 0; a < base->escape_map.alloc_count; a++) {
   294→   294→        vtx_nodeid_t alloc_id = base->escape_map.alloc_ids[a];
   295→   295→        vtx_escape_state_t proven = base->escape_map.states[alloc_id];
   296→   296→
   297→   297→        /* Count proven NoEscape allocations */
   298→   298→        if (proven == VTX_ESCAPE_NONE) {
   299→   299→            result->total_proven++;
   300→   300→            continue;  /* already NoEscape, no speculation needed */
   301→   301→        }
   302→   302→
   303→   303→        /* Check if this proven state is eligible for speculation */
   304→   304→        double threshold = get_no_escape_threshold(level, proven);
   305→   305→        if (isinf(threshold)) continue;  /* not eligible at this level */
   306→   306→
   307→   307→        /* Look up the allocation node to get its bytecode_pc */
   308→   308→        if (alloc_id >= table->count) continue;
   309→   309→        vtx_node_t *alloc_node = &table->nodes[alloc_id];
   310→   310→        if (alloc_node->dead) continue;
   311→   311→
   312→   312→        /* Look up profile data for this allocation site */
   313→   313→        uint64_t total_obs = 0;
   314→   314→        uint64_t no_escape_obs = 0;
   315→   315→        double no_escape_rate = 0.0;
   316→   316→
   317→   317→        if (!lookup_alloc_profile(value_profiles, alloc_node->bytecode_pc,
   318→   318→                                   &total_obs, &no_escape_obs, &no_escape_rate)) {
   319→   319→            continue;  /* no profile data available */
   320→   320→        }
   321→   321→
   322→   322→        /* Check if the no-escape rate meets the threshold */
   323→   323→        if (no_escape_rate >= threshold) {
   324→   324→            /* Speculation is justified! Create a decision. */
   325→   325→            if (grow_spec_decisions(result) != 0) {
   326→   326→                /* Allocation failure — continue without this speculation */
   327→   327→                continue;
   328→   328→            }
   329→   329→
   330→   330→            vtx_spec_pea_decision_t *decision =
   331→   331→                &result->spec_decisions[result->spec_count];
   332→   332→            memset(decision, 0, sizeof(*decision));
   333→   333→
   334→   334→            decision->alloc_node_id    = alloc_id;
   335→   335→            decision->proven_state     = proven;
   336→   336→            decision->speculated_state = VTX_ESCAPE_NONE;
   337→   337→            decision->level            = level;
   338→   338→            decision->guard_id         = 0;
   339→   339→            decision->total_observations = total_obs;
   340→   340→            decision->no_escape_obs    = no_escape_obs;
   341→   341→            decision->no_escape_rate   = no_escape_rate;
   342→   342→            decision->guard_installed  = false;
   343→   343→
   344→   344→            result->spec_count++;
   345→   345→            result->total_speculated++;
   346→   346→
   347→   347→            /* Update the speculative escape map */
   348→   348→            result->speculative_escape_map.states[alloc_id] = VTX_ESCAPE_NONE;
   349→   349→        }
   350→   350→    }
   351→   351→
   352→   352→    /* Step 4: Compute statistics */
   353→   353→    result->potential_sr_count = result->total_proven + result->total_speculated;
   354→   354→
   355→   355→    return result;
   356→   356→}

   459→   459→        analysis->speculative_escape_map.alloc_ids = NULL;
   460→   460→    }
   461→   461→
   462→   462→    analysis->spec_count = 0;
   463→   463→    analysis->spec_capacity = 0;
   464→   464→
   465→   465→    /* Note: base_analysis was arena-allocated and will be freed
   466→   466→     * when the arena is destroyed. We don't free it here. */
   467→   467→}

   494→   494→void vtx_lock_elision_result_destroy(vtx_lock_elision_result_t *result)

   499→   499→        free(result->profiles);
   500→   500→        result->profiles = NULL;

   502→   502→
   503→   503→    result->profile_count = 0;
   504→   504→    result->profile_capacity = 0;
   505→   505→    result->total_sites = 0;
   506→   506→    result->eligible_sites = 0;
   507→   507→    result->elided_sites = 0;
   508→   508→}

   510→   510→/* ========================================================================== */
   511→   511→/* Lock Elision: internal: find or create a profile for a monitor site          */
   512→   512→/* ========================================================================== */
   513→   513→
   514→   514→/**
   515→   515→ * Find the profile for a monitor site, or create a new one if it
   516→   516→ * doesn't exist yet.
   517→   517→ *
   518→   518→ * Returns the profile index in the profiles array, or -1 on failure.
   519→   519→ */
   520→   520→static int32_t find_or_create_profile(vtx_lock_elision_result_t *result,
   521→   521→                                        uint32_t monitor_site_id)
   522→   522→{
   523→   523→    /* Search for existing profile */
   524→   524→    for (uint32_t i = 0; i < result->profile_count; i++) {
   525→   525→        if (result->profiles[i].monitor_site_id == monitor_site_id) {
   526→   526→            return (int32_t)i;
   527→   527→        }
   528→   528→    }
   529→   529→
   530→   530→    /* Create new profile */
   531→   531→    if (grow_lock_profiles(result) != 0) return -1;
   532→   532→
   533→   533→    vtx_lock_contention_profile_t *profile =
   534→   534→        &result->profiles[result->profile_count];
   535→   535→    memset(profile, 0, sizeof(*profile));
   536→   536→
   537→   537→    profile->monitor_site_id    = monitor_site_id;
   538→   538→    profile->total_acquisitions = 0;
   539→   539→    profile->contention_count   = 0;
   540→   540→    profile->contention_rate    = 0.0;
   541→   541→    profile->sample_counter     = VTX_LOCK_SAMPLE_INTERVAL_DEFAULT;
   542→   542→    profile->sample_interval    = VTX_LOCK_SAMPLE_INTERVAL_DEFAULT;
   543→   543→    profile->is_eligible        = false;
   544→   544→    profile->guard_id           = 0;
   545→   545→    profile->guard_installed    = false;
   546→   546→
   547→   547→    result->profile_count++;
   548→   548→    result->total_sites++;
   549→   549→
   550→   550→    return (int32_t)(result->profile_count - 1);
   551→   551→}
   552→   552→
   553→   553→/* ========================================================================== */
   554→   554→/* Lock Elision: internal: update eligibility after a sample                    */
   555→   555→/* ========================================================================== */
   556→   556→
   557→   557→/**
   558→   558→ * Recompute the contention rate and eligibility for a profile.
   559→   559→ * Called at sample boundaries.
   560→   560→ */
   561→   561→static void update_eligibility(vtx_lock_contention_profile_t *profile)
   562→   562→{
   563→   563→    /* Recompute contention rate */
   564→   564→    if (profile->total_acquisitions > 0) {
   565→   565→        profile->contention_rate =
   566→   566→            (double)profile->contention_count / (double)profile->total_acquisitions;
   567→   567→    } else {
   568→   568→        profile->contention_rate = 0.0;
   569→   569→    }
   570→   570→
   571→   571→    /* Check eligibility:
   572→   572→     *   - contention_rate < VTX_LOCK_CONTENTION_THRESHOLD (0.1%)
   573→   573→     *   - total_acquisitions >= VTX_LOCK_MIN_OBSERVATIONS (1000) */
   574→   574→    bool was_eligible = profile->is_eligible;
   575→   575→    profile->is_eligible =
   576→   576→        (profile->total_acquisitions >= VTX_LOCK_MIN_OBSERVATIONS) &&
   577→   577→        (profile->contention_rate < VTX_LOCK_CONTENTION_THRESHOLD);
   578→   578→
   579→   579→    /* Update statistics: if eligibility changed, update the result's
   580→   580→     * eligible_sites count. (This is approximate — the caller should
   581→   581→     * recompute stats periodically.) */
   582→   582→    (void)was_eligible;  /* used for incremental update in a full integration */
   583→   583→}
   584→   584→
   585→   585→/* ========================================================================== */
   586→   586→/* Lock Elision: internal: adapt sampling interval                              */
   587→   587→/* ========================================================================== */
   588→   588→
   589→   589→/**
   590→   590→ * Adapt the sampling interval based on the contention stability.
   591→   591→ *
   592→   592→ * Stable sites (contention_rate near 0) get longer intervals
   593→   593→ * (less frequent checking → less overhead).
   594→   594→ * Unstable sites (contention_rate near threshold) get shorter intervals
   595→   595→ * (more responsive tracking).
   596→   596→ *
   597→   597→ * This mirrors the adaptive sampling in guard metadata and value profiles.
   598→   598→ */
   599→   599→static void adapt_sample_interval(vtx_lock_contention_profile_t *profile,
   600→   600→                                    bool had_contention)
   601→   601→{
   602→   602→    if (had_contention) {
   603→   603→        /* Contention observed → shorten interval for rapid detection */
   604→   604→        if (profile->sample_interval > VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE) {
   605→   605→            profile->sample_interval = VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE;
   606→   606→        }
   607→   607→    } else if (profile->contention_rate < VTX_LOCK_CONTENTION_THRESHOLD * 0.1) {
   608→   608→        /* Very stable (contention rate < 0.01%) → lengthen interval */
   609→   609→        profile->sample_interval = VTX_LOCK_SAMPLE_INTERVAL_STABLE;
   610→   610→    } else if (profile->contention_rate < VTX_LOCK_CONTENTION_THRESHOLD * 0.5) {
   611→   611→        /* Moderately stable → default interval */
   612→   612→        profile->sample_interval = VTX_LOCK_SAMPLE_INTERVAL_DEFAULT;
   613→   613→    } else {
   614→   614→        /* Near threshold → keep short for responsiveness */
   615→   615→        profile->sample_interval = VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE;
   616→   616→    }
   617→   617→}
   618→   618→
   619→   619→/* ========================================================================== */
   620→   620→/* Lock Elision: record acquisition (hot path)                                  */
   621→   621→/* ========================================================================== */
   622→   622→
   623→   623→void vtx_lock_elision_record_acquisition(vtx_lock_elision_result_t *result,
   624→   624→                                           uint32_t monitor_site_id,
   625→   625→                                           bool is_contended)
   626→   626→{
   627→   627→    if (result == NULL) return;
   628→   628→
   629→   629→    int32_t idx = find_or_create_profile(result, monitor_site_id);
   630→   630→    if (idx < 0) return;  /* allocation failure */
   631→   631→
   632→   632→    vtx_lock_contention_profile_t *profile = &result->profiles[idx];
   633→   633→
   634→   634→    /* Sampling: decrement counter, only record at sample boundary.
   635→   635→     * This is the zero-cost deopt hot path pattern:
   636→   636→     *   1. Decrement counter (1 cycle)
   637→   637→     *   2. If counter > 1, return (predicted branch)
   638→   638→     *   3. At boundary, do full update and reset counter */
   639→   639→    if (__builtin_expect(profile->sample_counter > 1, 1)) {
   640→   640→        profile->sample_counter--;
   641→   641→
   642→   642→        /* Even between samples, we accumulate contention info.
   643→   643→         * This ensures we don't miss rare contention events. */
   644→   644→        if (is_contended) {
   645→   645→            profile->contention_count++;
   646→   646→            /* Shorten interval on contention for rapid detection */
   647→   647→            if (profile->sample_interval > VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE) {
   648→   648→                profile->sample_interval = VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE;
   649→   649→                if (profile->sample_counter > VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE) {
   650→   650→                    profile->sample_counter = VTX_LOCK_SAMPLE_INTERVAL_UNSTABLE;
   651→   651→                }
   652→   652→            }
   653→   653→        }
   654→   654→        return;
   655→   655→    }
   656→   656→
   657→   657→    /* Sample boundary — do the full update */
   658→   658→    profile->total_acquisitions += profile->sample_interval;
   659→   659→    if (is_contended) {
   660→   660→        profile->contention_count++;
   661→   661→    }
   662→   662→
   663→   663→    /* Recompute eligibility */
   664→   664→    update_eligibility(profile);
   665→   665→
   666→   666→    /* Adapt sampling interval based on stability */
   667→   667→    adapt_sample_interval(profile, is_contended);
   668→   668→
   669→   669→    /* Reset sample counter */
   670→   670→    profile->sample_counter = profile->sample_interval;
   671→   671→}
   672→   672→
   673→   673→/* ========================================================================== */
   674→   674→/* Lock Elision: eligibility check                                              */
   675→   675→/* ========================================================================== */
   676→   676→
   677→   677→bool vtx_lock_elision_is_eligible(const vtx_lock_elision_result_t *result,
   678→   678→                                    uint32_t monitor_site_id)
   679→   679→{
   680→   680→    if (result == NULL || result->profiles == NULL) return false;
   681→   681→
   682→   682→    for (uint32_t i = 0; i < result->profile_count; i++) {
   683→   683→        if (result->profiles[i].monitor_site_id == monitor_site_id) {
   684→   684→            return result->profiles[i].is_eligible;
   685→   685→        }
   686→   686→    }
   687→   687→
   688→   688→    return false;  /* no profile for this site */
   689→   689→}

   693→   693→/* ========================================================================== */
   694→   694→
   695→   695→int vtx_lock_elision_install_guard(vtx_lock_elision_result_t *result,
   696→   696→                                     uint32_t monitor_site_id,
   697→   697→                                     uint32_t guard_id)
   698→   698→{
   699→   699→    VTX_ASSERT(result != NULL, "result must not be NULL");
   700→   700→    VTX_ASSERT(guard_id != 0, "guard_id must not be 0");
   701→   701→
   702→   702→    if (result == NULL || result->profiles == NULL) return -1;
   703→   703→
   704→   704→    for (uint32_t i = 0; i < result->profile_count; i++) {
   705→   705→        if (result->profiles[i].monitor_site_id == monitor_site_id) {
   706→   706→            result->profiles[i].guard_id = guard_id;
   707→   707→            result->profiles[i].guard_installed = true;
   708→   708→            result->elided_sites++;
   709→   709→            return 0;
   710→   710→        }
   711→   711→    }
   712→   712→
   713→   713→    return -1;  /* no profile for this site */
   714→   714→}

   718→   718→/* ========================================================================== */
   719→   719→
   720→   720→void vtx_lock_elision_get_stats(const vtx_lock_elision_result_t *result,
   721→   721→                                  uint32_t *total_sites,
   722→   722→                                  uint32_t *eligible_sites,
   723→   723→                                  uint32_t *elided_sites)
   724→   724→{
   725→   725→    if (result == NULL) {
   726→   726→        if (total_sites)    *total_sites = 0;
   727→   727→        if (eligible_sites) *eligible_sites = 0;
   728→   728→        if (elided_sites)   *elided_sites = 0;
   729→   729→        return;
   730→   730→    }
   731→   731→
   732→   732→    /* Recompute eligible_sites from profiles (may have changed since
   733→   733→     * last check due to profile updates) */
   734→   734→    uint32_t eligible = 0;
   735→   735→    for (uint32_t i = 0; i < result->profile_count; i++) {
   736→   736→        if (result->profiles[i].is_eligible) {
   737→   737→            eligible++;
   738→   738→        }
   739→   739→    }
   740→   740→
   741→   741→    /* Update the cached count */
   742→   742→    ((vtx_lock_elision_result_t *)result)->eligible_sites = eligible;
   743→   743→
   744→   744→    if (total_sites)    *total_sites = result->total_sites;
   745→   745→    if (eligible_sites) *eligible_sites = eligible;
   746→   746→    if (elided_sites)   *elided_sites = result->elided_sites;
   747→   747→}
