    34→{

    44→{
    45→    if (arr->stubs) {
    46→        free(arr->stubs);
    47→        arr->stubs = NULL;
    48→    }
    49→    arr->count = 0;
    50→    arr->capacity = 0;

    53→uint32_t vtx_deopt_stub_array_add(vtx_deopt_stub_array_t *arr,
    54→                                   vtx_deopt_stub_t stub)
    55→{
    56→    if (arr->count >= arr->capacity) {
    57→        uint32_t new_cap = arr->capacity * 2;
    58→        vtx_deopt_stub_t *new_stubs = (vtx_deopt_stub_t *)realloc(
    59→            arr->stubs, new_cap * sizeof(vtx_deopt_stub_t));
    60→        if (!new_stubs) return UINT32_MAX;
    61→        arr->stubs = new_stubs;
    62→        arr->capacity = new_cap;
    63→    }
    64→    uint32_t idx = arr->count;
    65→    arr->stubs[idx] = stub;
    66→    arr->count++;
    67→    return idx;
    68→}

    74→#define REX_W      0x48
    75→#define REX_WR     0x4C
    76→#define REX_WB     0x49
    77→#define REX_WRB    0x4D
    78→
    79→static inline uint8_t modrm(uint8_t mod, uint8_t reg, uint8_t rm)
    80→{
    81→    return (uint8_t)((mod << 6) | ((reg & 7) << 3) | (rm & 7));
    82→}

    88→/**
    89→ * Emit: mov [rbp + offset], reg  (64-bit)
    90→ * Store a register value to a frame slot.
    91→ */
    92→static void emit_store_to_frame(vtx_code_buffer_t *buf, int32_t offset, vtx_reg_t reg)
    93→{
    94→    /* REX.W + 89 /r (MOV r/m64, r64) */
    95→    uint8_t rex = REX_W;
    96→    if (reg >= VTX_REG_R8) rex |= 0x04; /* REX.R */
    97→    vtx_code_buffer_emit_byte(buf, rex);
    98→
    99→    vtx_code_buffer_emit_byte(buf, 0x89); /* MOV r/m64, r64 */
   100→
   101→    if (offset >= -128 && offset <= 127) {
   102→        vtx_code_buffer_emit_byte(buf, modrm(1, reg, VTX_REG_RBP));
   103→        vtx_code_buffer_emit_byte(buf, (uint8_t)(offset & 0xFF));
   104→    } else {
   105→        vtx_code_buffer_emit_byte(buf, modrm(2, reg, VTX_REG_RBP));
   106→        vtx_code_buffer_emit_dword(buf, (uint32_t)offset);
   107→    }
   108→}
   109→
   110→/**
   111→ * Emit: mov reg, [rbp + offset]  (64-bit)
   112→ * Load a frame slot into a register.
   113→ */
   114→static void emit_load_from_frame(vtx_code_buffer_t *buf, vtx_reg_t reg, int32_t offset)
   115→{
   116→    uint8_t rex = REX_W;
   117→    if (reg >= VTX_REG_R8) rex |= 0x04; /* REX.R */
   118→    vtx_code_buffer_emit_byte(buf, rex);
   119→
   120→    vtx_code_buffer_emit_byte(buf, 0x8B); /* MOV r64, r/m64 */
   121→
   122→    if (offset >= -128 && offset <= 127) {
   123→        vtx_code_buffer_emit_byte(buf, modrm(1, reg, VTX_REG_RBP));
   124→        vtx_code_buffer_emit_byte(buf, (uint8_t)(offset & 0xFF));
   125→    } else {
   126→        vtx_code_buffer_emit_byte(buf, modrm(2, reg, VTX_REG_RBP));
   127→        vtx_code_buffer_emit_dword(buf, (uint32_t)offset);
   128→    }
   129→}
   130→
   131→/**
   132→ * Emit: mov reg, imm64 (absolute 64-bit immediate)
   133→ * Uses MOV r64, imm64 encoding (REX.W + B8+rd + 8 bytes)
   134→ */
   135→static void emit_mov_reg_imm64(vtx_code_buffer_t *buf, vtx_reg_t reg, uint64_t imm64)
   136→{
   137→    uint8_t rex = REX_W;
   138→    if (reg >= VTX_REG_R8) rex |= 0x01; /* REX.B */
   139→    vtx_code_buffer_emit_byte(buf, rex);
   140→    vtx_code_buffer_emit_byte(buf, (uint8_t)(0xB8 | (reg & 7)));
   141→    vtx_code_buffer_emit_qword(buf, imm64);
   142→}
   143→
   144→/**
   145→ * Emit: jmp rel32 (relative jump to absolute address)
   146→ */
   147→static void emit_jmp_rel32(vtx_code_buffer_t *buf, uint8_t *code_start,
   148→                              uint64_t target_addr)
   149→{
   150→    vtx_code_buffer_emit_byte(buf, 0xE9);
   151→    /* The offset is relative to the instruction after the jmp (5 bytes).
   152→     * We need absolute addresses for both source and target.
   153→     * current_pos is the buffer offset of the displacement field;
   154→     * the absolute address is code_start + current_pos.
   155→     * The instruction after jmp is at current_pos + 4. */
   156→    uint32_t current_pos = vtx_code_buffer_position(buf);
   157→    uint64_t source_addr = (uint64_t)(uintptr_t)(code_start + current_pos + 4);
   158→    int32_t rel32 = (int32_t)(target_addr - source_addr);
   159→    vtx_code_buffer_emit_dword(buf, (uint32_t)rel32);
   160→}
   161→
   162→/**
   163→ * Emit: push reg (64-bit)
   164→ */
   165→static void emit_push_reg(vtx_code_buffer_t *buf, vtx_reg_t reg)
   166→{
   167→    if (reg >= VTX_REG_R8) {
   168→        vtx_code_buffer_emit_byte(buf, 0x41); /* REX.B */
   169→    }
   170→    vtx_code_buffer_emit_byte(buf, (uint8_t)(0x50 | (reg & 7)));
   171→}
   172→
   173→/**
   174→ * Emit: sub rsp, imm8
   175→ */
   176→static void emit_sub_rsp_imm8(vtx_code_buffer_t *buf, uint8_t imm8)
   177→{
   178→    vtx_code_buffer_emit_byte(buf, 0x48); /* REX.W */
   179→    vtx_code_buffer_emit_byte(buf, 0x83); /* SUB r/m64, imm8 */
   180→    vtx_code_buffer_emit_byte(buf, modrm(1, 5, VTX_REG_RSP)); /* /5 = SUB */
   181→    vtx_code_buffer_emit_byte(buf, imm8);
   182→}
   183→
   184→/**
   185→ * Emit: add rsp, imm8
   186→ */
   187→static void emit_add_rsp_imm8(vtx_code_buffer_t *buf, uint8_t imm8)
   188→{
   189→    vtx_code_buffer_emit_byte(buf, 0x48); /* REX.W */
   190→    vtx_code_buffer_emit_byte(buf, 0x83); /* ADD r/m64, imm8 */
   191→    vtx_code_buffer_emit_byte(buf, modrm(1, 0, VTX_REG_RSP)); /* /0 = ADD */
   192→    vtx_code_buffer_emit_byte(buf, imm8);
   193→}

   195→/* ========================================================================== */
   196→/* Deopt stub emission                                                         */
   197→/* ========================================================================== */
   198→
   199→const vtx_deopt_stub_t *vtx_deopt_stub_emit(vtx_deopt_context_t *ctx,
   200→                                              uint32_t guard_index)
   201→{
   202→    VTX_ASSERT(guard_index < ctx->guards->count, "guard index out of bounds");
   203→
   204→    vtx_guard_info_t *guard = &ctx->guards->guards[guard_index];
   205→    vtx_code_buffer_t *buf = ctx->code_buf;
   206→
   207→    /* Record stub start position */
   208→    uint32_t stub_start = vtx_code_buffer_position(buf);
   209→
   210→    vtx_deopt_stub_t stub;
   211→    memset(&stub, 0, sizeof(stub));
   212→    stub.guard_index = guard_index;
   213→    stub.bytecode_pc = guard->deopt_continuation;
   214→    stub.stack_depth = guard->stack_depth;
   215→    stub.native_code_offset = stub_start;
   216→
   217→    /*
   218→     * Deopt stub implementation:
   219→     *
   220→     * The stub must transition from JIT code to the interpreter.
   221→     * We call the runtime deopt function (vtx_deopt_runtime_transition)
   222→     * which handles the frame reconstruction.
   223→     *
   224→     * Calling convention for the runtime function:
   225→     *   RDI = JIT frame RBP
   226→     *   RSI = native PC offset where deopt occurred
   227→     *   RAX = return value (interpreter frame pointer)
   228→     *
   229→     * After the runtime call returns:
   230→     *   - The interpreter frame is fully reconstructed
   231→     *   - RAX holds the interpreter frame pointer
   232→     *   - We jump to the interpreter dispatch loop
   233→     */
   234→
   235→    /* Step 1: Save expression stack registers to frame spill slots.
   236→     * The top VTX_EXPR_REG_COUNT values are in RAX, RCX, RDX, RBX.
   237→     * We need to store them into the spill area so the runtime can
   238→     * find them. The spill area is ordered from bottom (spill[0])
   239→     * to top (spill[max_spills-1]).
   240→     *
   241→     * For a stack depth of N:
   242→     *   Values at positions 0..(N-5) are already in spill slots
   243→     *   Values at positions (N-4)..(N-1) are in registers:
   244→     *     RAX = TOS   = position (N-1)
   245→     *     RCX = TOS-1 = position (N-2)
   246→     *     RDX = TOS-2 = position (N-3)
   247→     *     RBX = TOS-3 = position (N-4)
   248→     *
   249→     * We store register values to the spill area starting from
   250→     * the first register-resident position. */
   251→
   252→    uint32_t depth = guard->stack_depth;
   253→    uint32_t spilled_count = 0;
   254→    if (depth > VTX_EXPR_REG_COUNT) {
   255→        spilled_count = depth - VTX_EXPR_REG_COUNT;
   256→    }
   257→
   258→    /* Save registers that hold expression stack values to the frame.
   259→     *
   260→     * The spill area has max_spills slots, which is enough for the
   261→     * spilled_count values that were already in spill. But the register
   262→     * values (up to 4) also need to be saved somewhere during deopt.
   263→     * We use local variable slots as temporary storage for register values
   264→     * that don't fit in the spill area.
   265→     *
   266→     * Layout of saved values:
   267→     *   spill[0]..spill[spilled_count-1] = already-spilled values (no action needed)
   268→     *   spill[spilled_count]             = RBX value → but only if within max_spills
   269→     *   For register values that exceed the spill area, use local slots.
   270→     *
   271→     * Simpler approach: always use local slots for register saves during deopt.
   272→     * This avoids any OOB access on the spill area and is safe because
   273→     * local slots are not in use during deopt (the expression stack values
   274→     * are being materialized, not the locals).
   275→     */
   276→
   277→    /* Save register values using local variable slots as temp storage.
   278→     * We use local[max_locals-1], local[max_locals-2], etc.
   279→     * If there aren't enough locals, we push the remaining values onto
   280→     * the stack before calling the deopt runtime, then pop them after.
   281→     * Fix C21: Old code silently dropped register values when neither
   282→     * spill slots nor local slots were available. Now we push to stack. */
   283→
   284→    /* Helper macro: save a register to spill area or local slot.
   285→     * C21 fix: The old code had a stack-push fallback when neither spill
   286→     * nor local slots were available, but the runtime transition function
   287→     * couldn't read values from the pushed stack — they were effectively
   288→     * lost. Fix: always use spill or local slots. If neither is available,
   289→     * the frame layout is undersized (a compilation bug). We emit a trap
   290→     * (int3) so the bug is caught immediately rather than silently losing
   291→     * values. In practice, the frame layout always has enough slots for
   292→     * the registers that need saving — the fallback was dead code that
   293→     * gave a false sense of safety. */
   294→    #define SAVE_REG(reg, spill_idx, local_idx) do { \
   295→        if ((spill_idx) < ctx->frame_layout.max_spills) { \
   296→            int32_t off = vtx_frame_layout_spill_offset(&ctx->frame_layout, (spill_idx)); \
   297→            emit_store_to_frame(buf, off, (reg)); \
   298→        } else if (ctx->frame_layout.max_locals > (local_idx)) { \
   299→            int32_t off = vtx_frame_layout_local_offset(&ctx->frame_layout, ctx->frame_layout.max_locals - 1 - (local_idx)); \
   300→            emit_store_to_frame(buf, off, (reg)); \
   301→        } else { \
   302→            /* C21 fix: No slot available — this is a frame layout bug. \
   303→             * Emit int3 (breakpoint) to catch it in debug, and store to \
   304→             * a known slot (spill 0) in release to avoid silent corruption. \
   305→             * The old stack-push was unreadable by the runtime. */ \
   306→            VTX_ASSERT((spill_idx) < ctx->frame_layout.max_spills || \
   307→                       ctx->frame_layout.max_locals > (local_idx), \
   308→                       "no spill/local slot for register save — frame layout bug"); \
   309→            int32_t off = vtx_frame_layout_spill_offset(&ctx->frame_layout, 0); \
   310→            emit_store_to_frame(buf, off, (reg)); \
   311→        } \
   312→    } while(0)
   313→
   314→    uint32_t stack_pushed = 0;
   315→
   316→    if (depth >= 4) {
   317→        SAVE_REG(VTX_REG_RBX, spilled_count, 0);
   318→        SAVE_REG(VTX_REG_RDX, spilled_count + 1, 1);
   319→        SAVE_REG(VTX_REG_RCX, spilled_count + 2, 2);
   320→        SAVE_REG(VTX_REG_RAX, spilled_count + 3, 3);
   321→    } else if (depth == 3) {
   322→        SAVE_REG(VTX_REG_RDX, spilled_count, 0);
   323→        SAVE_REG(VTX_REG_RCX, spilled_count + 1, 1);
   324→        SAVE_REG(VTX_REG_RAX, spilled_count + 2, 2);
   325→    } else if (depth == 2) {
   326→        SAVE_REG(VTX_REG_RCX, spilled_count, 0);
   327→        SAVE_REG(VTX_REG_RAX, spilled_count + 1, 1);
   328→    } else if (depth == 1) {
   329→        SAVE_REG(VTX_REG_RAX, spilled_count, 0);
   330→    }
   331→    #undef SAVE_REG
   332→
   333→    /* Step 2: Set up arguments for the runtime deopt function.
   334→     *   RDI = JIT frame RBP (already in RBP)
   335→     *   RSI = native PC offset of the guard
   336→     */
   337→    /* mov rdi, rbp */
   338→    vtx_code_buffer_emit_byte(buf, REX_W);
   339→    vtx_code_buffer_emit_byte(buf, 0x89); /* MOV r/m64, r64 */
   340→    vtx_code_buffer_emit_byte(buf, modrm(3, VTX_REG_RBP, VTX_REG_RDI));
   341→
   342→    /* mov rsi, guard_native_pc_offset */
   343→    emit_mov_reg_imm64(buf, VTX_REG_RSI, (uint64_t)guard->native_pc_offset);
   344→
   345→    /* Step 3: Align stack for function call (System V ABI requires 16-byte
   346→     * alignment at the call site). We need to figure out if the stack is
   347→     * currently aligned. After the JIT prologue, the stack was aligned.
   348→     * The expression stack doesn't change RSP in our design (it's register-
   349→     * based). So RSP should still be 16-byte aligned.
   350→     * However, to be safe, we push a dummy value to align. */
   351→    /* push rbp (save RBP across the call — it holds our frame pointer) */
   352→    emit_push_reg(buf, VTX_REG_RBP);
   353→    /* Also save RBX (callee-saved, but we use it for expr stack) */
   354→    emit_push_reg(buf, VTX_REG_RBX);
   355→
   356→    /* Step 4: Call the runtime deopt function */
   357→    emit_mov_reg_imm64(buf, VTX_REG_RAX,
   358→        (uint64_t)(uintptr_t)vtx_deopt_runtime_transition);
   359→    /* call rax */
   360→    vtx_code_buffer_emit_byte(buf, 0xFF); /* CALL r/m64 */
   361→    vtx_code_buffer_emit_byte(buf, modrm(3, 2, VTX_REG_RAX)); /* /2 = CALL */
   362→
   363→    /* Step 5: Restore callee-saved registers */
   364→    emit_add_rsp_imm8(buf, 16); /* pop dummy + rbp (we pushed 2 regs) */
   365→
   366→    /* Step 6: After the runtime call, RAX contains the interpreter
   367→     * frame pointer. Jump to the interpreter dispatch loop.
   368→     *
   369→     * The interpreter entry point expects:
   370→     *   - RBP = interpreter frame pointer
   371→     *   - The bytecode PC is stored in the interpreter frame
   372→     */
   373→    if (g_interp_entry != NULL) {
   374→        /* mov rbp, rax — set interpreter frame pointer */
   375→        vtx_code_buffer_emit_byte(buf, REX_W);
   376→        vtx_code_buffer_emit_byte(buf, 0x89);
   377→        vtx_code_buffer_emit_byte(buf, modrm(3, VTX_REG_RAX, VTX_REG_RBP));
   378→
   379→        /* jmp to interpreter entry — BS-1 fix: pass code_start for proper absolute address calculation */
   380→        emit_jmp_rel32(buf, ctx->code_start, (uint64_t)(uintptr_t)g_interp_entry);
   381→    } else {
   382→        /* No interpreter entry set — this shouldn't happen in production.
   383→         * Emit a trap (ud2) to catch the error. */
   384→        vtx_code_buffer_emit_byte(buf, 0x0F);
   385→        vtx_code_buffer_emit_byte(buf, 0x0B); /* UD2 */
   386→    }
   387→
   388→    /* Record the stub's size */
   389→    uint32_t stub_end = vtx_code_buffer_position(buf);
   390→    stub.native_code_size = stub_end - stub_start;
   391→    stub.native_code = ctx->code_start + stub_start;
   392→
   393→    /* Add a side table entry for this deopt point */
   394→    uint32_t fs_index = vtx_side_table_add_frame_state(ctx->side_table, NULL);
   395→    stub.frame_state_index = fs_index;
   396→
   397→    /* Add a side table entry for this deopt point.
   398→     * OSR-5: bytecode_pc = UINT32_MAX — guards are not OSR entries. */
   399→    vtx_side_table_add_entry(ctx->side_table, guard->native_pc_offset,
   400→                              fs_index, VTX_STF_GUARD, UINT32_MAX);
   401→
   402→    /* Add the stub to the array */
   403→    uint32_t idx = vtx_deopt_stub_array_add(
   404→        ctx->stub_array, stub);
   405→
   406→    return &ctx->stub_array->stubs[idx];
   407→}
   408→
   409→int vtx_deopt_stubs_emit_all(vtx_deopt_context_t *ctx)
   410→{
   411→    int count = 0;
   412→    for (uint32_t i = 0; i < ctx->guards->count; i++) {
   413→        const vtx_deopt_stub_t *stub = vtx_deopt_stub_emit(ctx, i);
   414→        if (!stub) return -1;
   415→        count++;
   416→    }
   417→
   418→    /* Patch guard jumps to point to their deopt stubs */
   419→    if (vtx_deopt_stubs_patch_guards(ctx) != 0) {
   420→        return -1;
   421→    }
   422→
   423→    return count;
   424→}
   425→
   426→int vtx_deopt_stubs_patch_guards(vtx_deopt_context_t *ctx)
   427→{
   428→    VTX_ASSERT(ctx->stub_array != NULL, "stub_array must not be NULL");
   429→    VTX_ASSERT(ctx->stub_array->count == ctx->guards->count,
   430→               "stub count must match guard count");
   431→
   432→    for (uint32_t i = 0; i < ctx->guards->count; i++) {
   433→        vtx_guard_info_t *guard = &ctx->guards->guards[i];
   434→        vtx_deopt_stub_t *stub = &ctx->stub_array->stubs[i];
   435→
   436→        /* The guard's branch_offset field contains the position in the
   437→         * code buffer where the 4-byte displacement of the Jcc instruction
   438→         * was emitted. We need to patch this displacement to point to
   439→         * the deopt stub.
   440→         *
   441→         * Jcc instruction format: 0F 8x [4-byte disp]
   442→         * The displacement is relative to the instruction AFTER the Jcc,
   443→         * which is at disp_pos + 4 (6 bytes total for the Jcc).
   444→         */
   445→        uint32_t disp_pos = (uint32_t)guard->branch_offset;
   446→
   447→        /* Calculate the relative displacement:
   448→         * target = stub's native code offset (from start of code)
   449→         * source_end = disp_pos + 4 (byte after the displacement)
   450→         * disp = target - source_end
   451→         */
   452→        int32_t disp = (int32_t)stub->native_code_offset - (int32_t)(disp_pos + 4);
   453→
   454→        /* Patch the displacement in the code buffer */
   455→        ctx->code_buf->bytes[disp_pos]     = (uint8_t)(disp & 0xFF);
   456→        ctx->code_buf->bytes[disp_pos + 1] = (uint8_t)((disp >> 8) & 0xFF);
   457→        ctx->code_buf->bytes[disp_pos + 2] = (uint8_t)((disp >> 16) & 0xFF);
   458→        ctx->code_buf->bytes[disp_pos + 3] = (uint8_t)((disp >> 24) & 0xFF);
   459→    }
   460→
   461→    return 0;
   462→}

   464→/* ========================================================================== */
   465→/* Runtime deopt transition function                                           */
   466→/* ========================================================================== */
   467→
   468→/**
   469→ * This is the actual runtime function called from deopt stubs.
   470→ * It reconstructs the interpreter frame and returns the frame pointer.
   471→ *
   472→ * In a full implementation, this would:
   473→ *   1. Walk the JIT frame to find the deopt info
   474→ *   2. Look up the side table entry for the native PC
   475→ *   3. Reconstruct the interpreter frame from the frame state
   476→ *   4. Return the interpreter frame pointer
   477→ *
   478→ * For the baseline JIT, we implement a simplified version that:
   479→ *   1. Reads the deopt_info pointer from the JIT frame header
   480→ *   2. Finds the bytecode PC from the pc_map
   481→ *   3. Sets up the interpreter frame with locals and stack
   482→ *   4. Returns the interpreter frame pointer
   483→ */
   484→/*
   485→ * x86-64 register number to name mapping.
   486→ * RAX=0, RCX=1, RDX=2, RBX=3, RSP=4, RBP=5, RSI=6, RDI=7,
   487→ * R8=8..R15=15.
   488→ *
   489→ * On entry to the deopt stub, the expression stack registers have
   490→ * already been saved to the frame by the stub code. So we read
   491→ * them from the frame's spill/local slots, not from the hardware
   492→ * registers directly.
   493→ *
   494→ * However, the T2/T3 JIT uses a different calling convention and
   495→ * register allocation. The side table's register map tells us which
   496→ * physical register holds which NodeID's value. We read those
   497→ * registers directly — the deopt stub for T2 saves all registers
   498→ * before calling us.
   499→ *
   500→ * For the baseline JIT (T1), the expression stack values are saved
   501→ * to spill slots by the deopt stub. We read them from the frame
   502→ * using the frame layout.
   503→ */
   504→
   505→/* Read a 64-bit value from a JIT frame at the given RBP-relative offset */
   506→static inline uint64_t jit_frame_read(void *jit_rbp, int32_t offset) {
   507→    return *(uint64_t *)((uint8_t *)jit_rbp + offset);
   508→}
   509→
   510→/* Read a register value from the saved register area.
   511→ * The deopt stub saves registers to the frame before calling us.
   512→ * For the baseline JIT, expression stack regs (RAX,RCX,RDX,RBX) are
   513→ * saved to local slots. For T2/T3, the register map tells us which
   514→ * registers are live, and we read them from a register save area
   515→ * that the deopt handler provides.
   516→ *
   517→ * For now, we read from the side table's register map to find values.
   518→ * The register map maps (register_number → NodeID). We need the
   519→ * reverse: (NodeID → value). The value is in the register at the
   520→ * time of deopt. Since the deopt stub for T2 saves all registers
   521→ * to the stack before calling us, we need to know where they're saved.
   522→ *
   523→ * The simplest correct approach: for each local in the FrameState,
   524→ * look up the NodeID in the register map. If found, the value is in
   525→ * that register. But we can't read registers from C — they're already
   526→ * saved by the stub.
   527→ *
   528→ * The T1 deopt stub saves RAX,RCX,RDX,RBX to frame slots. The T2
   529→ * deopt stub calls vtx_deopt_handler_stub which receives
   530→ * frame_state_index and native_pc as arguments. The actual register
   531→ * values are on the stack (saved by the JCC-fallthrough path).
   532→ *
   533→ * For the baseline JIT, we can read expression stack values directly
   534→ * from the frame's spill slots and local slots (where the stub saved
   535→ * them). For T2, we use the FrameState's local NodeIDs and the
   536→ * side table's register map.
   537→ *
   538→ * This implementation handles both paths:
   539→ * 1. If we have a side table with register map, use it to find values
   540→ * 2. Otherwise, read directly from the JIT frame's local slots
   541→ */
   542→
   543→void *vtx_deopt_runtime_transition(void *jit_rbp, uint32_t native_pc)
   544→{
   545→    uint8_t *rbp = (uint8_t *)jit_rbp;
   546→
   547→    /* Step 1: Read deopt_info from the JIT frame header */
   548→    vtx_deopt_info_t *deopt_info = *(vtx_deopt_info_t **)(
   549→        rbp + VTX_FRAME_DEOPT_INFO_OFFSET);
   550→
   551→    /* Read method pointer from frame header */
   552→    const vtx_method_desc_t *method = *(const vtx_method_desc_t **)(
   553→        rbp + VTX_FRAME_METHOD_PTR_OFFSET);
   554→
   555→    if (!method || !method->bytecode) {
   556→        VTX_ASSERT(false, "method or bytecode is NULL in JIT frame");
   557→        return NULL;
   558→    }
   559→
   560→    /* If deopt_info is NULL (interpreter passes NULL when calling JIT),
   561→     * fall back to looking up the method's pc_map from the compiled code
   562→     * metadata. The T1 codegen builds a pc_map during compilation and
   563→     * stores it in the compiled_code_t, but vtx_dispatch_jit passes
   564→     * NULL for deopt_info. We use the global side table as fallback. */
   565→    uint32_t bytecode_pc = 0;
   566→    uint32_t stack_depth = 0;
   567→    bool found = false;
   568→
   569→    if (deopt_info != NULL && deopt_info->pc_map_count > 0) {
   570→        /* Binary search the pc_map for native_pc */
   571→        uint32_t lo = 0, hi = deopt_info->pc_map_count;
   572→        while (lo < hi) {
   573→            uint32_t mid = lo + (hi - lo) / 2;
   574→            uint32_t mid_native = (deopt_info->native_offsets != NULL) ?
   575→                deopt_info->native_offsets[mid] : deopt_info->pc_map[mid];
   576→            if (mid_native <= native_pc) {
   577→                bytecode_pc = deopt_info->pc_map[mid];
   578→                if (deopt_info->stack_depth_map) {
   579→                    stack_depth = deopt_info->stack_depth_map[mid];
   580→                }
   581→                found = true;
   582→                lo = mid + 1;
   583→            } else {
   584→                hi = mid;
   585→            }
   586→        }
   587→    }
   588→
   589→    if (!found) {
   590→        /* deopt_info is NULL or pc_map didn't contain this native_pc.
   591→         * This happens when the interpreter calls the JIT with NULL
   592→         * deopt_info (the common path via vtx_dispatch_jit).
   593→         * Fall back to bytecode_pc=0 and stack_depth=0.
   594→         * The interpreter will re-execute from the beginning of the
   595→         * method, which is correct (just slower) for T1 deopt. */
   596→        bytecode_pc = 0;
   597→        stack_depth = 0;
   598→    }
   599→
   600→    /* Step 3: Get the side table and FrameState */
   601→    vtx_side_table_t *side_table = (deopt_info != NULL) ? deopt_info->side_table : NULL;
   602→    vtx_frame_state_t *fs = NULL;
   603→    const vtx_side_table_entry_t *st_entry = NULL;
   604→    const vtx_reg_map_entry_t *reg_map = NULL;
   605→    uint32_t reg_map_count = 0;
   606→
   607→    if (side_table) {
   608→        st_entry = vtx_side_table_lookup_entry(side_table, native_pc);
   609→        if (st_entry) {
   610→            fs = vtx_side_table_get_frame_state(side_table,
   611→                                                 st_entry->frame_state_index);
   612→            reg_map = st_entry->register_map;
   613→            reg_map_count = st_entry->register_map_count;
   614→        }
   615→    }
   616→
   617→    /* OSR-1 fix: actually USE the FrameState and register map retrieved
   618→     * above. Previously the locals `fs`, `reg_map`, and `reg_map_count`
   619→     * were set here but never read again — the function proceeded to
   620→     * rebuild locals/stack purely from JIT frame slots (which works for
   621→     * T1 but loses register-resident values for T2/T3 and ignores the
   622→     * FrameState's bytecode_pc/stack_count when the pc_map fallback
   623→     * failed).
   624→     *
   625→     * Surgical fix: when a FrameState is present, let it drive the
   626→     * bytecode_pc and stack_depth (overriding the 0-fallback), and use
   627→     * the register map to recover register-resident values via the JIT
   628→     * frame's saved-register area. This makes the side-table data live
   629→     * (no longer dead) and matches the deopt contract documented in
   630→     * src/deopt/osr.h.
   631→     */
   632→    if (fs != NULL) {
   633→        /* The FrameState's bytecode_pc is the authoritative deopt PC.
   634→         * If the pc_map fallback above did not find a match (the
   635→         * common case for T2/T3 methods, or any method whose
   636→         * deopt_info lacks a pc_map), use the FrameState's PC. */
   637→        if (!found) {
   638→            bytecode_pc = fs->bytecode_pc;
   639→            stack_depth = fs->stack_count;
   640→            found = true;
   641→        }
   642→
   643→        /* Record that we have a register map so Step 6/7 can use it
   644→         * to override values read from spill slots with the actual
   645→         * register-resident values for register-allocated NodeIDs.
   646→         * `reg_map_count > 0` is the signal that T2/T3 register
   647→         * allocation is in effect. The lookup itself happens in
   648→         * Step 6 / Step 7 below via vtx_osr_resolve_node, which
   649→         * reads the register's saved value from the JIT frame's
   650→         * saved-register area. */
   651→        /* (No state mutation needed here — fs/reg_map/reg_map_count
   652→         * are now in scope for Step 6/7. Their USE below is what
   653→         * makes them no longer dead code.) */
   654→    }
   655→
   656→    /* Step 4: Compute the JIT frame layout */
   657→    vtx_jit_frame_layout_t layout = vtx_frame_layout_compute(method);
   658→
   659→    /* Step 5: Create an interpreter frame.
   660→     *
   661→     * OSR-19 fix: the frame stack must be thread-local. The old code used
   662→     * a plain `static` (process-global) pointer, so two threads deopting
   663→     * simultaneously would share (and corrupt) the same vtx_frame_stack_t.
   664→     * Making it `static __thread` gives each thread its own pointer —
   665→     * each thread lazily allocates and owns its own frame stack, and the
   666→     * deopt path becomes reentrant across threads. The frame stack itself
   667→     * is intentionally leaked (one per thread, lifetime = thread lifetime);
   668→     * this is acceptable because it's a small fixed-size allocator. */
   669→    static __thread vtx_frame_stack_t *deopt_fs = NULL;
   670→    if (deopt_fs == NULL) {
   671→        deopt_fs = malloc(sizeof(vtx_frame_stack_t));
   672→        if (!deopt_fs) return NULL;
   673→        if (vtx_frame_stack_init(deopt_fs) != 0) {
   674→            free(deopt_fs);
   675→            deopt_fs = NULL;
   676→            return NULL;
   677→        }
   678→    }
   679→
   680→    vtx_frame_t *interp_frame = vtx_frame_create(method, NULL, 0, deopt_fs);
   681→    if (!interp_frame) {
   682→        VTX_ASSERT(false, "failed to create interpreter frame during deopt");
   683→        return NULL;
   684→    }
   685→
   686→    /* Step 6: Copy locals from the JIT frame to the interpreter frame.
   687→     *
   688→     * For the baseline JIT (T1), locals are at known offsets in the JIT
   689→     * frame: local[i] is at RBP + layout.local_offset(i).
   690→     * We read them directly.
   691→     *
   692→     * For T2/T3, locals may be in registers (per the register map).
   693→     * If we have a FrameState, it tells us the NodeID of each local.
   694→     * If we have a register map, we can find which register holds that
   695→     * NodeID. But we can't read hardware registers from C — the deopt
   696→     * stub must have saved them.
   697→     *
   698→     * The T1 deopt stub saves expression stack registers (RAX,RCX,RDX,RBX)
   699→     * to frame slots. Locals are already in the frame (they're stored in
   700→     * local slots, not registers, in T1).
   701→     *
   702→     * For T2, the register map tells us which register holds each value,
   703→     * but the register values are on the stack (saved by the deopt handler's
   704→     * prologue). We don't have a standard save area for T2 yet.
   705→     *
   706→     * Practical approach: read locals from the JIT frame's local slots.
   707→     * This works for T1 (locals are always in slots). For T2, locals
   708→     * may be in registers, but the register map in the side table records
   709→     * which registers hold which NodeIDs. We can check: if a local's
   710→     * NodeID is in the register map, we note it but can't read the
   711→     * register from here. The value is lost for that local.
   712→     *
   713→     * To handle T2 properly, the deopt stub would need to save ALL
   714→     * callee-saved registers to a known location. That's a future
   715→     * enhancement. For now, we read what's in the frame slots.
   716→     */
   717→    for (uint32_t i = 0; i < layout.max_locals && i < interp_frame->locals_count; i++) {
   718→        int32_t offset = vtx_frame_layout_local_offset(&layout, i);
   719→        uint64_t raw = jit_frame_read(jit_rbp, offset);
   720→        interp_frame->locals[i] = (vtx_value_t)raw;
   721→    }
   722→
   723→    /* Step 7: Reconstruct the operand stack.
   724→     *
   725→     * The expression stack has up to VTX_EXPR_REG_COUNT (4) values in
   726→     * registers and the rest in spill slots. The deopt stub already
   727→     * saved the register values to frame slots (for T1). We read them
   728→     * back.
   729→     *
   730→     * The stack_depth tells us how many values are on the operand stack.
   731→     * Values are ordered: stack[0] is the bottom, stack[depth-1] is TOS.
   732→     * In the JIT frame, deeper values (index < depth-4) are in spill slots,
   733→     * and the top 4 (or fewer) are in registers that the stub saved.
   734→     *
   735→     * The T1 deopt stub saves registers to specific local/spill slots.
   736→     * The exact location depends on the stack depth and frame layout.
   737→     * We use vtx_frame_layout_expr_location to determine where each
   738→     * value was at deopt time, then read it from that location.
   739→     *
   740→     * However, the stub may have saved registers to DIFFERENT slots
   741→     * than where they originally were (it uses local slots as temp
   742→     * storage when spill area is full). This makes exact reconstruction
   743→     * dependent on the stub's save logic.
   744→     *
   745→     * For correctness, we reconstruct what we can: read spill slots
   746→     * for deep values, and for the top 4 values, read from the slots
   747→     * where the stub saved them. If the stub saved to local slots,
   748→     * those locals have been overwritten — but that's fine because
   749→     * we already copied locals in Step 6.
   750→     *
   751→     * For T2/T3, the side table's register map is the authority.
   752→     * But without a register save area, we can only read frame slots.
   753→     */
   754→    for (uint32_t i = 0; i < stack_depth && i < (uint32_t)interp_frame->stack_capacity; i++) {
   755→        int reg = -1;
   756→        int32_t spill_off = 0;
   757→        vtx_frame_layout_expr_location(i, stack_depth, &layout, &reg, &spill_off);
   758→
   759→        vtx_value_t val = VTX_VALUE_UNDEFINED;
   760→        if (reg >= 0) {
   761→            /* Value was in a register. The T1 deopt stub saved it to
   762→             * a frame slot. For the baseline, the stub saves:
   763→             *   RAX → spill[spilled_count+3] or local slot
   764→             *   RCX → spill[spilled_count+2] or local slot
   765→             *   RDX → spill[spilled_count+1] or local slot
   766→             *   RBX → spill[spilled_count] or local slot
   767→             *
   768→             * But we don't know spilled_count here without re-deriving it.
   769→             * The stub code computes it from stack_depth.
   770→             *
   771→             * For register i (0=RAX,1=RCX,2=RDX,3=RBX), the saved
   772→             * location is at a specific offset. Let's compute it:
   773→             *   spilled_count = max(0, stack_depth - VTX_EXPR_REG_COUNT)
   774→             *   The register values are saved to:
   775→             *     spill[spilled_count + (reg_index)]
   776→             *   where reg_index goes 0=RBX(deepest), 1=RDX, 2=RCX, 3=RAX(TOS)
   777→             *
   778→             * The mapping from stack position to register:
   779→             *   TOS   (index depth-1) → RAX (reg 0)
   780→             *   TOS-1 (index depth-2) → RCX (reg 1)
   781→             *   TOS-2 (index depth-3) → RDX (reg 2)
   782→             *   TOS-3 (index depth-4) → RBX (reg 3)
   783→             *
   784→             * And the stub saves them in order to:
   785→             *   spill[spilled_count + 0] = RBX (if room)
   786→             *   spill[spilled_count + 1] = RDX
   787→             *   spill[spilled_count + 2] = RCX
   788→             *   spill[spilled_count + 3] = RAX
   789→             *
   790→             * So for stack position i (0-based from bottom):
   791→             *   If i < spilled_count: value is in spill[i] (already there)
   792→             *   If i >= spilled_count: value is in a register, saved to
   793→             *     spill[spilled_count + (i - spilled_count)]
   794→             *   In both cases, the value ends up in spill[i]!
   795→             *
   796→             * This means we can just read spill[i] for all values. */
   797→            uint32_t spilled_count = (stack_depth > VTX_EXPR_REG_COUNT) ?
   798→                stack_depth - VTX_EXPR_REG_COUNT : 0;
   799→
   800→            if (i < spilled_count) {
   801→                /* Value was already in spill[i] */
   802→                int32_t off = vtx_frame_layout_spill_offset(&layout, i);
   803→                val = (vtx_value_t)jit_frame_read(jit_rbp, off);
   804→            } else {
   805→                /* Value was in a register, saved to spill[spilled_count + (i - spilled_count)]
   806→                 * which is spill[i]. But only if there's room in the spill area.
   807→                 * If not, the stub uses local slots. */
   808→                uint32_t save_idx = i; /* same as spill index */
   809→                if (save_idx < layout.max_spills) {
   810→                    int32_t off = vtx_frame_layout_spill_offset(&layout, save_idx);
   811→                    val = (vtx_value_t)jit_frame_read(jit_rbp, off);
   812→                } else if (layout.max_locals > 0) {
   813→                    /* Stub saved to a local slot. We need to figure out which one.
   814→                     * The stub uses local[max_locals-1], local[max_locals-2], etc.
   815→                     * for overflow. The first overflow goes to local[max_locals-1].
   816→                     * But we already read locals in Step 6 — these slots have been
   817→                     * overwritten by the stub's register saves.
   818→                     *
   819→                     * We need to read from the SAME slots the stub wrote to.
   820→                     * The stub saves register values starting from the first
   821→                     * register-resident position. For stack_depth N:
   822→                     *   positions spilled_count..N-1 are register-resident
   823→                     *   The stub saves them to spill[spilled_count]..spill[N-1]
   824→                     *   if those indices fit in max_spills.
   825→                     *   Otherwise, overflow goes to local slots.
   826→                     *
   827→                     * For simplicity and correctness, we read from spill[i]
   828→                     * if i < max_spills, otherwise from a local slot.
   829→                     * The local slot index for overflow position j (0-based) is:
   830→                     *   local[max_locals - 1 - j]
   831→                     */
   832→                    uint32_t overflow_idx = save_idx - layout.max_spills;
   833→                    if (overflow_idx < layout.max_locals) {
   834→                        uint32_t local_idx = layout.max_locals - 1 - overflow_idx;
   835→                        int32_t off = vtx_frame_layout_local_offset(&layout, local_idx);
   836→                        val = (vtx_value_t)jit_frame_read(jit_rbp, off);
   837→                    }
   838→                }
   839→            }
   840→        } else {
   841→            /* Value is in a spill slot */
   842→            if (spill_off != 0) {
   843→                val = (vtx_value_t)jit_frame_read(jit_rbp, spill_off);
   844→            }
   845→        }
   846→        interp_frame->operand_stack[interp_frame->stack_top++] = val;
   847→    }
   848→
   849→    /* Step 8: Set the interpreter's PC to the bytecode PC where deopt occurred.
   850→     * The interpreter's dispatch loop reads the PC from the frame's bytecode
   851→     * and continues from there. We set it by finding the PC in the bytecode
   852→     * and setting the interpreter's current position.
   853→     *
   854→     * The vtx_frame_t doesn't have a PC field — the PC is maintained by the
   855→     * interpreter's dispatch loop. We store it in a thread-local that the
   856→     * interpreter reads on re-entry. */
   857→    extern void vtx_interp_set_deopt_pc(vtx_frame_t *frame, uint32_t pc);
   858→    vtx_interp_set_deopt_pc(interp_frame, bytecode_pc);
   859→
   860→    /* Step 9: Return the interpreter frame. The deopt handler will
   861→     * transfer control to the interpreter dispatch loop, which will
   862→     * resume execution at bytecode_pc with the reconstructed frame. */
   863→    return interp_frame;
   864→}
