    14→/**
    15→ * VORTEX Optimization Pipeline Driver
    16→ *
    17→ * Wires together ALL optimization passes in the correct order,
    18→ * producing optimized x86-64 machine code from a Sea-of-Nodes IR graph.
    19→ *
    20→ * Pipeline tiers:
    21→ *   T1 — Baseline: GVN + DCE (quick, no expensive analysis)
    22→ *   T2 — Optimizing: Full pipeline with inlining, PEA, LICM, bounds check
    23→ *   T3 — Speculative: T2 + speculative guards + vectorization hints
    24→ *
    25→ * Pass ordering rationale:
    26→ *   1. GVN first — enables SCCP and DCE by merging redundant nodes
    27→ *   2. SCCP — constant propagation simplifies conditions for DCE/inlining
    28→ *   3. DCE — remove dead code exposed by GVN/SCCP
    29→ *   4. Inlining — expand call sites using ML model
    30→ *   5. GVN+SCCP+DCE again — clean up after inlining exposes redundancies
    31→ *   6. PEA — escape analysis + scalar replacement of allocations
    32→ *   7. DCE — remove dead allocations exposed by scalar replacement
    33→ *   8. Schedule — convert SoN back to basic blocks
    34→ *   9. LICM — hoist loop-invariant code (needs schedule for loop structure)
    35→ *  10. Bounds check elimination (needs schedule for dominance info)
    36→ *  11. Instruction selection — map SoN nodes to x86-64 instructions
    37→ *  12. Register allocation — assign physical registers
    38→ *  13. Code emission — emit x86-64 machine code
    39→ */

    87→
    88→static inline int64_t now_ns(void)
    89→{
    90→    struct timespec ts;
    91→    clock_gettime(CLOCK_MONOTONIC, &ts);
    92→    return (int64_t)ts.tv_sec * 1000000000LL + (int64_t)ts.tv_nsec;
    93→}
    94→
    95→static inline int64_t elapsed_ns(int64_t start)
    96→{
    97→    return now_ns() - start;
    98→}

   104→/**
   105→ * Conditionally verify the graph between passes.
   106→ * Only active when config->run_verify is true.
   107→ * Returns 0 on success (or verification disabled), -1 on verification failure.
   108→ */
   109→static int verify_between_passes(const vtx_graph_t *graph,
   110→                                  const vtx_pipeline_config_t *config,
   111→                                  const char *pass_name)
   112→{
   113→    if (!config->run_verify) {
   114→        return 0;
   115→    }
   116→    if (!vtx_verify_graph(graph)) {
   117→        fprintf(stderr, "[pipeline] VERIFICATION FAILED after %s\n", pass_name);
   118→        return -1;
   119→    }
   120→    return 0;
   121→}

   280→/* ========================================================================== */
   281→/* GVN pass (iterative)                                                        */
   282→/* ========================================================================== */
   283→
   284→/**
   285→ * Run GVN for up to max_iter iterations, or until no more nodes are merged.
   286→ * Returns total nodes merged across all iterations.
   287→ */
   288→static uint32_t run_gvn_pass(vtx_graph_t *graph,
   289→                              int max_iter,
   290→                              int64_t *time_ns)
   291→{
   292→    int64_t start = now_ns();
   293→    uint32_t total_merged = 0;
   294→
   295→    for (int i = 0; i < max_iter; i++) {
   296→        uint32_t merged = vtx_gvn_run(graph);
   297→        total_merged += merged;
   298→        if (merged == 0) {
   299→            break;  /* fixed point reached */
   300→        }
   301→    }
   302→
   303→    *time_ns = elapsed_ns(start);
   304→    return total_merged;
   305→}
   306→
   307→/* ========================================================================== */
   308→/* SCCP pass (iterative)                                                       */
   309→/* ========================================================================== */
   310→
   311→/**
   312→ * Run SCCP for up to max_iter iterations, or until no more constants are found.
   313→ * Returns total constants propagated across all iterations.
   314→ */
   315→static uint32_t run_sccp_pass(vtx_graph_t *graph,
   316→                               int max_iter,
   317→                               int64_t *time_ns)
   318→{
   319→    int64_t start = now_ns();
   320→    uint32_t total_propagated = 0;
   321→
   322→    for (int i = 0; i < max_iter; i++) {
   323→        uint32_t propagated = vtx_constant_prop_run(graph);
   324→        total_propagated += propagated;
   325→        if (propagated == 0) {
   326→            break;  /* fixed point reached */
   327→        }
   328→    }
   329→
   330→    *time_ns = elapsed_ns(start);
   331→    return total_propagated;
   332→}
   333→
   334→/* ========================================================================== */
   335→/* DCE pass (iterative)                                                        */
   336→/* ========================================================================== */
   337→
   338→/**
   339→ * Run DCE for up to max_iter iterations, or until no more nodes are removed.
   340→ * Returns total nodes removed across all iterations.
   341→ */
   342→static uint32_t run_dce_pass(vtx_graph_t *graph,
   343→                              int max_iter,
   344→                              int64_t *time_ns,
   345→                              bool verify)
   346→{
   347→    int64_t start = now_ns();
   348→    uint32_t total_removed = 0;
   349→
   350→    for (int i = 0; i < max_iter; i++) {
   351→        uint32_t removed = vtx_dce_run(graph);
   352→        total_removed += removed;
   353→        if (removed == 0) {
   354→            break;  /* no more dead code */
   355→        }
   356→        /* Clean up dead node references so subsequent passes and verify
   357→         * don't see stale inputs pointing to dead nodes. */
   358→        vtx_node_table_clear_dead(&graph->node_table);
   359→    }
   360→    /* Final clear_dead after all DCE iterations */
   361→    vtx_node_table_clear_dead(&graph->node_table);
   362→
   363→    *time_ns = elapsed_ns(start);
   364→
   365→    if (verify) {
   366→        vtx_verify_graph_post_dce(graph);
   367→    }
   368→
   369→    return total_removed;
   370→}

   372→/* ========================================================================== */
   373→/* Inlining pass                                                               */
   374→/* ========================================================================== */
   375→
   376→/**
   377→ * Run the ML-guided inlining pass.
   378→ *
   379→ * Walks all CallStatic/CallVirtual/CallInterface nodes in the graph,
   380→ * computes features, runs GBDT inference, and inlines callees whose
   381→ * score exceeds VTX_INLINE_THRESHOLD.
   382→ *
   383→ * Returns the number of inlining decisions made (both inline and no-inline).
   384→ */
   385→static uint32_t run_inlining_pass(vtx_graph_t *graph,
   386→                                   const vtx_pipeline_config_t *config,
   387→                                   vtx_arena_t *arena,
   388→                                   int64_t *time_ns)
   389→{
   390→    int64_t start = now_ns();
   391→    uint32_t decisions = 0;
   392→    uint32_t inlined = 0;
   393→
   394→    /* Use the shared GBDT model if available, otherwise create a
   395→     * temporary one for this compilation (backward compatibility).
   396→     *
   397→     * BUGFIX (audit): The old code loaded the GBDT model on EVERY
   398→     * compile when shared_gbdt_model was NULL. The model is ~128KB
   399→     * and takes ~1ms to load. With 100 compiles/sec, that's 100ms/sec
   400→     * wasted on model loading.
   401→     *
   402→     * Fix: Use a static cached model so it's loaded only once per
   403→     * process. This is thread-safe because the GBDT model is read-only
   404→     * after initialization (inference doesn't mutate it). */
   405→    static vtx_gbdt_model_t *s_cached_model = NULL;
   406→    vtx_gbdt_model_t *model = config->shared_gbdt_model;
   407→    bool owns_temp_model = false;
   408→
   409→    if (model == NULL) {
   410→        /* Use the cached model if available */
   411→        if (s_cached_model != NULL) {
   412→            model = s_cached_model;
   413→        } else {
   414→            /* Load and cache the model */
   415→            vtx_gbdt_model_t *temp = (vtx_gbdt_model_t *)malloc(sizeof(vtx_gbdt_model_t));
   416→            if (!temp) {
   417→                *time_ns = elapsed_ns(start);
   418→                return 0;
   419→            }
   420→            if (vtx_gbdt_model_init(temp) != 0) {
   421→                free(temp);
   422→                *time_ns = elapsed_ns(start);
   423→                return 0;
   424→            }
   425→            if (vtx_gbdt_load_default_model(temp) != 0) {
   426→                vtx_gbdt_model_destroy(temp);
   427→                free(temp);
   428→                *time_ns = elapsed_ns(start);
   429→                return 0;
   430→            }
   431→            s_cached_model = temp;
   432→            model = temp;
   433→        }
   434→    }
   435→
   436→    /* Determine the effective inline size limit */
   437→    uint32_t effective_size_limit = (config && config->inline_size_limit > 0)
   438→                                    ? (uint32_t)config->inline_size_limit
   439→                                    : VTX_INLINE_SIZE_LIMIT;
   440→
   441→    /* Scan the graph for call nodes and make inlining decisions.
   442→     *
   443→     * We iterate over all nodes in the node table. For each call node
   444→     * (CallStatic, CallVirtual, CallInterface), we compute the feature
   445→     * vector, run GBDT inference, and inline if the score is high enough.
   446→     *
   447→     * Note: We collect call nodes first, then inline, because inlining
   448→     * modifies the graph and would invalidate iteration.
   449→     */
   450→    uint32_t node_count = graph->node_table.count;
   451→    vtx_nodeid_t *call_nodes = (vtx_nodeid_t *)vtx_arena_alloc(arena,
   452→        sizeof(vtx_nodeid_t) * (node_count + 1));
   453→    if (!call_nodes) {
   454→        if (owns_temp_model) {
   455→            vtx_gbdt_model_destroy(model);
   456→            free(model);
   457→        }
   458→        *time_ns = elapsed_ns(start);
   459→        return 0;
   460→    }
   461→
   462→    uint32_t call_count = 0;
   463→    for (uint32_t i = 0; i < node_count; i++) {
   464→        vtx_node_t *node = &graph->node_table.nodes[i];
   465→        if (node->dead) continue;
   466→
   467→        vtx_node_opcode_t op = node->opcode;
   468→        if (op == VTX_OP_CallStatic ||
   469→            op == VTX_OP_CallVirtual ||
   470→            op == VTX_OP_CallInterface) {
   471→            call_nodes[call_count++] = (vtx_nodeid_t)i;
   472→        }
   473→    }
   474→
   475→    /* Attempt inlining for each call site.
   476→     * If a callee graph lookup callback is provided, we look up the
   477→     * callee graph, check can_inline(), and call vtx_inline_transform(). */
   478→    for (uint32_t c = 0; c < call_count; c++) {
   479→        vtx_nodeid_t call_id = call_nodes[c];
   480→        vtx_node_t *call_node = vtx_graph_node(graph, call_id);
   481→        if (!call_node || call_node->dead) continue;
   482→
   483→        /* Look up the callee graph via the callback, if provided.
   484→         * We do this BEFORE feature extraction so the extractor can use
   485→         * the callee graph for precise features (instruction count, has_loops,
   486→         * has_try_catch, allocates, calls_virtual). */
   487→        const vtx_graph_t *callee_graph = NULL;
   488→        if (config && config->callee_lookup) {
   489→            callee_graph = config->callee_lookup(
   490→                call_node->method_index,
   491→                config->callee_lookup_context);
   492→        }
   493→
   494→        /* Build the feature extraction context from real data.
   495→         *
   496→         * BUGFIX (audit #8): The old code hand-filled the features struct
   497→         * with constants (callee_size=64, callee_is_hot=0.5, etc.), making
   498→         * the GBDT model receive essentially constant inputs and produce
   499→         * essentially constant output. The "ML inliner" was a heuristic
   500→         * in disguise. Now we call the real vtx_features_extract() with
   501→         * a properly populated context. */
   502→        vtx_inline_context_t ctx;
   503→        memset(&ctx, 0, sizeof(ctx));
   504→        ctx.caller_bytecode_size = (config && config->method && config->method->bytecode)
   505→                                    ? (uint32_t)config->method->bytecode->length : 0;
   506→        ctx.call_depth = 1;
   507→        ctx.inline_history = 0;
   508→        ctx.callee_bytecode_size = 64;  /* conservative default if unknown */
   509→        ctx.callee_method_id = call_node->method_index;
   510→        ctx.callee_deopt_count = 0;
   511→        ctx.callee_invocation_count = 0;
   512→        ctx.caller_node_count = graph->node_table.count;
   513→        ctx.callee_graph = callee_graph;
   514→
   515→        /* Extract real features from the graph, call node, and context.
   516→         * Profile data is not available in the pipeline config (T2 compiles
   517→         * without profile), so pass NULL — the extractor falls back to
   518→         * conservative estimates for profile-dependent features. */
   519→        vtx_inline_features_t features = vtx_features_extract(
   520→            graph, call_id, NULL, &ctx);
   521→
   522→        /* Run GBDT inference */
   523→        double score = vtx_gbdt_infer(model, &features);
   524→        decisions++;
   525→
   526→        if (vtx_gbdt_should_inline(score)) {
   527→            if (callee_graph != NULL) {
   528→                /* Check if inlining is legal */
   529→                if (callee_graph->node_table.count <= effective_size_limit &&
   530→                    vtx_inline_can_inline(graph, call_id, callee_graph, 1)) {
   531→                    /* Perform the inlining transform */
   532→                    vtx_inline_result_t iresult = vtx_inline_transform(
   533→                        graph, call_id, callee_graph, arena);
   534→                    if (iresult.success) {
   535→                        inlined++;
   536→                    }
   537→                }
   538→            }
   539→            /* If no callee graph is available, the decision is recorded
   540→             * but the actual transform is deferred until a callee graph
   541→             * is provided (e.g., via a method registry at runtime). */
   542→        }
   543→    }
   544→
   545→    if (owns_temp_model) {
   546→        vtx_gbdt_model_destroy(model);
   547→        free(model);
   548→    }
   549→    *time_ns = elapsed_ns(start);
   550→    return (decisions << 16) | inlined;  /* pack decisions + inlined count */
   551→}
   552→
   553→/* ========================================================================== */
   554→/* PEA pass (escape analysis + scalar replacement + materialization)            */
   555→/* ========================================================================== */
   556→
   557→/**
   558→ * Run the Partial Escape Analysis pipeline:
   559→ *   1. Flow-sensitive escape analysis
   560→ *   2. Cross-object scalar replacement
   561→ *   3. Object materialization at escape/deopt points
   562→ *
   563→ * Returns the number of allocations eliminated (scalar-replaced).
   564→ */
   565→static uint32_t run_pea_pass(vtx_graph_t *graph,
   566→                              vtx_arena_t *arena,
   567→                              int64_t *time_ns)
   568→{
   569→    int64_t start = now_ns();
   570→    uint32_t allocs_eliminated = 0;
   571→
   572→    /* Step 1: Run escape analysis */
   573→    vtx_pea_analysis_t *analysis = vtx_pea_run(graph, arena);
   574→    if (!analysis) {
   575→        *time_ns = elapsed_ns(start);
   576→        return 0;
   577→    }
   578→
   579→    /* Step 2: Cross-object scalar replacement */
   580→    vtx_cross_sr_result_t *sr_result = vtx_cross_object_sr_run(graph, analysis, arena);
   581→    if (sr_result) {
   582→        allocs_eliminated = sr_result->allocs_replaced;
   583→    }
   584→
   585→    /* Step 3: Run virtual object tracking (rewrites field accesses to
   586→     * locals, marks StoreField nodes as dead). Must run before
   587→     * materialization so that materialize reads field values from
   588→     * the virtual field maps instead of dead StoreField nodes. */
   589→    vtx_virtual_result_t *virtual_result = vtx_virtual_run(graph, analysis, arena);
   590→
   591→    /* Step 4: Materialize objects at escape/deopt points */
   592→    vtx_materialize_result_t *mat_result = vtx_materialize_run(graph, analysis, virtual_result, arena);
   593→    /* mat_result tracks how many objects needed materialization; we don't
   594→     * count those as "eliminated" since they still exist on the heap. */
   595→    (void)mat_result;
   596→
   597→    *time_ns = elapsed_ns(start);
   598→    return allocs_eliminated;
   599→}
   600→
   601→/* ========================================================================== */
   602→/* Schedule pass                                                               */
   603→/* ========================================================================== */
   604→
   605→/**
   606→ * Run the SoN -> basic block scheduler.
   607→ * Returns 0 on success, -1 on failure.
   608→ */
   609→static int run_schedule_pass(vtx_graph_t *graph,
   610→                              vtx_arena_t *arena,
   611→                              vtx_schedule_t *schedule,
   612→                              int64_t *time_ns)
   613→{
   614→    int64_t start = now_ns();
   615→    int rc = vtx_schedule_run(graph, arena, schedule);
   616→    *time_ns = elapsed_ns(start);
   617→    return rc;
   618→}
   619→
   620→/* ========================================================================== */
   621→/* LICM pass                                                                   */
   622→/* ========================================================================== */
   623→
   624→/**
   625→ * Run Loop-Invariant Code Motion.
   626→ * Returns the number of nodes hoisted.
   627→ */
   628→static uint32_t run_licm_pass(vtx_graph_t *graph,
   629→                               const vtx_schedule_t *schedule,
   630→                               vtx_arena_t *arena,
   631→                               int64_t *time_ns)
   632→{
   633→    int64_t start = now_ns();
   634→    int hoisted = vtx_licm_run(graph, schedule, arena);
   635→    *time_ns = elapsed_ns(start);
   636→    return (hoisted >= 0) ? (uint32_t)hoisted : 0;
   637→}
   638→
   639→/* ========================================================================== */
   640→/* Bounds check elimination pass                                                */
   641→/* ========================================================================== */
   642→
   643→/**
   644→ * Run bounds check elimination.
   645→ * Returns the number of guards eliminated.
   646→ */
   647→static uint32_t run_bounds_check_pass(vtx_graph_t *graph,
   648→                                       const vtx_schedule_t *schedule,
   649→                                       vtx_arena_t *arena,
   650→                                       int64_t *time_ns)
   651→{
   652→    int64_t start = now_ns();
   653→    int eliminated = vtx_bounds_check_run(graph, schedule, arena);
   654→    *time_ns = elapsed_ns(start);
   655→    return (eliminated >= 0) ? (uint32_t)eliminated : 0;
   656→}
   657→
   658→/* ========================================================================== */
   659→/* Loop specialization/vectorization pass                                        */
   660→/* ========================================================================== */
   661→
   662→/**
   663→ * Run loop specialization and vectorization.
   664→ *
   665→ * Finds all LoopBegin nodes, checks if they are vectorizable using
   666→ * the loop_spec analyzer, and transforms vectorizable loops by
   667→ * replacing scalar operations with SIMD-wide operations.
   668→ *
   669→ * Returns the number of loops vectorized.
   670→ */
   671→static uint32_t run_loop_spec_pass(vtx_graph_t *graph,
   672→                                    vtx_arena_t *arena,
   673→                                    int64_t *time_ns)
   674→{
   675→    int64_t start = now_ns();
   676→    uint32_t vectorized = 0;
   677→
   678→    /* Initialize loop spec analyzer */
   679→    vtx_sota_loop_spec_t spec;
   680→    if (vtx_sota_loop_spec_init(&spec) != 0) {
   681→        *time_ns = elapsed_ns(start);
   682→        return 0;
   683→    }
   684→
   685→    /* Detect CPU features */
   686→    uint32_t cpu_features = vtx_sota_loop_detect_cpu_features();
   687→
   688→    /* Find all LoopBegin nodes in the graph */
   689→    vtx_node_table_t *table = &graph->node_table;
   690→
   691→    /* First pass: collect LoopBegin node IDs (iteration may be
   692→     * invalidated if we modify the graph during transformation) */
   693→    uint32_t loop_count = 0;
   694→    for (uint32_t i = 0; i < table->count; i++) {
   695→        vtx_node_t *node = &table->nodes[i];
   696→        if (!node->dead && node->opcode == VTX_OP_LoopBegin) {
   697→            loop_count++;
   698→        }
   699→    }
   700→
   701→    if (loop_count == 0) {
   702→        vtx_sota_loop_spec_destroy(&spec);
   703→        *time_ns = elapsed_ns(start);
   704→        return 0;
   705→    }
   706→
   707→    vtx_nodeid_t *loop_nodes = (vtx_nodeid_t *)vtx_arena_alloc(
   708→        arena, sizeof(vtx_nodeid_t) * loop_count);
   709→    if (!loop_nodes) {
   710→        vtx_sota_loop_spec_destroy(&spec);
   711→        *time_ns = elapsed_ns(start);
   712→        return 0;
   713→    }
   714→
   715→    uint32_t idx = 0;
   716→    for (uint32_t i = 0; i < table->count; i++) {
   717→        vtx_node_t *node = &table->nodes[i];
   718→        if (!node->dead && node->opcode == VTX_OP_LoopBegin) {
   719→            loop_nodes[idx++] = node->id;
   720→        }
   721→    }
   722→
   723→    /* Second pass: analyze and transform each loop */
   724→    for (uint32_t l = 0; l < loop_count; l++) {
   725→        vtx_nodeid_t loop_id = loop_nodes[l];
   726→
   727→        /* Verify the node still exists and is a LoopBegin */
   728→        vtx_node_t *loop_node = vtx_node_get(table, loop_id);
   729→        if (!loop_node || loop_node->dead || loop_node->opcode != VTX_OP_LoopBegin) {
   730→            continue;
   731→        }
   732→
   733→        /* Analyze this loop for vectorization.
   734→         * We pass NULL for profile since profiling data may not be
   735→         * available at this point; the analyzer will fall back to
   736→         * heuristic trip count estimation. */
   737→        vtx_loop_spec_result_t result = vtx_sota_loop_spec_check(
   738→            &spec, NULL, graph, loop_id);
   739→
   740→        /* If the loop is vectorizable, transform it */
   741→        if (result.vectorizability >= VTX_LOOP_CAN_VECTORIZE_SSE2 &&
   742→            result.vector_width > 1) {
   743→            bool ok = vtx_sota_loop_spec_transform(
   744→                graph, loop_id, cpu_features, &result, arena);
   745→            if (ok) {
   746→                vectorized++;
   747→            }
   748→        }
   749→    }
   750→
   751→    vtx_sota_loop_spec_destroy(&spec);
   752→    *time_ns = elapsed_ns(start);
   753→    return vectorized;
   754→}
   755→
   756→/* ========================================================================== */
   757→/* Lowering pipeline (isel -> regalloc -> emit)                                */
   758→/* ========================================================================== */
   759→
   760→/**
   761→ * Run the complete lowering pipeline:
   762→ *   1. Instruction selection (SoN -> x86-64 virtual register instructions)
   763→ *   2. Register allocation (virtual -> physical registers)
   764→ *   3. Peephole optimization
   765→ *   4. Branch optimization
   766→ *   5. Code emission
   767→ *
   768→ * Returns 0 on success, -1 on failure.
   769→ * The emitted code is stored in result->native_code / native_size.
   770→ */
   771→static int run_lowering_pipeline(vtx_graph_t *graph,
   772→                                  const vtx_schedule_t *schedule,
   773→                                  vtx_arena_t *arena,
   774→                                  vtx_compile_result_t *result,
   775→                                  int64_t *time_ns,
   776→                                  uint32_t method_arg_count,
   777→                                  uint32_t method_max_locals)
   778→{
   779→    int64_t start = now_ns();
   780→
   781→    /* Step 1: Instruction selection */
   782→    vtx_inst_stream_t *inst_stream = vtx_isel_select(schedule, graph, arena);
   783→    if (!inst_stream) {
   784→        fprintf(stderr, "[pipeline] instruction selection failed\n");
   785→        *time_ns = elapsed_ns(start);
   786→        return -1;
   787→    }
   788→
   789→    /* Step 2: Register allocation */
   790→    vtx_regalloc_result_t *ra_result = vtx_regalloc_run(inst_stream, arena);
   791→    if (!ra_result) {
   792→        fprintf(stderr, "[pipeline] register allocation failed\n");
   793→        *time_ns = elapsed_ns(start);
   794→        return -1;
   795→    }
   796→
   797→    /* Step 3: Apply register allocation (rewrites virtual -> physical regs) */
   798→    if (vtx_regalloc_apply(inst_stream, ra_result, arena) != 0) {
   799→        fprintf(stderr, "[pipeline] register allocation apply failed\n");
   800→        *time_ns = elapsed_ns(start);
   801→        return -1;
   802→    }
   803→
   804→    /* Step 4: Peephole optimization on the allocated instructions */
   805→    vtx_peephole_optimize(inst_stream, ra_result);
   806→
   807→    /* Perf 5: Branch optimization — merge conditional jumps over unconditional
   808→     * jumps (inverted branch chains), eliminate dead branches, and compute
   809→     * short-jump offsets. Called after peephole but before emission so the
   810→     * emitter sees the optimized instruction stream. */
   811→    {
   812→        vtx_x86_emit_t tmp_emit;
   813→        vtx_x86_emit_init(&tmp_emit, 256);
   814→        vtx_branch_optimize(inst_stream, &tmp_emit, ra_result);
   815→        vtx_x86_emit_destroy(&tmp_emit);
   816→    }
   817→
   818→    /* Step 5: Build guard descriptor array from the graph.
   819→     * Scan for Guard/DeoptGuard nodes and create descriptors so we can
   820→     * emit deopt stubs and build the side table during code emission. */
   821→    vtx_guard_desc_array_t guard_arr;
   822→    if (vtx_guard_desc_array_init(&guard_arr, arena) != 0) {
   823→        fprintf(stderr, "[pipeline] guard array init failed\n");
   824→        *time_ns = elapsed_ns(start);
   825→        return -1;
   826→    }
   827→
   828→    vtx_node_table_t *ntable = &graph->node_table;
   829→    for (uint32_t i = 0; i < ntable->count; i++) {
   830→        vtx_node_t *node = &ntable->nodes[i];
   831→        if (node->dead) continue;
   832→        if (node->opcode != VTX_OP_Guard && node->opcode != VTX_OP_DeoptGuard) continue;
   833→
   834→        vtx_guard_desc_t gd;
   835→        memset(&gd, 0, sizeof(gd));
   836→        gd.guard_node = i;  /* node index in the table */
   837→        gd.cond = node->cond;
   838→        gd.bytecode_pc = node->bytecode_pc;
   839→        gd.type_id = node->type_id;
   840→        /* frame_state_index: use the node's frame_state if available,
   841→         * otherwise use the node index as a unique identifier */
   842→        gd.frame_state_index = (node->frame_state != VTX_NODEID_INVALID)
   843→                                ? node->frame_state
   844→                                : i;
   845→
   846→        vtx_guard_desc_array_add(&guard_arr, gd, arena);
   847→    }
   848→
   849→    /* Step 6: Build side table for deoptimization.
   850→     * The side table maps native PC offsets → FrameState, enabling
   851→     * the deopt runtime to reconstruct interpreter state when a
   852→     * guard fails or a safepoint is hit. */
   853→    vtx_side_table_t *side_table = vtx_side_table_build(arena);
   854→    if (!side_table) {
   855→        fprintf(stderr, "[pipeline] side table build failed\n");
   856→        *time_ns = elapsed_ns(start);
   857→        return -1;
   858→    }
   859→
   860→    /* Step 7: Code emission */
   861→    vtx_x86_emit_t emitter;
   862→    if (vtx_x86_emit_init(&emitter, VTX_EMIT_INITIAL_CAPACITY) != 0) {
   863→        fprintf(stderr, "[pipeline] emitter init failed\n");
   864→        *time_ns = elapsed_ns(start);
   865→        return -1;
   866→    }
   867→
   868→    /* Emit prologue — includes arg-to-register copy for T2/T3.
   869→     * arg_count and max_locals are needed so the prologue can copy
   870→     * args from the JIT entry's args array into the registers that
   871→     * the T2 instruction selector expects for Parameter nodes. */
   872→    vtx_x86_emit_prologue(&emitter, ra_result->frame_size,
   873→                           ra_result->callee_saved_mask,
   874→                           method_arg_count, method_max_locals,
   875→                           ra_result->is_leaf);
   876→
   877→    /* Note: SMI constants (R10=HEADER, R11=DATA_MASK) are loaded as the
   878→     * first instructions in block 0 of the instruction stream (emitted by
   879→     * isel), NOT here. This way the regalloc sees the definitions and
   880→     * doesn't spill the fixed vregs. */
   881→
   882→    /* Note: Epilogue is emitted by the RET instruction handler in
   883→     * emit_single_inst(), which calls vtx_x86_emit_epilogue() to
   884→     * restore callee-saved registers before the ret instruction.
   885→     * This correctly handles functions with multiple return points.
   886→     * We do NOT emit a separate epilogue at the end of the function. */
   887→
   888→    /* Emit the function body */
   889→    if (vtx_x86_emit_function(&emitter, inst_stream, ra_result, arena) != 0) {
   890→        fprintf(stderr, "[pipeline] code emission failed\n");
   891→        vtx_x86_emit_destroy(&emitter);
   892→        *time_ns = elapsed_ns(start);
   893→        return -1;
   894→    }
   895→
   896→    /* Step 8: Lower guards — record guard positions in the side table.
   897→     * This scans the instruction stream for guard JCCs (marked with
   898→     * VTX_INST_FLAG_IS_GUARD) and records their native PC offsets,
   899→     * live register maps, and FrameState indices in the side table. */
   900→    if (guard_arr.count > 0) {
   901→        int lowered = vtx_guard_emit_lower(&guard_arr, inst_stream,
   902→                                            &emitter, side_table, arena,
   903→                                            ra_result);
   904→        if (lowered < 0) {
   905→            fprintf(stderr, "[pipeline] guard lowering failed\n");
   906→            vtx_x86_emit_destroy(&emitter);
   907→            *time_ns = elapsed_ns(start);
   908→            return -1;
   909→        }
   910→
   911→        /* Step 9: Emit deopt stubs after the main code.
   912→         * Each guard gets a deopt stub that saves state and transfers
   913→         * to the deopt runtime when the guard fails. */
   914→        uint8_t *code_start = (uint8_t *)vtx_x86_emit_code(&emitter);
   915→        int stubs = vtx_guard_emit_deopt_stubs(&guard_arr, &emitter,
   916→                                                 side_table, code_start, arena);
   917→        if (stubs < 0) {
   918→            fprintf(stderr, "[pipeline] deopt stub emission failed\n");
   919→            vtx_x86_emit_destroy(&emitter);
   920→            *time_ns = elapsed_ns(start);
   921→            return -1;
   922→        }
   923→
   924→        /* Step 10: Patch guard JCC instructions to point to deopt stubs.
   925→         * After deopt stubs are emitted, we patch the 32-bit displacement
   926→         * in each guard's JCC instruction to jump to its deopt stub. */
   927→        int patched = vtx_guard_emit_patch(&guard_arr, &emitter,
   928→                                            code_start, arena);
   929→        if (patched < 0) {
   930→            fprintf(stderr, "[pipeline] guard patching failed\n");
   931→            vtx_x86_emit_destroy(&emitter);
   932→            *time_ns = elapsed_ns(start);
   933→            return -1;
   934→        }
   935→
   936→        /* Step 10.5: Populate real FrameState objects in the side table.
   937→         *
   938→         * The side table entries reference frame_state_index, which is
   939→         * the NodeID of the Guard's FrameState node in the IR. We now
   940→         * create real vtx_frame_state_t objects from the IR FrameState
   941→         * nodes and store them in the side table's frame_states array.
   942→         *
   943→         * Without this, the side table has NULL frame_states and any
   944→         * deopt would abort with "can't reconstruct interpreter state."
   945→         *
   946→         * The FrameState IR node has inputs:
   947→         *   [0] = control node
   948→         *   [1] = memory node
   949→         *   [2..2+max_locals] = local variable values (as NodeIDs)
   950→         *
   951→         * We create a vtx_frame_state_t with:
   952→         *   - bytecode_pc from the guard node
   953→         *   - method_id from the method
   954→         *   - locals array from the FrameState node's inputs
   955→         *   - empty stack (T2 doesn't track operand stack at deopt points
   956→         *     yet — the live register map in the side table handles this)
   957→         */
   958→        for (uint32_t g = 0; g < guard_arr.count; g++) {
   959→            vtx_guard_desc_t *guard = &guard_arr.guards[g];
   960→            vtx_node_t *guard_node = &graph->node_table.nodes[guard->guard_node];
   961→
   962→            /* Find the FrameState node for this guard */
   963→            vtx_nodeid_t fs_id = guard_node->frame_state;
   964→            if (fs_id == VTX_NODEID_INVALID || fs_id >= graph->node_table.count) {
   965→                /* No FrameState — create a minimal one with just the PC */
   966→                vtx_frame_state_t *fs = vtx_frame_state_create(arena,
   967→                    guard->bytecode_pc, 0, 0, 0);
   968→                uint32_t fs_idx2 = vtx_side_table_add_frame_state(side_table, fs);
   969→                guard->frame_state_index = fs_idx2;
   970→                continue;
   971→            }
   972→
   973→            vtx_node_t *fs_node = &graph->node_table.nodes[fs_id];
   974→            if (fs_node->dead) {
   975→                vtx_frame_state_t *fs = vtx_frame_state_create(arena,
   976→                    guard->bytecode_pc, 0, 0, 0);
   977→                uint32_t fs_idx2 = vtx_side_table_add_frame_state(side_table, fs);
   978→                guard->frame_state_index = fs_idx2;
   979→                continue;
   980→            }
   981→
   982→            /* Count locals from the FrameState node's inputs.
   983→             * Inputs: [0]=control, [1]=memory, [2..]=locals */
   984→            uint32_t local_count = 0;
   985→            if (fs_node->input_count > 2) {
   986→                local_count = fs_node->input_count - 2;
   987→            }
   988→
   989→            /* Create the FrameState */
   990→            vtx_frame_state_t *fs = vtx_frame_state_create(arena,
   991→                fs_node->bytecode_pc != 0 ? fs_node->bytecode_pc : guard->bytecode_pc,
   992→                0, /* method_id — filled at install time */
   993→                local_count,
   994→                0  /* stack_count — T2 uses live register maps instead */);
   995→
   996→            /* Fill in locals from the FrameState node's inputs */
   997→            for (uint32_t li = 0; li < local_count && li + 2 < fs_node->input_count; li++) {
   998→                vtx_nodeid_t local_node = fs_node->inputs[li + 2];
   999→                vtx_frame_state_set_local(fs, li, local_node);
  1000→            }
  1001→
  1002→            /* Store in the side table */
  1003→            uint32_t fs_idx2 = vtx_side_table_add_frame_state(side_table, fs);
  1004→            guard->frame_state_index = fs_idx2;
  1005→        }
  1006→    }
  1007→
  1008→    /* Copy the emitted code into the result.
  1009→     * We use malloc here (not arena) because the native code must outlive
  1010→     * the compilation arena -- it gets installed in the code cache.
  1011→     *
  1012→     * No hard size reject: the code cache is growable (see emit.c) and
  1013→     * L2/i-cache pressure is a tuning concern for the user, not a
  1014→     * correctness gate. A 2MB hot function is better than a 0-byte one
  1015→     * that fell back to the interpreter forever. */
  1016→    uint32_t code_size = vtx_x86_emit_position(&emitter);
  1017→
  1018→    uint8_t *native_code = (uint8_t *)malloc(code_size);
  1019→    if (!native_code) {
  1020→        fprintf(stderr, "[pipeline] failed to allocate %u bytes for native code\n",
  1021→                code_size);
  1022→        vtx_x86_emit_destroy(&emitter);
  1023→        *time_ns = elapsed_ns(start);
  1024→        return -1;
  1025→    }
  1026→
  1027→    memcpy(native_code, vtx_x86_emit_code(&emitter), code_size);
  1028→
  1029→    /* Propagate the relocation table to the result so vtx_install_method()
  1030→     * can call vtx_reloc_apply_external() at install time. Without this,
  1031→     * every external call recorded during emission — write-barrier calls,
  1032→     * GC helper calls, JIT-to-JIT direct calls — would be silently dropped
  1033→     * (reloc_apply_external is a no-op when given NULL).
  1034→     *
  1035→     * The reloc_table is arena-allocated (entries + struct), so it survives
  1036→     * the emitter's destruction and is owned by the result. */
  1037→    result->reloc_table = emitter.relocs;
  1038→
  1039→    vtx_x86_emit_destroy(&emitter);
  1040→
  1041→    result->native_code = native_code;
  1042→    result->native_size = code_size;
  1043→    result->side_table = side_table;
  1044→
  1045→    *time_ns = elapsed_ns(start);
  1046→    return 0;
  1047→}
  1048→
  1049→/* ========================================================================== */
  1050→/* Main pipeline entry point                                                   */
  1051→/* ========================================================================== */
  1052→
  1053→int vtx_pipeline_run(vtx_graph_t *graph,
  1054→                      const vtx_pipeline_config_t *config,
  1055→                      vtx_arena_t *arena,
  1056→                      vtx_compile_result_t *result)
  1057→{
  1058→    int64_t pipeline_start = now_ns();
  1059→
  1060→    /* Initialize result */
  1061→    memset(result, 0, sizeof(*result));
  1062→    result->graph = graph;
  1063→    result->success = false;
  1064→
  1065→    /* Accumulated stats */
  1066→    vtx_pipeline_stats_t stats;
  1067→    memset(&stats, 0, sizeof(stats));
  1068→
  1069→    /* Temporary accumulators for cleanup passes after inlining */
  1070→    uint32_t post_inline_gvn = 0;
  1071→    uint32_t post_inline_sccp = 0;
  1072→    uint32_t post_inline_dce = 0;
  1073→    uint32_t post_pea_dce = 0;
  1074→    int64_t  post_inline_gvn_ns = 0;
  1075→    int64_t  post_inline_sccp_ns = 0;
  1076→    int64_t  post_inline_dce_ns = 0;
  1077→    int64_t  post_pea_dce_ns = 0;
  1078→
  1079→    /* ================================================================== */
  1080→    /* Phase 1: GVN (Global Value Numbering)                              */
  1081→    /*                                                                    */
  1082→    /* GVN runs first to merge redundant computations. This enables       */
  1083→    /* downstream passes (SCCP, DCE, inlining) to work on a smaller,     */
  1084→    /* cleaner graph.                                                     */
  1085→    /* ================================================================== */
  1086→    if (config->run_gvn) {
  1087→        stats.gvn_nodes_merged = run_gvn_pass(
  1088→            graph, config->gvn_iterations, &stats.gvn_time_ns);
  1089→
  1090→        if (verify_between_passes(graph, config, "GVN") != 0) {
  1091→            result->stats = stats;
  1092→            return -1;
  1093→        }
  1094→    }
  1095→
  1096→    /* ================================================================== */
  1097→    /* Phase 2: SCCP (Sparse Conditional Constant Propagation)            */
  1098→    /*                                                                    */
  1099→    /* Propagates constants and folds conditional branches. This exposes   */
  1100→    /* more dead code for DCE and simplifies the graph for inlining.      */
  1101→    /* ================================================================== */
  1102→    if (config->run_sccp) {
  1103→        stats.sccp_constants_propagated = run_sccp_pass(
  1104→            graph, config->sccp_iterations, &stats.sccp_time_ns);
  1105→
  1106→        if (verify_between_passes(graph, config, "SCCP") != 0) {
  1107→            result->stats = stats;
  1108→            return -1;
  1109→        }
  1110→    }
  1111→
  1112→    /* ================================================================== */
  1113→    /* Phase 2.5: Strength Reduction                                      */
  1114→    /*                                                                    */
  1115→    /* Replaces expensive ops with cheaper equivalents:                   */
  1116→    /*   Mul(x, 2^k) → Shl(x, k)                                         */
  1117→    /*   Mul(x, 1) → x, Mul(x, 0) → 0, Div(x, 1) → x                    */
  1118→    /*                                                                    */
  1119→    /* Runs after SCCP (constants are propagated) and before DCE (dead    */
  1120→    /* nodes from the replacement are cleaned up).                        */
  1121→    /* ================================================================== */
  1122→    if (config->run_sccp) {
  1123→        uint32_t sr_replaced = vtx_strength_reduce_run(graph);
  1124→        if (sr_replaced > 0) {
  1125→            vtx_node_table_clear_dead(&graph->node_table);
  1126→        }
  1127→
  1128→        /* Algebraic simplification: fold x+0, x*1, x*0, x-x, !!x, etc.
  1129→         * Runs after SCCP (constants are known) and strength reduction. */
  1130→        uint32_t alg_simplified = vtx_algebraic_simplify_run(graph);
  1131→        if (alg_simplified > 0) {
  1132→            vtx_node_table_clear_dead(&graph->node_table);
  1133→        }
  1134→    }
  1135→
  1136→    /* ================================================================== */
  1137→    /* Phase 2.6: SMI Tag Elision + Representation Selection              */
  1138→    /*                                                                    */
  1139→    /* Marks arithmetic chains AND Phis as RAW_INT so the isel skips      */
  1140→    /* per-op untag/retag. The representation_selection pass (C++) does   */
  1141→    /* V8-style representation selection: it marks Phis as RAW_INT when   */
  1142→    /* safe, and the boundary conversions are handled by resolve_phis   */
  1143→    /* (INSERT_UNTAG on forward edge, INSERT_RETAG at tagged consumers). */
  1144→    /* This collapses per-iteration tag/untag overhead in hot loops.     */
  1145→    /* ================================================================== */
  1146→    if (config->run_sccp) {
  1147→        /* First run the old smi_tag_elision for backward compat */
  1148→        uint32_t elided = vtx_smi_tag_elision_run(graph);
  1149→        if (elided > 0) {
  1150→            verify_between_passes(graph, config, "SMITagElision");
  1151→        }
  1152→
  1153→        /* Then run the new representation selection pass which also
  1154→         * marks Phis as RAW_INT (the old pass didn't). This is the
  1155→         * V8 Simplified Lowering approach. */
  1156→        /* Representation selection — V8-style.
  1157→         * DISABLED: breaks fib (returns N instead of fib(N)) when Cmp
  1158→         * is allowed as a Phi consumer. The Cmp+If fusion path appears
  1159→         * to handle mixed representations correctly, but there's a
  1160→         * subtle bug. Without Cmp as a consumer, sum gets 12% faster
  1161→         * but the counter Phi stays tagged (only accumulator Phi is raw).
  1162→         * With Cmp, sum gets 39% faster but fib/collatz break.
  1163→         * TODO: debug the Cmp+If fusion with RAW_INT Phi inputs. */
  1164→        /* vtx_representation_selection_run(graph); */
  1165→    }
  1166→
  1167→    /* ================================================================== */
  1168→    /* Phase 3: DCE (Dead Code Elimination)                               */
  1169→    /*                                                                    */
  1170→    /* Removes nodes whose results are never used. GVN and SCCP expose    */
  1171→    /* many dead nodes (e.g., the second copy of a redundant computation, */
  1172→    /* or a branch condition that folded to a constant).                  */
  1173→    /* ================================================================== */
  1174→    if (config->run_dce) {
  1175→        stats.dce_nodes_removed = run_dce_pass(
  1176→            graph, config->dce_iterations, &stats.dce_time_ns, config->run_verify);
  1177→
  1178→        if (verify_between_passes(graph, config, "DCE") != 0) {
  1179→            result->stats = stats;
  1180→            return -1;
  1181→        }
  1182→    }
  1183→
  1184→    /* ================================================================== */
  1185→    /* Phase 2.9: CFG Simplification                                       */
  1186→    /*                                                                    */
  1187→    /* Collapses single-input Regions (from SCCP-folded Ifs), resolves    */
  1188→    /* Goto→Goto chains. Reduces CFG size for better scheduling.          */
  1189→    /* ================================================================== */
  1190→    if (config->run_sccp) {
  1191→        uint32_t cfg_simplified = vtx_cfg_simplify_run(graph);
  1192→        if (cfg_simplified > 0) {
  1193→            vtx_node_table_clear_dead(&graph->node_table);
  1194→        }
  1195→    }
  1196→
  1197→    /* ================================================================== */
  1198→    /* Phase 3.5: Mid-Tier Type Specialization (T1.5)                      */
  1199→    /*                                                                    */
  1200→    /* When run_midtier is true, applies type-specialization based on      */
  1201→    /* type feedback. This inserts guards at monomorphic call sites and    */
  1202→    /* annotates arithmetic nodes with observed types, enabling the        */
  1203→    /* lowering pipeline to skip type checks. This is the core of the     */
  1204→    /* T1.5 mid-tier: type specialization without full optimization.      */
  1205→    /* ================================================================== */
  1206→    if (config->run_midtier && config->type_feedback != NULL) {
  1207→        /* Walk all nodes and specialize based on type feedback.
  1208→         * For CALL_VIRTUAL with monomorphic receiver: insert Guard + direct dispatch.
  1209→         * For Add/Sub/Mul with known-integer inputs: annotate as integer-typed.
  1210→         * For LoadField with monomorphic holder shape: skip shape check. */
  1211→        vtx_node_table_t *ntable = &graph->node_table;
  1212→        const vtx_type_feedback_t *feedback = config->type_feedback;
  1213→
  1214→        for (uint32_t i = 0; i < ntable->count; i++) {
  1215→            vtx_node_t *node = &ntable->nodes[i];
  1216→            if (node->dead) continue;
  1217→
  1218→            /* Specialize CALL_VIRTUAL at monomorphic sites */
  1219→            if (node->opcode == VTX_OP_CallVirtual) {
  1220→                uint32_t site_index = node->bytecode_pc;
  1221→                uint32_t type_count = vtx_type_feedback_get_call_site_type_count(
  1222→                    feedback, site_index);
  1223→
  1224→                if (type_count == 1) {
  1225→                    vtx_typeid_t dominant_type = vtx_type_feedback_get_dominant_call_type(
  1226→                        feedback, site_index);
  1227→
  1228→                    if (dominant_type != VTX_TYPE_INVALID) {
  1229→                        /* Insert a Guard node checking the receiver type */
  1230→                        vtx_nodeid_t guard_id = vtx_node_create(ntable, VTX_OP_Guard);
  1231→                        if (guard_id != VTX_NODEID_INVALID) {
  1232→                            vtx_node_t *guard = vtx_node_get(ntable, guard_id);
  1233→                            if (guard) {
  1234→                                /* Copy the first data input (receiver) */
  1235→                                for (uint32_t inp = 0; inp < node->input_count; inp++) {
  1236→                                    const vtx_node_t *inp_node = vtx_node_get_const(
  1237→                                        ntable, node->inputs[inp]);
  1238→                                    if (inp_node && vtx_nf_has(inp_node->flags, VTX_NF_DATA)) {
  1239→                                        vtx_node_add_input(ntable, guard_id, node->inputs[inp]);
  1240→                                        break;
  1241→                                    }
  1242→                                }
  1243→                                guard->bytecode_pc = node->bytecode_pc;
  1244→                                guard->type_id = dominant_type;
  1245→                                guard->flags = VTX_NF_CONTROL | VTX_NF_DATA | VTX_NF_SIDE_EFFECT | VTX_NF_PINNED;
  1246→                            }
  1247→                        }
  1248→                        /* Annotate the call with the specialized type */
  1249→                        node->type_id = dominant_type;
  1250→                    }
  1251→                }
  1252→                continue;
  1253→            }
  1254→
  1255→            /* Specialize LoadField at monomorphic holder shape sites */
  1256→            if (node->opcode == VTX_OP_LoadField) {
  1257→                uint32_t site_index = node->bytecode_pc;
  1258→                vtx_shapeid_t dominant_shape = vtx_type_feedback_get_dominant_field_shape(
  1259→                    feedback, site_index);
  1260→
  1261→                if (dominant_shape != VTX_SHAPE_INVALID) {
  1262→                    node->type_id = dominant_shape; /* store shape in type_id as proxy */
  1263→                }
  1264→                continue;
  1265→            }
  1266→        }
  1267→
  1268→        if (verify_between_passes(graph, config, "MidTierSpecialization") != 0) {
  1269→            result->stats = stats;
  1270→            return -1;
  1271→        }
  1272→    }
  1273→
  1274→    /* ================================================================== */
  1275→    /* Phase 4: ML-guided Inlining                                        */
  1276→    /*                                                                    */
  1277→    /* Uses a GBDT model to decide which call sites to inline. Inlining   */
  1278→    /* expands the caller graph with the callee's body, enabling further  */
  1279→    /* optimizations across the call boundary.                            */
  1280→    /* ================================================================== */
  1281→    if (config->run_inlining) {
  1282→        uint32_t inline_result = run_inlining_pass(
  1283→            graph, config, arena, &stats.inlining_time_ns);
  1284→        stats.inlining_decisions = inline_result >> 16;
  1285→        stats.inlines_performed = inline_result & 0xFFFF;
  1286→
  1287→        if (verify_between_passes(graph, config, "Inlining") != 0) {
  1288→            result->stats = stats;
  1289→            return -1;
  1290→        }
  1291→
  1292→        /* ============================================================== */
  1293→        /* Phase 5: GVN + SCCP + DCE again (post-inlining cleanup)        */
  1294→        /*                                                                */
  1295→        /* Inlining introduces redundant nodes (parameter mappings,       */
  1296→        /* return Phi merges). Running GVN+SCCP+DCE again cleans up      */
  1297→        /* the expanded graph.                                            */
  1298→        /* ============================================================== */
  1299→        if (config->run_gvn) {
  1300→            post_inline_gvn = run_gvn_pass(
  1301→                graph, config->gvn_iterations, &post_inline_gvn_ns);
  1302→            stats.gvn_nodes_merged += post_inline_gvn;
  1303→            stats.gvn_time_ns += post_inline_gvn_ns;
  1304→
  1305→            if (verify_between_passes(graph, config, "Post-inline GVN") != 0) {
  1306→                result->stats = stats;
  1307→                return -1;
  1308→            }
  1309→        }
  1310→
  1311→        if (config->run_sccp) {
  1312→            post_inline_sccp = run_sccp_pass(
  1313→                graph, config->sccp_iterations, &post_inline_sccp_ns);
  1314→            stats.sccp_constants_propagated += post_inline_sccp;
  1315→            stats.sccp_time_ns += post_inline_sccp_ns;
  1316→
  1317→            if (verify_between_passes(graph, config, "Post-inline SCCP") != 0) {
  1318→                result->stats = stats;
  1319→                return -1;
  1320→            }
  1321→        }
  1322→
  1323→        if (config->run_dce) {
  1324→            post_inline_dce = run_dce_pass(
  1325→                graph, config->dce_iterations, &post_inline_dce_ns,
  1326→                config->run_verify);
  1327→            stats.dce_nodes_removed += post_inline_dce;
  1328→            stats.dce_time_ns += post_inline_dce_ns;
  1329→
  1330→            if (verify_between_passes(graph, config, "Post-inline DCE") != 0) {
  1331→                result->stats = stats;
  1332→                return -1;
  1333→            }
  1334→        }
  1335→    }
  1336→
  1337→    /* ================================================================== */
  1338→    /* Phase 5.5: Partial Virtualization (field → constant)                */
  1339→    /*                                                                    */
  1340→    /* Identifies object fields that always hold compile-time constant     */
  1341→    /* values and replaces field loads with the constants directly.        */
  1342→    /* This is the "cross-constant optimization + partial virtualization"  */
  1343→    /* combination: known constants propagate through the computation      */
  1344→    /* graph, killing branches and enabling further virtualization.        */
  1345→    /*                                                                    */
  1346→    /* Runs BEFORE PEA (Phase 6) so that PEA sees the simplified graph     */
  1347→    /* (objects with constant fields are more likely to become fully       */
  1348→    /* virtualizable). Also runs BEFORE SCCP's re-run so SCCP can         */
  1349→    /* propagate the exposed constants.                                   */
  1350→    /*                                                                    */
  1351→    /* T2-safe: no speculation — only replaces fields proven to always     */
  1352→    /* hold the same constant value. A T3 variant (future) can use type    */
  1353→    /* feedback to speculate field values with guards.                    */
  1354→    /* ================================================================== */
  1355→    if (config->run_partial_virt) {
  1356→        vtx_partial_virtualize_run(graph);
  1357→    }
  1358→
  1359→    /* ================================================================== */
  1360→    /* Phase 6: PEA (Partial Escape Analysis)                             */
  1361→    /*                                                                    */
  1362→    /* Determines which allocations can be scalar-replaced (eliminated).  */
  1363→    /* Cross-object SR extends this to objects reachable only through     */
  1364→    /* fields of other (escaping) objects. Materialization reifies        */
  1365→    /* scalar-replaced objects at escape/deopt points.                    */
  1366→    /* ================================================================== */
  1367→    if (config->run_pea) {
  1368→        stats.pea_allocs_eliminated = run_pea_pass(
  1369→            graph, arena, &stats.pea_time_ns);
  1370→
  1371→        if (verify_between_passes(graph, config, "PEA") != 0) {
  1372→            result->stats = stats;
  1373→            return -1;
  1374→        }
  1375→
  1376→        /* DCE after scalar replacement: dead allocations and field     */
  1377→        /* accesses become removable after PEA rewrites them.           */
  1378→        if (config->run_dce) {
  1379→            post_pea_dce = run_dce_pass(
  1380→                graph, config->dce_iterations, &post_pea_dce_ns,
  1381→                config->run_verify);
  1382→            stats.dce_nodes_removed += post_pea_dce;
  1383→            stats.dce_time_ns += post_pea_dce_ns;
  1384→
  1385→            if (verify_between_passes(graph, config, "Post-PEA DCE") != 0) {
  1386→                result->stats = stats;
  1387→                return -1;
  1388→            }
  1389→        }
  1390→    }
  1391→
  1392→    /* ================================================================== */
  1393→    /* Phase 6.5: Temporal Constant Propagation                            */
  1394→    /*                                                                    */
  1395→    /* Identifies "phase-stable" fields: values that are mutable during   */
  1396→    /* startup but become constant after initialization completes.        */
  1397→    /* ================================================================== */
  1398→    if (config->run_temporal_const) {
  1399→        vtx_temporal_constant_run(graph);
  1400→    }
  1401→
  1402→    /* ================================================================== */
  1403→    /* Phase 6.6: Cross-Function Virtual Object Continuation               */
  1404→    /*                                                                    */
  1405→    /* Pushes PEA across call boundaries. After inlining, identifies        */
  1406→    /* virtual objects that flow through the (now-inlined) call chain     */
  1407→    /* and maintains their virtual representation. Fully virtualizes       */
  1408→    /* objects where ALL fields are constant (removes the Allocate).       */
  1409→    /* ================================================================== */
  1410→    if (config->run_cross_func_virt) {
  1411→        vtx_cross_function_virtual_run(graph);
  1412→    }
  1413→
  1414→    /* ================================================================== */
  1415→    /* Phase 7: Scheduling (SoN -> Basic Blocks)                           */
  1416→    /*                                                                    */
  1417→    /* Converts the Sea-of-Nodes graph back into a scheduled form with    */
  1418→    /* basic blocks. This is required for LICM (needs loop structure),    */
  1419→    /* bounds check elimination (needs dominance), and lowering.          */
  1420→    /* ================================================================== */
  1421→    vtx_schedule_t schedule;
  1422→    memset(&schedule, 0, sizeof(schedule));
  1423→
  1424→    if (run_schedule_pass(graph, arena, &schedule, &stats.schedule_time_ns) != 0) {
  1425→        fprintf(stderr, "[pipeline] scheduling failed\n");
  1426→        result->stats = stats;
  1427→        return -1;
  1428→    }
  1429→
  1430→    /* (Debug schedule printing disabled for production) */
  1431→
  1432→    if (verify_between_passes(graph, config, "Schedule") != 0) {
  1433→        vtx_schedule_destroy(&schedule);
  1434→        result->stats = stats;
  1435→        return -1;
  1436→    }
  1437→
  1438→    result->schedule = (vtx_schedule_t *)vtx_arena_alloc(arena, sizeof(vtx_schedule_t));
  1439→    if (result->schedule) {
  1440→        memcpy(result->schedule, &schedule, sizeof(vtx_schedule_t));
  1441→    }
  1442→
  1443→    /* ================================================================== */
  1444→    /* Phase 8: LICM (Loop-Invariant Code Motion)                         */
  1445→    /*                                                                    */
  1446→    /* Hoists loop-invariant computations out of loops into preheaders.   */
  1447→    /* Requires the schedule for loop structure identification.           */
  1448→    /* ================================================================== */
  1449→    if (config->run_licm) {
  1450→        stats.licm_nodes_hoisted = run_licm_pass(
  1451→            graph, &schedule, arena, &stats.licm_time_ns);
  1452→
  1453→        if (verify_between_passes(graph, config, "LICM") != 0) {
  1454→            result->stats = stats;
  1455→            return -1;
  1456→        }
  1457→
  1458→        /* After LICM hoists nodes, the schedule may be stale.
  1459→         * Re-schedule for correctness in subsequent passes. */
  1460→        vtx_schedule_destroy(&schedule);
  1461→        memset(&schedule, 0, sizeof(schedule));
  1462→        if (vtx_schedule_run(graph, arena, &schedule) != 0) {
  1463→            fprintf(stderr, "[pipeline] re-scheduling after LICM failed\n");
  1464→            result->stats = stats;
  1465→            return -1;
  1466→        }
  1467→
  1468→        if (result->schedule) {
  1469→            memcpy(result->schedule, &schedule, sizeof(vtx_schedule_t));
  1470→        }
  1471→
  1472→        /* Outer fixpoint: re-run GVN+SCCP+DCE after LICM to clean up
  1473→         * trivial Phis created by hoisting (Phi with both inputs now equal)
  1474→         * and dead code exposed by the hoisted invariants.
  1475→         * (audit #9: outer fixpoint after LICM) */
  1476→        if (config->run_gvn) {
  1477→            run_gvn_pass(graph, 1, &stats.gvn_time_ns);
  1478→        }
  1479→        if (config->run_sccp) {
  1480→            run_sccp_pass(graph, 1, &stats.sccp_time_ns);
  1481→        }
  1482→        /* BUGFIX (audit #22): Re-run strength reduction after LICM.
  1483→         * LICM may hoist Mul(x, const) out of loops, creating new
  1484→         * opportunities for strength reduction (e.g., Mul(x, 8) →
  1485→         * Shl(x, 3) outside the loop). The old code skipped this,
  1486→         * leaving hoisted Muls as IMULs.
  1487→         *
  1488→         * Safe because strength_reduce checks if the node is already
  1489→         * a shift — it won't create duplicate Sar chains. */
  1490→        if (config->run_sccp) {
  1491→            uint32_t sr_replaced2 = vtx_strength_reduce_run(graph);
  1492→            if (sr_replaced2 > 0) {
  1493→                vtx_node_table_clear_dead(&graph->node_table);
  1494→            }
  1495→        }
  1496→        if (config->run_dce) {
  1497→            run_dce_pass(graph, 1, &stats.dce_time_ns, false);
  1498→        }
  1499→
  1500→        /* Guard hoisting: move loop-invariant guards (type checks, null
  1501→         * checks) to the preheader so they execute once instead of every
  1502→         * iteration.
  1503→         * (audit #4: wire vtx_hoist_guards) */
  1504→        vtx_hoist_guards(graph, &schedule, arena);
  1505→
  1506→        /* Guard merging: merge adjacent guards on the same value into a
  1507→         * single guard. E.g., two null checks on the same reference become
  1508→         * one null check.
  1509→         * (audit #13: wire vtx_merge_guards) */
  1510→        vtx_merge_guards(graph, arena);
  1511→
  1512→        /* Loop unrolling: unroll small loops by factor 4 to reduce loop
  1513→         * overhead and enable more instruction-level parallelism.
  1514→         * BUGFIX (audit #8): Was factor=2 (too conservative). Factor 4
  1515→         * gives better ILP exposure without excessive code size growth.
  1516→         * The unroll pass now also handles multiple loops (was aborting
  1517→         * after the first). */
  1518→        vtx_loop_unroll_run(graph, &schedule, arena, 4);
  1519→    }
  1520→
  1521→    /* ================================================================== */
  1522→    /* Phase 8.5: Representation Inference                                */
  1523→    /*                                                                    */
  1524→    /* Inserts explicit UnboxInt/BoxInt nodes around arithmetic chains     */
  1525→    /* to eliminate per-op SMI tag/untag overhead. Runs after LICM and    */
  1526→    /* loop unrolling (so the full loop body is visible) and before        */
  1527→    /* scheduling (so the new nodes are placed correctly).                 */
  1528→    /* ================================================================== */
  1529→    if (config->run_rep_infer) {
  1530→        int64_t ri_start = now_ns();
  1531→        uint32_t ri_inserted = vtx_rep_infer_run(graph, arena);
  1532→        stats.block_layout_time_ns += elapsed_ns(ri_start); /* reuse stats field */
  1533→
  1534→        /* Run DCE only if rep_infer actually inserted nodes */
  1535→        if (ri_inserted > 0 && config->run_dce) {
  1536→            run_dce_pass(graph, 1, &stats.dce_time_ns, false);
  1537→        }
  1538→
  1539→        /* Only re-schedule if rep_infer actually inserted new nodes.
  1540→         * If no nodes were inserted (e.g., SCCP folded everything),
  1541→         * the old schedule is still valid and re-scheduling risks
  1542→         * introducing ordering bugs. */
  1543→        if (ri_inserted > 0) {
  1544→            vtx_schedule_destroy(&schedule);
  1545→            memset(&schedule, 0, sizeof(schedule));
  1546→            if (vtx_schedule_run(graph, arena, &schedule) != 0) {
  1547→                fprintf(stderr, "[pipeline] re-scheduling after rep_infer failed\n");
  1548→                result->stats = stats;
  1549→                return -1;
  1550→            }
  1551→            if (result->schedule) {
  1552→                memcpy(result->schedule, &schedule, sizeof(vtx_schedule_t));
  1553→            }
  1554→        }
  1555→
  1556→        if (verify_between_passes(graph, config, "RepInfer") != 0) {
  1557→            result->stats = stats;
  1558→            return -1;
  1559→        }
  1560→    }
  1561→
  1562→    /* ================================================================== */
  1563→    /* Phase 9: Bounds Check Elimination                                  */
  1564→    /*                                                                    */
  1565→    /* Eliminates redundant array bounds checks using range analysis.     */
  1566→    /* Requires the schedule for dominance and loop structure info.       */
  1567→    /* ================================================================== */
  1568→    if (config->run_bounds_check) {
  1569→        stats.bounds_checks_eliminated = run_bounds_check_pass(
  1570→            graph, &schedule, arena, &stats.bounds_check_time_ns);
  1571→
  1572→        if (verify_between_passes(graph, config, "BoundsCheck") != 0) {
  1573→            result->stats = stats;
  1574→            return -1;
  1575→        }
  1576→
  1577→        /* Outer fixpoint: re-run DCE after bounds check to remove dead
  1578→         * guards and Cmp nodes whose conditions were proven.
  1579→         * (audit #9: outer fixpoint after bounds check) */
  1580→        if (config->run_dce) {
  1581→            run_dce_pass(graph, 1, &stats.dce_time_ns, false);
  1582→        }
  1583→    }
  1584→
  1585→    /* ================================================================== */
  1586→    /* Phase 9.1: Loop Specialization / Vectorization                      */
  1587→    /*                                                                    */
  1588→    /* Analyzes loops for vectorization opportunities. Vectorizable loops  */
  1589→    /* are transformed by replacing scalar LoadIndexed/StoreIndexed with   */
  1590→    /* VectorLoad/VectorStore, and scalar Add/Mul with VectorAdd/VectorMul.*/
  1591→    /* DeoptGuard nodes guard the vectorized code. Runs after bounds check */
  1592→    /* elimination so that eliminated bounds checks don't inhibit          */
  1593→    /* vectorization, and before speculative guards to keep guard          */
  1594→    /* insertion simple.                                                  */
  1595→    /* ================================================================== */
  1596→    if (config->run_loop_spec && config->run_vectorize) {
  1597→        stats.loops_vectorized = run_loop_spec_pass(
  1598→            graph, arena, &stats.loop_spec_time_ns);
  1599→
  1600→        if (verify_between_passes(graph, config, "LoopSpec") != 0) {
  1601→            result->stats = stats;
  1602→            return -1;
  1603→        }
  1604→    }
  1605→
  1606→    /* ================================================================== */
  1607→    /* Phase 9.5: Speculative Guard Insertion (T3 only)                   */
  1608→    /*                                                                    */
  1609→    /* Inserts type-check guards at virtual call sites based on profiling  */
  1610→    /* data. If a dominant type is observed, a guard checks the receiver  */
  1611→    /* type and deoptimizes on mismatch. This enables speculative          */
  1612→    /* devirtualization and more aggressive inlining in the T3 tier.       */
  1613→    /* ================================================================== */
  1614→    if (config->run_speculative) {
  1615→        /* Scan for virtual call sites and insert type guards.
  1616→         * For each CallVirtual/CallInterface, we look up the dominant
  1617→         * receiver type from the type feedback system (if available),
  1618→         * or fall back to the IR node's type_id as a speculative hint.
  1619→         * A DeoptGuard is inserted that checks the receiver's type_id
  1620→         * against the speculated dominant type. On type mismatch,
  1621→         * execution falls back to the interpreter. */
  1622→        vtx_node_table_t *ntable = &graph->node_table;
  1623→        const vtx_type_feedback_t *feedback = config->type_feedback;
  1624→
  1625→        for (uint32_t i = 0; i < ntable->count; i++) {
  1626→            vtx_node_t *node = &ntable->nodes[i];
  1627→            if (node->dead) continue;
  1628→            if (node->opcode != VTX_OP_CallVirtual &&
  1629→                node->opcode != VTX_OP_CallInterface) continue;
  1630→            if (node->input_count < 1) continue;
  1631→
  1632→            /* The receiver is the first data input (after control+memory). */
  1633→            vtx_nodeid_t receiver_id = VTX_NODEID_INVALID;
  1634→            for (uint32_t inp = 0; inp < node->input_count; inp++) {
  1635→                const vtx_node_t *inp_node = vtx_node_get_const(ntable, node->inputs[inp]);
  1636→                if (inp_node && vtx_nf_has(inp_node->flags, VTX_NF_DATA)) {
  1637→                    receiver_id = node->inputs[inp];
  1638→                    break;
  1639→                }
  1640→            }
  1641→
  1642→            if (receiver_id == VTX_NODEID_INVALID) continue;
  1643→
  1644→            /* Determine the speculated dominant receiver type.
  1645→             *
  1646→             * Priority:
  1647→             *   1. If type feedback is available, look up the call site's
  1648→             *      bytecode_pc in the feedback and use the dominant observed
  1649→             *      type from profiling data. This is far more accurate than
  1650→             *      the IR node's type_id, which may be a generic base type.
  1651→             *   2. If no feedback is available (feedback pointer is NULL
  1652→             *      or VTX_TYPE_INVALID returned for the site), fall back
  1653→             *      to the node's type_id as a speculative hint. */
  1654→            vtx_typeid_t speculated_type = VTX_TYPE_INVALID;
  1655→
  1656→            if (feedback != NULL) {
  1657→                /* Look up the dominant receiver type from profiling data.
  1658→                 * The site_index is the node's bytecode_pc — the same
  1659→                 * index used when recording observations in the interpreter
  1660→                 * dispatch loop (see dispatch.c: vtx_type_feedback_record_call). */
  1661→                speculated_type = vtx_type_feedback_get_dominant_call_type(
  1662→                    feedback, node->bytecode_pc);
  1663→            }
  1664→
  1665→            /* Fall back to node->type_id if no feedback data is available */
  1666→            if (speculated_type == VTX_TYPE_INVALID) {
  1667→                speculated_type = node->type_id;
  1668→            }
  1669→
  1670→            /* Only insert a guard if we have a valid speculated type */
  1671→            if (speculated_type != VTX_TYPE_INVALID) {
  1672→                /* Insert a DeoptGuard before this call:
  1673→                 * Guard(receiver_type == speculated_type)
  1674→                 * On failure, deoptimize to interpreter. */
  1675→                vtx_nodeid_t guard_id = vtx_node_create(ntable, VTX_OP_DeoptGuard);
  1676→                if (guard_id != VTX_NODEID_INVALID) {
  1677→                    vtx_node_t *guard = vtx_node_get(ntable, guard_id);
  1678→                    if (guard) {
  1679→                        guard->cond = VTX_COND_EQ;
  1680→                        guard->type_id = speculated_type;
  1681→                        guard->flags = VTX_NF_CONTROL | VTX_NF_DATA | VTX_NF_SIDE_EFFECT | VTX_NF_PINNED;
  1682→                        guard->bytecode_pc = node->bytecode_pc;
  1683→                        /* Add the receiver as input */
  1684→                        vtx_node_add_input(ntable, guard_id, receiver_id);
  1685→                        /* Mark this as a speculative guard for T3 */
  1686→                        guard->frame_state = node->frame_state;
  1687→                    }
  1688→                }
  1689→            }
  1690→        }
  1691→
  1692→        if (verify_between_passes(graph, config, "SpeculativeGuards") != 0) {
  1693→            result->stats = stats;
  1694→            return -1;
  1695→        }
  1696→    }
  1697→
  1698→    /* ================================================================== */
  1699→    /* Phase 9.8: Profile-Guided Block Layout                              */
  1700→    /*                                                                    */
  1701→    /* Reorders basic blocks so that hot paths fall through (no branch    */
  1702→    /* needed) and cold paths are pushed to the end. Uses branch          */
  1703→    /* probability data from the profiler to decide which successor is    */
  1704→    /* "hot". This improves I-cache locality for hot code paths.          */
  1705→    /*                                                                    */
  1706→    /* Runs after all IR passes (which may add/remove/reorder blocks)     */
  1707→    /* and before lowering (which emits code in block-index order).       */
  1708→    /* ================================================================== */
  1709→    if (config->run_block_layout) {
  1710→        int64_t layout_start = now_ns();
  1711→        uint32_t blocks_reordered = vtx_block_layout_run(
  1712→            &schedule, graph,
  1713→            config->profiler,  /* may be NULL — falls back to heuristic */
  1714→            config->method);
  1715→        stats.block_layout_time_ns = elapsed_ns(layout_start);
  1716→
  1717→        if (result->schedule) {
  1718→            memcpy(result->schedule, &schedule, sizeof(vtx_schedule_t));
  1719→        }
  1720→
  1721→        if (verify_between_passes(graph, config, "BlockLayout") != 0) {
  1722→            result->stats = stats;
  1723→            return -1;
  1724→        }
  1725→        (void)blocks_reordered;  /* could log for debugging */
  1726→    }
  1727→
  1728→    /* ================================================================== */
  1729→    /* Phase 9.8: Post-optimization PEA re-run                            */
  1730→    /*                                                                    */
  1731→    /* §3.2/PEA audit: PEA should be re-run after inlining and loop       */
  1732→    /* unrolling because these transforms expose new allocation patterns.  */
  1733→    /* Inlining can merge allocation sites from callees into the caller,  */
  1734→    /* and loop unrolling can duplicate allocations that are now eligible */
  1735→    /* for scalar replacement.                                            */
  1736→    /*                                                                    */
  1737→    /* GraalVM runs PEA after inlining (EscapeAnalysisPhase runs after    */
  1738→    /* InliningPhase). V8 runs escape analysis after inlining and         */
  1739→    /* simplification (EscapeAnalysis runs after JSInlining).             */
  1740→    /*                                                                    */
  1741→    /* We only re-run if the first PEA pass found at least one            */
  1742→    /* scalar-replaceable allocation (no point re-running if there are   */
  1743→    /* no allocations to analyze).                                        */
  1744→    /* ================================================================== */
  1745→    if (config->run_pea && stats.pea_allocs_eliminated > 0) {
  1746→        int64_t pea2_start = now_ns();
  1747→        uint32_t pea2_eliminated = run_pea_pass(
  1748→            graph, arena, &stats.pea_time_ns);
  1749→        stats.pea_allocs_eliminated += pea2_eliminated;
  1750→        stats.pea_time_ns += elapsed_ns(pea2_start);
  1751→
  1752→        if (config->run_dce) {
  1753→            run_dce_pass(graph, 1, &stats.dce_time_ns, false);
  1754→        }
  1755→    }
  1756→
  1757→    /* ================================================================== */
  1758→    /* Phase 10: Lowering (isel -> regalloc -> emit)                        */
  1759→    /*                                                                    */
  1760→    /* Converts the optimized SoN graph into x86-64 machine code:         */
  1761→    /*   - Instruction selection maps SoN nodes to x86-64 instructions    */
  1762→    /*   - Register allocation assigns physical registers                 */
  1763→    /*   - Code emission produces the final machine code bytes            */
  1764→    /* ================================================================== */
  1765→    uint32_t m_arg_count = (config->method != NULL) ? config->method->arg_count : 0;
  1766→    uint32_t m_max_locals = (config->method != NULL && config->method->bytecode != NULL)
  1767→                             ? config->method->bytecode->max_locals : 0;
  1768→    if (run_lowering_pipeline(graph, &schedule, arena, result,
  1769→                               &stats.lowering_time_ns,
  1770→                               m_arg_count, m_max_locals) != 0) {
  1771→        fprintf(stderr, "[pipeline] lowering failed\n");
  1772→        result->stats = stats;
  1773→        return -1;
  1774→    }
  1775→
  1776→    /* ================================================================== */
  1777→    /* Finalize                                                           */
  1778→    /* ================================================================== */
  1779→    stats.total_pipeline_time_ns = elapsed_ns(pipeline_start);
  1780→    result->stats = stats;
  1781→    result->success = true;
  1782→
  1783→    /* ================================================================== */
  1784→    /* Code Installation                                                  */
  1785→    /*                                                                    */
  1786→    /* CRITICAL FIX: Previously, the T2/T3 pipeline produced native code  */
  1787→    /* but never installed it into the code cache. The code was simply     */
  1788→    /* freed in vtx_compile_result_destroy(), making the optimizing JIT    */
  1789→    /* completely non-functional. Now, if config provides code_cache,     */
  1790→    /* method_registry, and method, we install the compiled code so it    */
  1791→    /* becomes the active execution path for subsequent calls.            */
  1792→    /* ================================================================== */
  1793→    if (result->native_code != NULL && result->native_size > 0 &&
  1794→        config->code_cache != NULL && config->method_registry != NULL &&
  1795→        config->method != NULL) {
  1796→        uint32_t method_id = config->method->vtable_index;
  1797→        bool installed = vtx_install_method(
  1798→            config->code_cache, config->method_registry,
  1799→            config->method, method_id,
  1800→            result->native_code, result->native_size,
  1801→            result->side_table,  /* side_table from lowering pipeline */
  1802→            result->reloc_table, /* reloc_table — fixes #19: external calls now applied */
  1803→            NULL, 0, NULL, 0,  /* dep_type_ids, dep_shape_ids */
  1804→            config->install_arena ? config->install_arena : arena,
  1805→            NULL, 0,  /* poly_ics */
  1806→            NULL,     /* frame_layout — T2/T3 doesn't use T1's layout */
  1807→            NULL, 0   /* bc_pc_map — T2/T3 uses side_table for deopt, not bc_pc_map */
  1808→        );
  1809→        if (installed) {
  1810→            /* Code is now installed in the cache and method->compiled_code
  1811→             * is set. Transfer ownership: don't free native_code in
  1812→             * vtx_compile_result_destroy(). */
  1813→            result->native_code = NULL;
  1814→            result->native_size = 0;
  1815→            result->side_table = NULL;  /* ownership transferred to code cache */
  1816→            result->reloc_table = NULL; /* ownership transferred to code cache */
  1817→
  1818→            /* Notify the orchestrator that a compilation finished.
  1819→             * This wakes up the recomp monitor (saves a profile snapshot
  1820→             * for drift detection), FDI (records the new version), and
  1821→             * the phase-reactive version manager. Without this call,
  1822→             * those ~5 SOTA subsystems sit idle — they were implemented
  1823→             * but never received the events they need to act. */
  1824→            if (config->orchestrator != NULL) {
  1825→                vtx_orchestrator_on_compile_done(config->orchestrator,
  1826→                                                  method_id,
  1827→                                                  method_id /* version_id */);
  1828→            }
  1829→
  1830→            /* Register the new version with the versioned code cache.
  1831→             * This enables N+1 versioning: the old version (if any) is
  1832→             * marked retired but kept alive until no thread references it.
  1833→             * This prevents use-after-free when thread A is executing old
  1834→             * code while thread B installs a new version. */
  1835→            if (config->versioned_cache != NULL) {
  1836→                vtx_versioned_cache_install(config->versioned_cache,
  1837→                                             method_id,
  1838→                                             config->method->compiled_code,
  1839→                                             result->native_size);
  1840→            }
  1841→
  1842→            /* Create a deoptless continuation table for this method.
  1843→             * The deopt handler uses this to look up pre-compiled
  1844→             * continuations when a guard fails, avoiding the expensive
  1845→             * deopt→interpret→recompile cycle. The table is stored in
  1846→             * the deoptless_tables array indexed by method_id. */
  1847→            if (config->deoptless_tables != NULL &&
  1848→                method_id < config->deoptless_table_count) {
  1849→                if (config->deoptless_tables[method_id] == NULL) {
  1850→                    /* Allocate and initialize the table */
  1851→                    vtx_deoptless_table_t *dtable =
  1852→                        (vtx_deoptless_table_t *)malloc(sizeof(vtx_deoptless_table_t));
  1853→                    if (dtable != NULL) {
  1854→                        if (vtx_deoptless_table_init(dtable, method_id,
  1855→                                                       config->method->compiled_code,
  1856→                                                       graph) == 0) {
  1857→                            config->deoptless_tables[method_id] = dtable;
  1858→                        } else {
  1859→                            free(dtable);
  1860→                        }
  1861→                    }
  1862→                }
  1863→            }
  1864→        }
  1865→    }
  1866→
  1867→    /* ================================================================== */
  1868→    /* Phase 11: Markov Chain — Predictive Compilation (D10)               */
  1869→    /*                                                                    */
  1870→    /* If a Markov chain model is attached to the pipeline config, we     */
  1871→    /* check whether a phase transition is predicted. If so, we predict   */
  1872→    /* which methods will be hot in the next phase and can proactively    */
  1873→    /* compile them before the transition occurs, reducing warmup time.   */
  1874→    /*                                                                    */
  1875→    /* The actual proactive compilation is initiated asynchronously by    */
  1876→    /* recording the predicted hot methods. The compilation thread pool   */
  1877→    /* picks them up and compiles them at lower priority.                 */
  1878→    /* ================================================================== */
  1879→    if (config->markov != NULL && config->markov->is_trained) {
  1880→        uint32_t predicted_phase = vtx_markov_predict_next(
  1881→            config->markov, config->markov->current_phase);
  1882→
  1883→        if (predicted_phase != config->markov->current_phase) {
  1884→            /* A phase transition is predicted. Get the methods that will
  1885→             * be hot in the next phase. */
  1886→            uint32_t hot_methods[32];
  1887→            uint32_t hot_count = vtx_markov_predict_hot_methods(
  1888→                config->markov, predicted_phase, hot_methods, 32);
  1889→
  1890→            /* Log the prediction for debugging / monitoring.
  1891→             * In a production system, these method IDs would be enqueued
  1892→             * on the compilation thread pool for proactive compilation. */
  1893→            if (hot_count > 0) {
  1894→                (void)hot_methods;  /* Suppress unused variable warning;
  1895→                                     * in production, these would be enqueued
  1896→                                     * for proactive compilation. */
  1897→            }
  1898→        }
  1899→    }
  1900→
  1901→    return 0;
  1902→}
  1903→
  1904→/* ========================================================================== */
  1905→/* Compile result destruction                                                  */
  1906→/* ========================================================================== */
  1907→
  1908→void vtx_compile_result_destroy(vtx_compile_result_t *result)
  1909→{
  1910→    if (!result) return;
  1911→
  1912→    /* Free the native code buffer (allocated with malloc, not arena) */
  1913→    if (result->native_code) {
  1914→        free(result->native_code);
  1915→        result->native_code = NULL;
  1916→    }
  1917→
  1918→    /* Note: graph and schedule are NOT owned by the result.
  1919→     * - graph is owned by the caller
  1920→     * - schedule is arena-allocated and freed with the arena
  1921→     * We just clear the pointers to avoid dangling references. */
  1922→    result->graph = NULL;
  1923→    result->schedule = NULL;
  1924→    result->native_size = 0;
  1925→    result->success = false;
  1926→
  1927→    /* reloc_table and side_table are arena-allocated; ownership is
  1928→     * transferred to the code cache on successful install (set to NULL).
  1929→     * If install never happened, the arena reclaims the memory. We just
  1930→     * clear the pointer to avoid stale references. */
  1931→    result->reloc_table = NULL;
  1932→    result->side_table = NULL;
  1933→}

  1935→/* ========================================================================== */
  1936→/* Pipeline config lifecycle                                                    */
  1937→/* ========================================================================== */
  1938→
  1939→int vtx_pipeline_config_init_shared_model(vtx_pipeline_config_t *config)
  1940→{
  1941→    if (!config) return -1;
  1942→
  1943→    /* Only init if inlining is enabled and no shared model exists yet */
  1944→    if (!config->run_inlining) return 0;
  1945→    if (config->shared_gbdt_model != NULL) return 0;  /* already initialized */
  1946→
  1947→    vtx_gbdt_model_t *model = (vtx_gbdt_model_t *)malloc(sizeof(vtx_gbdt_model_t));
  1948→    if (!model) return -1;
  1949→
  1950→    if (vtx_gbdt_model_init(model) != 0) {
  1951→        free(model);
  1952→        return -1;
  1953→    }
  1954→
  1955→    if (vtx_gbdt_load_default_model(model) != 0) {
  1956→        vtx_gbdt_model_destroy(model);
  1957→        free(model);
  1958→        return -1;
  1959→    }
  1960→
  1961→    config->shared_gbdt_model = model;
  1962→    config->owns_gbdt_model = true;
  1963→    return 0;
  1964→}
  1965→
  1966→void vtx_pipeline_config_destroy(vtx_pipeline_config_t *config)
  1967→{
  1968→    if (!config) return;
  1969→
  1970→    if (config->owns_gbdt_model && config->shared_gbdt_model != NULL) {
  1971→        vtx_gbdt_model_destroy(config->shared_gbdt_model);
  1972→        free(config->shared_gbdt_model);
  1973→        config->shared_gbdt_model = NULL;
  1974→        config->owns_gbdt_model = false;
  1975→    }
  1976→}

  1978→/* ========================================================================== */
  1979→/* Heat-adapted speculation budgets (Proposal #12)                               */
  1980→/* ========================================================================== */
  1981→
  1982→vtx_pipeline_config_t vtx_pipeline_config_heat_adapted(uint64_t execution_count)
  1983→{
  1984→    /* Start with the T3 config as the base */
  1985→    vtx_pipeline_config_t config = vtx_pipeline_config_t3();
  1986→
  1987→    /* For very cold methods, use conservative defaults */
  1988→    if (execution_count < 1000) {
  1989→        return config;
  1990→    }
  1991→
  1992→    /* Compute logarithmic scaling factor.
  1993→     * log10(1000) = 3, log10(10M) = 7
  1994→     * We use this to gradually increase aggressiveness. */
  1995→    double log_exec = 0.0;
  1996→    if (execution_count > 0) {
