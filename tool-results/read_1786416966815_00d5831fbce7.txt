    26→    26→/**
    27→    27→ * The register map is a sparse array indexed by NodeID. For nodes that
    28→    28→ * are not in the register map (e.g., constants, nodes that were spilled
    29→    29→ * to the stack), we need to look up the value differently.
    30→    30→ *
    31→    31→ * For OSR down, the register_map is laid out as:
    32→    32→ *   register_map[0] = number of entries
    33→    33→ *   register_map[1..2*N] = (node_id, value) pairs
    34→    34→ *
    35→    35→ * This allows for sparse mapping without allocating a huge array.
    36→    36→ */
    37→    37→typedef struct {
    38→    38→    vtx_nodeid_t node_id;
    39→    39→    vtx_value_t  value;
    40→    40→} vtx_node_value_pair_t;
    41→    41→
    42→    42→vtx_value_t vtx_osr_resolve_node(vtx_nodeid_t node_id,
    43→    43→                                   const vtx_value_t *register_map,
    44→    44→                                   uint32_t map_size)
    45→    45→{
    46→    46→    /* DEOPT-004 fix: the old code did
    47→    47→     *   const vtx_node_value_pair_t *pairs = (const vtx_node_value_pair_t *)register_map;
    48→    48→     * which type-puns vtx_value_t[] as vtx_node_value_pair_t[] — strict
    49→    49→     * aliasing UB. Fix: read each pair via memcpy so the compiler can't
    50→    50→     * mis-optimize. The map layout is:
    51→    51→     *   [count: uint32_t in first vtx_value_t's low 32 bits]
    52→    52→     *   [pair[0].node_id, pair[0].value, pair[1].node_id, pair[1].value, ...]
    53→    53→     * Each element is sizeof(vtx_value_t) bytes. The first element's low
    54→    54→     * 32 bits are the count; the remaining elements form (node_id, value)
    55→    55→     * pairs of 2 vtx_value_t each. */
    56→    56→    if (!register_map || map_size == 0) return VTX_VALUE_UNDEFINED;
    57→    57→
    58→    58→    /* Validate we have at least the count header + 1 pair. */
    59→    59→    if (map_size < 3) return VTX_VALUE_UNDEFINED;
    60→    60→
    61→    61→    /* Read the count via memcpy (no type punning). */
    62→    62→    uint32_t pair_count;
    63→    63→    vtx_value_t first_elem = register_map[0];
    64→    64→    memcpy(&pair_count, &first_elem, sizeof(uint32_t));
    65→    65→
    66→    66→    /* Validate pair_count against map_size (1 header + 2*pair_count). */
    67→    67→    if ((uint64_t)1 + (uint64_t)pair_count * 2ULL > (uint64_t)map_size) {
    68→    68→        return VTX_VALUE_UNDEFINED;
    69→    69→    }
    70→    70→
    71→    71→    /* Linear scan via memcpy — no type punning. */
    72→    72→    for (uint32_t i = 0; i < pair_count; i++) {
    73→    73→        vtx_value_t id_elem = register_map[1 + i * 2];
    74→    74→        vtx_value_t val_elem = register_map[1 + i * 2 + 1];
    75→    75→        vtx_nodeid_t this_id;
    76→    76→        memcpy(&this_id, &id_elem, sizeof(vtx_nodeid_t));
    77→    77→        if (this_id == node_id) {
    78→    78→            return val_elem;
    79→    79→        }
    80→    80→    }
    81→    81→
    82→    82→    return VTX_VALUE_UNDEFINED;
    83→    83→}

   110→   110→
   111→   111→vtx_interp_frame_t *vtx_osr_build_interp_frame(
   112→   112→    const vtx_frame_state_t *fs,
   113→   113→    vtx_value_t (*node_to_value)(vtx_nodeid_t, void *),
   114→   114→    void *context)
   115→   115→{
   116→   116→    if (!fs) return NULL;
   117→   117→
   118→   118→    vtx_interp_frame_t *frame = calloc(1, sizeof(vtx_interp_frame_t));
   119→   119→    if (!frame) return NULL;
   120→   120→
   121→   121→    frame->method_id = fs->method_id;
   122→   122→    frame->bytecode_pc = fs->bytecode_pc;
   123→   123→    frame->local_count = fs->local_count;
   124→   124→    frame->stack_top = fs->stack_count;
   125→   125→    frame->stack_capacity = fs->stack_count;
   126→   126→    frame->caller = NULL;
   127→   127→
   128→   128→    /* Initialize enhanced fields */
   129→   129→    frame->monitors = NULL;
   130→   130→    frame->monitor_count = 0;
   131→   131→    frame->monitor_capacity = 0;
   132→   132→    frame->catch_handler_pc = (fs->exception.handler_pc != VTX_DEOPT_NO_HANDLER)
   133→   133→                              ? fs->exception.handler_pc : VTX_CATCH_NONE;
   134→   134→    frame->return_pc = 0;  /* Set by caller during frame chain reconstruction */
   135→   135→    frame->frame_kind = VTX_FRAME_INTERPRETED;
   136→   136→
   137→   137→    /* Allocate and fill locals */
   138→   138→    if (fs->local_count > 0) {
   139→   139→        frame->locals = calloc(fs->local_count, sizeof(vtx_value_t));
   140→   140→        if (!frame->locals) {
   141→   141→            free(frame);
   142→   142→            return NULL;
   143→   143→        }
   144→   144→        for (uint32_t i = 0; i < fs->local_count; i++) {
   145→   145→            if (fs->locals[i] != VTX_NODEID_INVALID) {
   146→   146→                frame->locals[i] = node_to_value(fs->locals[i], context);
   147→   147→            } else {
   148→   148→                frame->locals[i] = VTX_VALUE_UNDEFINED;
   149→   149→            }
   150→   150→        }
   151→   151→    }
   152→   152→
   153→   153→    /* Allocate and fill operand stack */
   154→   154→    if (fs->stack_count > 0) {
   155→   155→        frame->stack = calloc(fs->stack_count, sizeof(vtx_value_t));
   156→   156→        if (!frame->stack) {
   157→   157→            free(frame->locals);
   158→   158→            free(frame);
   159→   159→            return NULL;
   160→   160→        }
   161→   161→        for (uint32_t i = 0; i < fs->stack_count; i++) {
   162→   162→            if (fs->stack[i] != VTX_NODEID_INVALID) {
   163→   163→                frame->stack[i] = node_to_value(fs->stack[i], context);
   164→   164→            } else {
   165→   165→                frame->stack[i] = VTX_VALUE_UNDEFINED;
   166→   166→            }
   167→   167→        }
   168→   168→    }
   169→   169→
   170→   170→    /* Reconstruct monitor state from FrameState */
   171→   171→    if (fs->monitor_count > 0) {
   172→   172→        frame->monitors = calloc(fs->monitor_count, sizeof(vtx_osr_monitor_entry_t));
   173→   173→        if (!frame->monitors) {
   174→   174→            free(frame->stack);
   175→   175→            free(frame->locals);
   176→   176→            free(frame);
   177→   177→            return NULL;
   178→   178→        }
   179→   179→        frame->monitor_count = fs->monitor_count;
   180→   180→        frame->monitor_capacity = fs->monitor_count;
   181→   181→        for (uint32_t i = 0; i < fs->monitor_count; i++) {
   182→   182→            frame->monitors[i].local_index = 0; /* will be resolved during deopt */
   183→   183→            if (fs->monitors[i].monitor_object != VTX_NODEID_INVALID) {
   184→   184→                frame->monitors[i].object = node_to_value(fs->monitors[i].monitor_object, context);
   185→   185→            } else {
   186→   186→                frame->monitors[i].object = VTX_VALUE_UNDEFINED;
   187→   187→            }
   188→   188→        }
   189→   189→    }
   190→   190→
   191→   191→    return frame;
   192→   192→}

   194→   194→/* ========================================================================== */
   195→   195→/* OSR Up: Interpreter → Compiled Code                                        */

   197→   197→
   198→   198→__attribute__((optimize("O0")))
   199→   199→void vtx_osr_up(vtx_interp_frame_t *interp,
   200→   200→                 uint32_t method_id,
   201→   201→                 const vtx_compiled_code_t *compiled_code,
   202→   202→                 uint32_t loop_header_pc,
   203→   203→                 vtx_method_registry_t *registry,
   204→   204→                 vtx_gc_t *gc)
   205→   205→{
   206→   206→    if (!interp || !compiled_code || !compiled_code->entry_point) {
   207→   207→        fprintf(stderr, "[osr] FAIL: null check (interp=%p code=%p entry=%p)\n",
   208→   208→                (void*)interp, (void*)compiled_code,
   209→   209→                compiled_code ? (void*)compiled_code->entry_point : NULL);
   210→   210→        return;
   211→   211→    }
   212→   212→
   213→   213→    /* OSR-3: this function returns void. On a successful transition the
   214→   214→     * inline-asm trampoline jumps to the JIT entry and never returns to
   215→   215→     * C — the JIT method's NaN-boxed return value propagates back to the
   216→   216→     * caller of vtx_interp_run through RAX exactly as if the JIT had
   217→   217→     * been called normally. Returning `bool` was broken because the
   218→   218→     * post-asm RAX (the JIT return value) could be SMI 0 / undefined /
   219→   219→     * null and be misread as `false`, triggering re-execution.
   220→   220→     *
   221→   221→     * If we get here at all, OSR failed; the caller falls back to
   222→   222→     * whole-method re-enter via jit_reenter_pending. */
   223→   223→
   224→   224→    /* Verify the method matches */
   225→   225→    if (compiled_code->method_id != method_id) {
   226→   226→        fprintf(stderr, "[osr] FAIL: method_id mismatch (code=%u frame=%u)\n",
   227→   227→                compiled_code->method_id, method_id);
   228→   228→        return;
   229→   229→    }
   230→   230→
   231→   231→    /* Verify the interpreter is at the loop header PC */
   232→   232→    if (interp->bytecode_pc != loop_header_pc) {
   233→   233→        fprintf(stderr, "[osr] FAIL: pc mismatch (frame_pc=%u loop_header=%u)\n",
   234→   234→                interp->bytecode_pc, loop_header_pc);
   235→   235→        return;
   236→   236→    }
   237→   237→
   238→   238→    /* Verify frame size compatibility */
   239→   239→    if (interp->local_count > compiled_code->local_slots) {
   240→   240→        fprintf(stderr, "[osr] FAIL: locals overflow (frame=%u code=%u)\n",
   241→   241→                interp->local_count, compiled_code->local_slots);
   242→   242→        return;
   243→   243→    }
   244→   244→    if (interp->stack_top > compiled_code->stack_slots) {
   245→   245→        fprintf(stderr, "[osr] FAIL: stack overflow (frame=%u code=%u)\n",
   246→   246→                interp->stack_top, compiled_code->stack_slots);
   247→   247→        return;
   248→   248→    }
   249→   249→
   250→   250→    /* OSR-16: refuse OSR into inlined code. The trampoline only knows
   251→   251→     * how to copy a single interpreter frame into the JIT frame; if the
   252→   252→     * JIT code at the OSR entry point has inlined callees, the JIT
   253→   253→     * frame's shape (locals + spills for each inlined frame) does not
   254→   254→     * match what the asm sets up, and the JIT would read garbage from
   255→   255→     * the wrong slots. Fall through to the failure path so the dispatch
   256→   256→     * loop re-enters the JIT from method entry (which is correct for
   257→   257→     * inlined code since the prologue handles inlined frame setup). */
   258→   258→    if (compiled_code->has_inlined_frames) {
   259→   259→        fprintf(stderr, "[osr] FAIL: compiled code for method %u has inlined "
   260→   260→                "frames — refusing OSR (frame shape mismatch)\n", method_id);
   261→   261→        return;
   262→   262→    }
   263→   263→
   264→   264→    /* OSR-33: verify the JIT entry register convention matches the one
   265→   265→     * the trampoline hardcodes. The asm loads TOS/TOS-1/TOS-2/TOS-3 into
   266→   266→     * RAX/RCX/RDX/RBX (VTX_OSR_CONV_DEFAULT). If the JIT code uses a
   267→   267→     * different convention, refuse OSR instead of loading values into
   268→   268→     * the wrong registers. */
   269→   269→    if (compiled_code->entry_register_convention != VTX_OSR_CONV_DEFAULT) {
   270→   270→        fprintf(stderr, "[osr] FAIL: compiled code for method %u uses entry "
   271→   271→                "register convention %u (trampoline only supports "
   272→   272→                "VTX_OSR_CONV_DEFAULT) — refusing OSR\n",
   273→   273→                method_id, (unsigned)compiled_code->entry_register_convention);
   274→   274→        return;
   275→   275→    }
   276→   276→
   277→   277→    /* ---- Step 1: Look up the OSR entry point in the side table ----
   278→   278→     *
   279→   279→     * OSR-2 fix: codegen now records VTX_STF_OSR_ENTRY entries for each
   280→   280→     * loop-header (backward-branch target). Previously no codegen wrote
   281→   281→     * this flag, so the side-table lookup never found an entry.
   282→   282→     *
   283→   283→     * OSR-5 fix: the lookup matches the requested loop_header_pc against
   284→   284→     * the entry's bytecode_pc field, so a method with multiple OSR entry
   285→   285→     * points picks the correct loop header.
   286→   286→     *
   287→   287→     * OSR-23 fix: we use the dedicated vtx_side_table_lookup_osr_entry
   288→   288→     * (filters by VTX_STF_OSR_ENTRY flag) instead of the generic
   289→   289→     * vtx_side_table_lookup_entry which uses "largest ≤ target" semantics
   290→   290→     * and could return a non-OSR entry (e.g., a safepoint) near the
   291→   291→     * requested PC.
   292→   292→     *
   293→   293→     * DEOPT-003 fix: start with osr_entry = NULL and return false if no
   294→   294→     * entry is found, so the interpreter continues interpreting instead
   295→   295→     * of silently jumping to the function entry point. */
   296→   296→    void *osr_entry = NULL;
   297→   297→
   298→   298→    if (compiled_code->side_table != NULL) {
   299→   299→        const vtx_side_table_entry_t *osr_e = vtx_side_table_lookup_osr_entry(
   300→   300→            compiled_code->side_table, loop_header_pc);
   301→   301→        if (osr_e != NULL) {
   302→   302→            osr_entry = (uint8_t *)compiled_code->entry_point + osr_e->native_pc_offset;
   303→   303→        }
   304→   304→    }
   305→   305→
   306→   306→    /* ---- Step 2: Look up the bytecode-to-native PC mapping ----
   307→   307→     * If bc_pc_map is available, find the native offset for the loop header.
   308→   308→     * This gives us the exact entry point in the compiled code.
   309→   309→     *
   310→   310→     * OSR-27: compiled_code->code may be NULL (the install path at
   311→   311→     * codegen.c:3689 sets code = NULL after copying into the executable
   312→   312→     * cache). The current dispatch.c caller papers over this by setting
   313→   313→     * cc.code = cm->code_start, but the function's contract does not
   314→   314→     * enforce it. Add a NULL check so a future caller that forgets to
   315→   315→     * set cc.code does not dereference NULL here. */
   316→   316→    if (compiled_code->bc_pc_map != NULL && compiled_code->bc_pc_map_count > 0) {
   317→   317→        for (uint32_t i = 0; i < compiled_code->bc_pc_map_count; i++) {
   318→   318→            if (compiled_code->bc_pc_map[i].bytecode_pc == loop_header_pc) {
   319→   319→                /* OSR-27: NULL check on compiled_code->code. */
   320→   320→                if (compiled_code->code == NULL) {
   321→   321→                    fprintf(stderr, "[osr] FAIL: compiled_code->code is NULL "
   322→   322→                            "for method %u (caller did not set cc.code) — "
   323→   323→                            "cannot compute OSR entry\n", method_id);
   324→   324→                    return;
   325→   325→                }
   326→   326→                /* OSR-24: verify the interpreter's stack_top matches the
   327→   327→                 * stack depth the JIT code expects at this OSR entry
   328→   328→                 * point. If they differ, the JIT would read garbage from
   329→   329→                 * the spill slots (or miss values it expects in
   330→   330→                 * registers). Fall through to the failure path on
   331→   331→                 * mismatch — the dispatch loop re-enters from method
   332→   332→                 * entry, which is always correct. */
   333→   333→                if (compiled_code->bc_pc_map[i].stack_depth != interp->stack_top) {
   334→   334→                    fprintf(stderr, "[osr] FAIL: stack depth mismatch at "
   335→   335→                            "loop_header_pc=%u (interp=%u expected=%u) — "
   336→   336→                            "refusing OSR\n",
   337→   337→                            loop_header_pc,
   338→   338→                            interp->stack_top,
   339→   339→                            compiled_code->bc_pc_map[i].stack_depth);
   340→   340→                    return;
   341→   341→                }
   342→   342→                /* Found the mapping — use this as the OSR entry */
   343→   343→                osr_entry = (uint8_t *)compiled_code->code +
   344→   344→                            compiled_code->bc_pc_map[i].native_offset;
   345→   345→                break;
   346→   346→            }
   347→   347→        }
   348→   348→    }
   349→   349→
   350→   350→    /* DEOPT-003 fix: if no OSR entry was found, fail instead of silently
   351→   351→     * jumping to the function entry (which would re-execute the prologue
   352→   352→     * and all code before the loop, violating side-effect-once semantics
   353→   353→     * and the interpreter's loop-header assumption). */
   354→   354→    if (osr_entry == NULL) {
   355→   355→        fprintf(stderr, "[osr] no OSR entry for loop_header_pc=%u — "
   356→   356→                "continuing in interpreter\n", loop_header_pc);
   357→   357→        return;
   358→   358→    }
   359→   359→
   360→   360→    /* ---- OSR-11 fix: re-check the registry's current cm before the asm jump ----
   361→   361→     *
   362→   362→     * The dispatch loop fetched `cm` from the registry earlier
   363→   363→     * (dispatch.c:3063). Between then and now, the method may have been
   364→   364→     * invalidated or recompiled (e.g., by a concurrent retrace triggered
   365→   365→     * by a guard failure elsewhere). The old cm may have been freed
   366→   366→     * (install.c:303), and its code_start memory may have been reclaimed.
   367→   367→     *
   368→   368→     * We re-fetch the current cm via the registry and compare its
   369→   369→     * code_start against the cached `compiled_code->entry_point`. If
   370→   370→     * they differ, the cached code is stale — return and let the
   371→   371→     * dispatch loop re-enter the JIT from method entry on the next call
   372→   372→     * (when it'll fetch the fresh cm).
   373→   373→     *
   374→   374→     * Note: there's still a small TOCTOU window between this check and
   375→   375→     * the asm jump, but this narrows it dramatically and catches the
   376→   376→     * common case (compile thread finished retrace between the original
   377→   377→     * dispatch fetch and now). True elimination would require hazard
   378→   378→     * pointers / epoch-based reclamation, which is out of scope. */
   379→   379→    if (registry != NULL) {
   380→   380→        vtx_compiled_method_t *current_cm = vtx_method_registry_get(registry, method_id);
   381→   381→        if (current_cm == NULL ||
   382→   382→            current_cm->code_start != (uint8_t *)compiled_code->entry_point ||
   383→   383→            !current_cm->is_valid) {
   384→   384→            fprintf(stderr, "[osr] FAIL: version mismatch — method %u was "
   385→   385→                    "recompiled/invalidated since the dispatch loop fetched cm "
   386→   386→                    "(current=%p cached=%p valid=%d)\n",
   387→   387→                    method_id,
   388→   388→                    current_cm ? (void*)current_cm->code_start : NULL,
   389→   389→                    (void*)compiled_code->entry_point,
   390→   390→                    current_cm ? (int)current_cm->is_valid : 0);
   391→   391→            return;
   392→   392→        }
   393→   393→    }
   394→   394→
   395→   395→    /* ---- Step 3: Set up the JIT frame with interpreter values ----
   396→   396→     *
   397→   397→     * The JIT frame layout (from the compiled_code's frame_layout) is:
   398→   398→     *   [RBP+32]  = return address
   399→   399→     *   [RBP+24]  = method pointer
   400→   400→     *   [RBP+16]  = deopt_info pointer
   401→   401→     *   [RBP+8]   = profile_data pointer
   402→   402→     *   [RBP+0]   = caller RBP
   403→   403→     *   [RBP-8]   = first local slot
   404→   404→     *   [RBP-16]  = second local slot
   405→   405→     *   ...
   406→   406→     *   [RBP-8*N] = Nth local slot
   407→   407→     *   [RBP+spill_base] = first spill slot (operand stack)
   408→   408→     *   ...
   409→   409→     *
   410→   410→     * We build the JIT frame on the native stack using inline assembly,
   411→   411→     * copy interpreter locals and operand stack into the frame slots,
   412→   412→     * then jump to osr_entry. The C function never returns normally
   413→   413→     * after a successful transition.
   414→   414→     */
   415→   415→
   416→   416→    const vtx_jit_frame_layout_t *layout = &compiled_code->frame_layout;
   417→   417→    uint32_t frame_sz  = layout->total_frame_size;
   418→   418→    int32_t  l_base    = layout->locals_base;   /* negative offset of local[0] from RBP */
   419→   419→    int32_t  s_base    = layout->spill_base;    /* negative offset of spill[0] from RBP */
   420→   420→    uint32_t nlocals   = interp->local_count;
   421→   421→    uint32_t nstack    = interp->stack_top;
   422→   422→
   423→   423→    /* OSR-32 fix: the `osr_active` flag was set here but never read by
   424→   424→     * any code (the GC and stack walker don't check it). The intended use
   425→   425→     * was to skip OSR'd interpreter frames during root scanning, but this
   426→   426→     * was never implemented. Per the surgical-removal rule, we delete the
   427→   427→     * dead setter rather than carry the dead flag forward. The interp
   428→   428→     * frame remains valid (the JIT code returns to the interpreter's
   429→   429→     * caller via the patched return address) and the GC continues to scan
   430→   430→     * it as before. If duplicate-root scanning becomes a real performance
   431→   431→     * problem, a future patch should add the check at the GC site, not
   432→   432→     * via a half-wired runtime flag. */
   433→   433→
   434→   434→    /* Prepare ALL parameters in a struct on the stack to reduce
   435→   435→     * register pressure on the inline asm. x86-64 has limited
   436→   436→     * registers and we clobber many, so passing a single pointer
   437→   437→     * avoids "impossible constraints" errors.
   438→   438→     *
   439→   439→     * Struct layout (all 8-byte slots, natural alignment):
   440→   440→     *   [0]   frame_sz      (uint64_t)
   441→   441→     *   [8]   l_base        (int64_t, sign-extended)
   442→   442→     *   [16]  s_base        (int64_t, sign-extended)
   443→   443→     *   [24]  nlocals       (uint64_t)
   444→   444→     *   [32]  nstack        (uint64_t)
   445→   445→     *   [40]  src_locals    (pointer)
   446→   446→     *   [48]  src_stack     (pointer)
   447→   447→     *   [56]  target        (pointer)
   448→   448→     *   [64]  method_desc   (pointer)
   449→   449→     *   [72]  deopt_ptr     (pointer)
   450→   450→     *   [80]  profile_data  (pointer)        -- OSR-30
   451→   451→     */
   452→   452→    struct osr_params {
   453→   453→        uint64_t frame_sz;
   454→   454→        int64_t  l_base;
   455→   455→        int64_t  s_base;
   456→   456→        uint64_t nlocals;
   457→   457→        uint64_t nstack;
   458→   458→        vtx_value_t             *src_locals;
   459→   459→        vtx_value_t             *src_stack;
   460→   460→        void                    *target;
   461→   461→        const vtx_method_desc_t *method_desc;
   462→   462→        vtx_deopt_info_t        *deopt_ptr;
   463→   463→        void                    *profile_data;   /* OSR-30: written to [RBP+8] */
   464→   464→    };
   465→   465→
   466→   466→    struct osr_params params;
   467→   467→    params.frame_sz    = (uint64_t)frame_sz;
   468→   468→    params.l_base      = (int64_t)l_base;
   469→   469→    params.s_base      = (int64_t)s_base;
   470→   470→    params.nlocals     = (uint64_t)nlocals;
   471→   471→    params.nstack      = (uint64_t)nstack;
   472→   472→    params.src_locals  = interp->locals;
   473→   473→    params.src_stack   = interp->stack;
   474→   474→    params.target      = osr_entry;
   475→   475→    params.method_desc = compiled_code->method;
   476→   476→    params.deopt_ptr   = compiled_code->deopt_info;
   477→   477→    /* OSR-30: pass profile_data through params so the asm can write it
   478→   478→     * to [RBP+8] in the JIT frame header. NULL is acceptable (matches
   479→   479→     * pre-fix behavior); the dispatch caller populates this from the
   480→   480→     * interp's profiler when known. */
   481→   481→    params.profile_data = compiled_code->profile_data;
   482→   482→
   483→   483→    /* OSR-12: poll for a pending safepoint immediately before the asm
   484→   484→     * jump. The dispatch loop's last safepoint check happened at the
   485→   485→     * backward branch; a GC or invalidation may have been requested
   486→   486→     * between then and now. Without this poll, the JIT code entered via
   487→   487→     * OSR would not observe the safepoint until its own loop-back-edge
   488→   488→     * poll, which may be far away if the OSR entry is at a loop header
   489→   489→     * with a long body. vtx_gc_safepoint is the existing fast-path poll
   490→   490→     * used at backward branches in dispatch.c — it just reads an atomic
   491→   491→     * flag and only does real work when a GC was requested. */
   492→   492→    if (gc != NULL) {
   493→   493→        vtx_gc_safepoint(gc);
   494→   494→    }
   495→   495→
   496→   496→    /*
   497→   497→     * Inline assembly trampoline (x86-64, System V ABI, Linux).
   498→   498→     *
   499→   499→     * We:
   500→   500→     *   1. Read the current C frame's caller RBP and return address
   501→   501→     *      (so the JIT frame returns to the right place)
   502→   502→     *   2. Allocate the JIT frame on the stack
   503→   503→     *   3. Write the frame header (caller RBP, profile_data, deopt_info,
   504→   504→     *      method_ptr, return address)
   505→   505→     *   4. Copy interpreter locals into JIT local slots
   506→   506→     *   5. Copy interpreter operand stack into JIT spill slots
   507→   507→     *   6. Load top stack values into expression registers (RAX, RCX, RDX, RBX)
   508→   508→     *   7. Set RBP/RSP to the new JIT frame
   509→   509→     *   8. Jump to osr_entry
   510→   510→     *
   511→   511→     * After this asm block, the C function never returns.
   512→   512→     *
   513→   513→     * Register usage:
   514→   514→     *   r12 = caller RBP
   515→   515→     *   r13 = return address
   516→   516→     *   r14 = new RBP
   517→   517→     *   r15 = pointer to osr_params struct (loaded once, used throughout)
   518→   518→     *   rax, rcx, rdx, rsi, r8, r9, r10 = temporaries
   519→   519→     *
   520→   520→     * OSR-33 (entry register convention): the asm loads TOS, TOS-1,
   521→   521→     * TOS-2, TOS-3 into RAX, RCX, RDX, RBX respectively. This matches
   522→   522→     * the VTX_OSR_CONV_DEFAULT convention declared in
   523→   523→     * codecache/types.h and the expression-stack register assignment
   524→   524→     * documented in baseline/frame_layout.h:54-55:
   525→   525→     *
   526→   526→     *     TOS   -> RAX (reg 0)
   527→   527→     *     TOS-1 -> RCX (reg 1)
   528→   528→     *     TOS-2 -> RDX (reg 2)
   529→   529→     *     TOS-3 -> RBX (reg 3)
   530→   530→     *
   531→   531→     * Values deeper than TOS-3 are spilled to the frame and read by
   532→   532→     * the JIT code from spill slots. vtx_osr_up verifies
   533→   533→     * compiled_code->entry_register_convention == VTX_OSR_CONV_DEFAULT
   534→   534→     * before reaching this asm, so a future codegen change that emits
   535→   535→     * a different convention will fail the OSR transition cleanly
   536→   536→     * (falling back to whole-method re-enter) instead of silently
   537→   537→     * loading values into the wrong registers.
   538→   538→     *
   539→   539→     * OSR-6 (callee-saved register preservation): the JIT epilogue
   540→   540→     * (baseline/codegen.c:1146-1170) restores RBX from [RBP-8] and
   541→   541→     * R12 from [RBP-16]. Before this fix, the OSR-up asm never wrote
   542→   542→     * the caller's values to those slots, so the JIT epilogue restored
   543→   543→     * garbage into RBX and R12 — corrupting the caller of vtx_interp_run.
   544→   544→     * Step 2.5 below writes the caller's current RBX and R12 to the
   545→   545→     * saved-register slots BEFORE the asm clobbers them, so the JIT
   546→   546→     * epilogue restores the correct values.
   547→   547→     *
   548→   548→     * R13/R14/R15 are also clobbered by the asm trampoline, but the
   549→   549→     * baseline JIT codegen does NOT save/restore them (no slots in the
   550→   550→     * prologue/epilogue) and does not use them as callee-saved. They
   551→   551→     * are caller-saved from the JIT codegen's perspective. The asm
   552→   552→     * trampoline follows the same convention — it clobbers them and
   553→   553→     * does not restore them. This is consistent with the JIT codegen's
   554→   554→     * behavior for non-OSR JIT calls (the caller of vtx_dispatch_jit
   555→   555→     * also sees R13/R14/R15 clobbered). Extending the JIT frame to
   556→   556→     * save/restore R13/R14/R15 would be a codegen-wide change and is
   557→   557→     * out of scope for the OSR-up fix; this is documented as a known
   558→   558→     * limitation matching the existing JIT convention.
   559→   559→     */
   560→   560→    __asm__ __volatile__ (
   561→   561→        /* ---- Read current frame link data before modifying RBP ---- */
   562→   562→        "movq (%%rbp), %%r12\n\t"           /* r12 = caller's RBP (from current C frame) */
   563→   563→        "movq 8(%%rbp), %%r13\n\t"          /* r13 = return address   (from current C frame) */
   564→   564→
   565→   565→        /* ---- Load params pointer into r15 (single input register) ---- */
   566→   566→        "movq %[params], %%r15\n\t"
   567→   567→
   568→   568→        /* ---- Step 1: Allocate the JIT frame on the stack ---- */
   569→   569→        "movq 0(%%r15), %%rax\n\t"          /* load frame_sz from params[0] */
   570→   570→        "addq $48, %%rax\n\t"               /* 48 = 40 header + 8 for alignment margin */
   571→   571→        "addq $15, %%rax\n\t"               /* Fix C24: round UP for alignment */
   572→   572→        "andq $-16, %%rax\n\t"              /* align to 16 bytes (rounds up, not down) */
   573→   573→        "subq %%rax, %%rsp\n\t"             /* allocate frame on stack */
   574→   574→
   575→   575→        /* ---- Step 2: Compute new RBP ---- */
   576→   576→        "movq 0(%%r15), %%rax\n\t"          /* reload frame_sz */
   577→   577→        "leaq 8(%%rsp, %%rax), %%r14\n\t"   /* r14 = new RBP */
   578→   578→
   579→   579→        /* ---- Step 2.5: OSR-6 — save callee-saved registers to JIT frame ----
   580→   580→         * The JIT epilogue restores RBX from [RBP-8] and R12 from [RBP-16].
   581→   581→         * The OSR-up asm must initialize these slots with the caller's
   582→   582→         * CURRENT values BEFORE clobbering them, otherwise the JIT
   583→   583→         * epilogue restores garbage. At this point r14 = new RBP, and
   584→   584→         * RBX and R12 still hold the caller's values (the asm hasn't
   585→   585→         * touched them yet). VTX_FRAME_SAVED_RBX_OFFSET = -8 and
   586→   586→         * VTX_FRAME_SAVED_R12_OFFSET = -16 (from baseline/frame_layout.h). */
   587→   587→        "movq %%rbx, -8(%%r14)\n\t"         /* [RBP-8]  = caller's RBX */
   588→   588→        "movq %%r12, -16(%%r14)\n\t"        /* [RBP-16] = caller's R12 */
   589→   589→
   590→   590→        /* ---- Step 3: Write frame header above RBP ---- */
   591→   591→        "movq %%r12, 0(%%r14)\n\t"          /* [RBP+0]  = caller RBP */
   592→   592→        "movq 80(%%r15), %%rax\n\t"         /* OSR-30: load profile_data from params[80] */
   593→   593→        "movq %%rax, 8(%%r14)\n\t"          /* [RBP+8]  = profile_data (was hardcoded NULL) */
   594→   594→        "movq 72(%%r15), %%rax\n\t"         /* load deopt_ptr from params[72] */
   595→   595→        "movq %%rax, 16(%%r14)\n\t"         /* [RBP+16] = deopt_info */
   596→   596→        "movq 64(%%r15), %%rax\n\t"         /* load method_desc from params[64] */
   597→   597→        "movq %%rax, 24(%%r14)\n\t"         /* [RBP+24] = method_ptr */
   598→   598→        "movq %%r13, 32(%%r14)\n\t"         /* [RBP+32] = return address */
   599→   599→
   600→   600→        /* ---- Step 4: Copy interpreter locals into JIT frame local slots ---- */
   601→   601→        "movq 24(%%r15), %%rcx\n\t"         /* load nlocals from params[24] */
   602→   602→        "testq %%rcx, %%rcx\n\t"
   603→   603→        "jz 1f\n\t"                         /* skip if no locals */
   604→   604→        "movq 40(%%r15), %%rsi\n\t"         /* src_locals from params[40] */
   605→   605→        "movq 8(%%r15), %%rax\n\t"          /* l_base from params[8] */
   606→   606→        "0:\n\t"
   607→   607→        "movq (%%rsi), %%rdx\n\t"           /* load local value */
   608→   608→        "movq %%rdx, (%%r14, %%rax)\n\t"    /* store at [RBP + offset] */
   609→   609→        "addq $8, %%rsi\n\t"                /* next source slot */
   610→   610→        "subq $8, %%rax\n\t"                /* next offset (more negative) */
   611→   611→        "decq %%rcx\n\t"
   612→   612→        "jnz 0b\n\t"
   613→   613→        "1:\n\t"
   614→   614→
   615→   615→        /* ---- Step 5: Copy interpreter operand stack into JIT frame spill slots ---- */
   616→   616→        "movq 32(%%r15), %%rcx\n\t"         /* load nstack from params[32] */
   617→   617→        "testq %%rcx, %%rcx\n\t"
   618→   618→        "jz 2f\n\t"                         /* skip if no stack values */
   619→   619→        "movq 48(%%r15), %%rsi\n\t"         /* src_stack from params[48] */
   620→   620→        "movq 16(%%r15), %%rax\n\t"         /* s_base from params[16] */
   621→   621→        "0:\n\t"
   622→   622→        "movq (%%rsi), %%rdx\n\t"           /* load stack value */
   623→   623→        "movq %%rdx, (%%r14, %%rax)\n\t"    /* store at [RBP + offset] */
   624→   624→        "addq $8, %%rsi\n\t"                /* next source slot */
   625→   625→        "subq $8, %%rax\n\t"                /* next offset */
   626→   626→        "decq %%rcx\n\t"
   627→   627→        "jnz 0b\n\t"
   628→   628→        "2:\n\t"
   629→   629→
   630→   630→        /* ---- Step 6: Load top expression stack values into JIT registers ---- */
   631→   631→        "movq 32(%%r15), %%r8\n\t"          /* save nstack for reuse */
   632→   632→        "movq 16(%%r15), %%r9\n\t"          /* save sbase for reuse */
   633→   633→        "testq %%r8, %%r8\n\t"
   634→   634→        "jz 4f\n\t"                         /* skip if no stack values */
   635→   635→
   636→   636→        /* Load TOS (stack[stack_top-1]) → RAX */
   637→   637→        "movq %%r8, %%rax\n\t"
   638→   638→        "decq %%rax\n\t"
   639→   639→        "shlq $3, %%rax\n\t"               /* rax = (stack_top-1) * 8 */
   640→   640→        "movq %%r9, %%rdx\n\t"
   641→   641→        "subq %%rax, %%rdx\n\t"            /* rdx = s_base - (stack_top-1)*8 */
   642→   642→        "movq (%%r14, %%rdx), %%rax\n\t"   /* RAX = TOS value */
   643→   643→
   644→   644→        "cmpq $1, %%r8\n\t"
   645→   645→        "je 4f\n\t"                         /* only 1 stack value */
   646→   646→
   647→   647→        /* Load TOS-1 (stack[stack_top-2]) → RCX */
   648→   648→        "movq %%r8, %%rcx\n\t"
   649→   649→        "subq $2, %%rcx\n\t"
   650→   650→        "shlq $3, %%rcx\n\t"               /* rcx = (stack_top-2) * 8 */
   651→   651→        "movq %%r9, %%rdx\n\t"
   652→   652→        "subq %%rcx, %%rdx\n\t"
   653→   653→        "movq (%%r14, %%rdx), %%rcx\n\t"   /* RCX = TOS-1 value */
   654→   654→
   655→   655→        "cmpq $2, %%r8\n\t"
   656→   656→        "je 4f\n\t"                         /* only 2 stack values */
   657→   657→
   658→   658→        /* Load TOS-2 (stack[stack_top-3]) → RDX */
   659→   659→        "movq %%r8, %%rdx\n\t"
   660→   660→        "subq $3, %%rdx\n\t"
   661→   661→        "shlq $3, %%rdx\n\t"               /* rdx = (stack_top-3) * 8 */
   662→   662→        "movq %%r9, %%r10\n\t"
   663→   663→        "subq %%rdx, %%r10\n\t"
   664→   664→        "movq (%%r14, %%r10), %%rdx\n\t"   /* RDX = TOS-2 value */
   665→   665→
   666→   666→        "cmpq $3, %%r8\n\t"
   667→   667→        "je 4f\n\t"                         /* only 3 stack values */
   668→   668→
   669→   669→        /* Load TOS-3 (stack[stack_top-4]) → RBX */
   670→   670→        "movq %%r8, %%rbx\n\t"
   671→   671→        "subq $4, %%rbx\n\t"
   672→   672→        "shlq $3, %%rbx\n\t"               /* rbx = (stack_top-4) * 8 */
   673→   673→        "movq %%r9, %%r10\n\t"
   674→   674→        "subq %%rbx, %%r10\n\t"
   675→   675→        "movq (%%r14, %%r10), %%rbx\n\t"   /* RBX = TOS-3 value */
   676→   676→        "jmp 4f\n\t"
   677→   677→
   678→   678→        "4:\n\t"
   679→   679→
   680→   680→        /* ---- Step 7: Set RBP and RSP to the new JIT frame ---- */
   681→   681→        "movq %%r14, %%rbp\n\t"             /* RBP = new frame base */
   682→   682→        "movq 0(%%r15), %%rax\n\t"          /* reload frame_sz */
   683→   683→        "negq %%rax\n\t"
   684→   684→        "leaq (%%rbp, %%rax), %%rsp\n\t"   /* RSP = RBP - frame_sz */
   685→   685→
   686→   686→        /* ---- Step 8: Jump to the OSR entry point ---- */
   687→   687→        "movq 56(%%r15), %%rax\n\t"         /* load target from params[56] */
   688→   688→        "jmp *%%rax\n\t"
   689→   689→
   690→   690→        /* OSR-31: safety trap; should never execute — the preceding
   691→   691→         * `jmp *%%rax` is unconditional and `__builtin_unreachable()`
   692→   692→         * below tells the compiler the asm doesn't fall through. Kept
   693→   693→         * as a defensive trap so that if a future code change breaks the
   694→   694→         * asm's control flow (e.g., accidentally writes a `ret` instead
   695→   695→         * of `jmp`), we trap loudly here instead of executing whatever
   696→   696→         * bytes happen to follow in the code section. Do NOT delete. */
   697→   697→        "int3\n\t"
   698→   698→
   699→   699→        : /* no outputs — we never return */
   700→   700→        : [params]  "r"(&params)
   701→   701→        /* OSR-28: rbp and rsp are modified by the asm (movq %%r14,%%rbp
   702→   702→         * and the leaq/subq that adjust %%rsp). They were not in the
   703→   703→         * clobber list because the asm ends with an unconditional jmp +
   704→   704→         * __builtin_unreachable — the compiler never generates code
   705→   705→         * after the asm, so it doesn't matter that rbp/rsp are wrong.
   706→   706→         * Adding them is defensive: if a future change adds a fall-
   707→   707→         * through path after the asm, the compiler would generate
   708→   708→         * wrong code without these in the clobber list. */
   709→   709→        : "rax", "rbx", "rcx", "rdx", "rsi", "r8", "r9", "r10",
   710→   710→          "r12", "r13", "r14", "r15", "rbp", "rsp", "memory"
   711→   711→    );
   712→   712→
   713→   713→    /* The asm block jumps to osr_entry and never returns here. */
   714→   714→    __builtin_unreachable();
   715→   715→}

   717→   717→/* ========================================================================== */
   718→   718→/* OSR Down: Compiled Code → Interpreter                                      */
   719→   719→/* ========================================================================== */
   720→   720→
   721→   721→vtx_interp_frame_t *vtx_osr_down(vtx_interp_frame_t *interp,
   722→   722→                                   const vtx_osr_deopt_context_t *deopt_ctx)
   723→   723→{
   724→   724→    if (!deopt_ctx || !deopt_ctx->frame_state) {
   725→   725→        return NULL;
   726→   726→    }
   727→   727→
   728→   728→    /* Step 1: Look up FrameState from side table (already provided in deopt_ctx) */
   729→   729→    const vtx_frame_state_t *fs = deopt_ctx->frame_state;
   730→   730→
   731→   731→    /* Step 2: Set up the resolution context */
   732→   732→    vtx_resolve_context_t ctx;
   733→   733→    ctx.register_map = deopt_ctx->register_map;
   734→   734→    ctx.register_map_size = deopt_ctx->register_count;
   735→   735→
   736→   736→    /* Step 3: Build the interpreter frame for the innermost method */
   737→   737→    vtx_interp_frame_t *new_frame = vtx_osr_build_interp_frame(
   738→   738→        fs, resolve_node_callback, &ctx);
   739→   739→    if (!new_frame) return NULL;
   740→   740→
   741→   741→    /* Step 4: Handle monitors — relock if needed.
   742→   742→     * For each monitor in the FrameState, we need to reacquire the lock
   743→   743→     * on the monitor object. The monitor object's value is resolved from
   744→   744→     * the register map. */
   745→   745→    if (fs->monitor_count > 0) {
   746→   746→        for (uint32_t i = 0; i < fs->monitor_count; i++) {
   747→   747→            vtx_nodeid_t mon_node = fs->monitors[i].monitor_object;
   748→   748→            if (mon_node != VTX_NODEID_INVALID) {
   749→   749→                vtx_value_t mon_val = resolve_node_callback(mon_node, &ctx);
   750→   750→                /* In a real implementation, we would call the runtime
   751→   751→                 * to re-enter the monitor on the resolved object.
   752→   752→                 * The object must be a heap pointer. */
   753→   753→                if (vtx_is_heap_ptr(mon_val)) {
   754→   754→                    /* vtx_runtime_monitor_enter(vtx_heap_ptr(mon_val)); */
   755→   755→                }
   756→   756→            }
   757→   757→        }
   758→   758→    }
   759→   759→
   760→   760→    /* Step 5: Handle exception handler state */
   761→   761→    if (fs->exception.handler_pc != VTX_DEOPT_NO_HANDLER) {
   762→   762→        /* The interpreter will pick up the exception handler from
   763→   763→         * the method's exception table at the deopt PC. No additional
   764→   764→         * work needed here — the bytecode_pc is set correctly. */
   765→   765→    }
   766→   766→
   767→   767→    /* Step 6: Walk the caller chain and reconstruct caller frames.
   768→   768→     *
   769→   769→     * B24 fix: The `caller` field is typed as `vtx_frame_state_t*` but
   770→   770→     * stores `vtx_interp_frame_t*`. Use a union/cast through void* to
   771→   771→     * avoid undefined behavior. The field is only used as a linked-list
   772→   772→     * next pointer — it never accesses vtx_frame_state_t fields through it. */
   773→   773→    vtx_interp_frame_t *current = new_frame;
   774→   774→    const vtx_frame_state_t *caller_fs = fs->caller;
   775→   775→    while (caller_fs != NULL) {
   776→   776→        vtx_interp_frame_t *caller_frame = vtx_osr_build_interp_frame(
   777→   777→            caller_fs, resolve_node_callback, &ctx);
   778→   778→        if (!caller_frame) {
   779→   779→            /* Clean up already-built frames */
   780→   780→            vtx_interp_frame_t *f = new_frame;
   781→   781→            while (f) {
   782→   782→                /* DEOPT-005 fix: caller is void * now, cast directly. */
   783→   783→                vtx_interp_frame_t *next = (vtx_interp_frame_t *)f->caller;
   784→   784→                free(f->locals);
   785→   785→                free(f->stack);
   786→   786→                free(f);
   787→   787→                f = next;
   788→   788→            }
   789→   789→            return NULL;
   790→   790→        }
   791→   791→        /* DEOPT-005 fix: caller is void *, store the interp frame pointer
   792→   792→         * directly (no type punning). */
   793→   793→        current->caller = caller_frame;
   794→   794→        current = caller_frame;
   795→   795→        caller_fs = caller_fs->caller;
   796→   796→    }
   797→   797→
   798→   798→    /* Step 7: Transfer to interpreter.
   799→   799→     *
   800→   800→     * B25 fix: Use the frame_state's bytecode_pc (the actual deopt PC)
   801→   801→     * not the return_pc (which is the caller's resume PC). The old code
   802→   802→     * used new_frame->bytecode_pc which was set from return_pc in
   803→   803→     * vtx_osr_build_interp_frame — that resumes at the wrong instruction. */
   804→   804→    if (interp) {
   805→   805→        interp->method_id = new_frame->method_id;
   806→   806→        /* B25 fix: Use the deopt context's bytecode_pc, which is the
   807→   807→         * PC where the guard failed — the interpreter should resume
   808→   808→         * at that point, not at the caller's return PC. */
   809→   809→        interp->bytecode_pc = fs->bytecode_pc;
   810→   810→    }
   811→   811→
   812→   812→    return new_frame;
   813→   813→}
