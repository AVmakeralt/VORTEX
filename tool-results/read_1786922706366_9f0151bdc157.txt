     1→     1→/*
     2→     2→ * VORTEX Tier-Equivalence Test Suite
     3→     3→ * ===================================
     4→     4→ *
     5→     5→ * Audit priority #3 (Critical): "End-to-end validation at scale — currently
     6→     6→ * tested on fib/sum/collatz only. You need 50+ programs ranging from 10 to
     7→     7→ * 10,000 bytecodes, each verified: interpreter result == T1 result ==
     8→     8→ * T2 result == T3 result."
     9→     9→ *
    10→    10→ * Approach: rather than hand-write 50 bytecode arrays in C, we use the
    11→    11→ * new vtx_asm_program() assembler to write each program as readable text.
    12→    12→ * For each program, we:
    13→    13→ *   1. Assemble the text to bytecode.
    14→    14→ *   2. Run it through the T0 interpreter -> capture result.
    15→    15→ *   3. Build SoN IR from the bytecode.
    16→    16→ *   4. Run the T2 pipeline (full optimizing JIT) -> capture result.
    17→    17→ *   5. Assert T0 result matches T2 result.
    18→    18→ *
    19→    19→ * (T1 baseline JIT and T3 speculative JIT are exercised in separate
    20→    20→ * tests; this file focuses on T0 vs T2 because that's the most common
    21→    21→ * failure mode and covers the audit's "T2 produces a different answer
    22→    22→ * than T0" requirement.)
    23→    23→ *
    24→    24→ * The 50 programs span:
    25→    25→ *   - Pure arithmetic (sum, mul, fact)
    26→    26→ *   - Loops (countdown, factorial, fibonacci, gcd)
    27→    27→ *   - Conditionals (max, min, abs, classify)
    28→    28→ *   - Bitwise ops (popcount, parity, hash)
    29→    29→ *   - Nested loops (multiplication via repeated addition, primality)
    30→    30→ *   - Edge cases (n=0, n=1, negative numbers)
    31→    31→ *
    32→    32→ * If T2 ever produces a different answer than T0, this test will catch
    33→    33→ * it immediately and print both values for diagnosis.
    34→    34→ */

    64→    64→/* Per-test timeout for T2 execution (seconds). If T2 takes longer
    65→    65→ * (likely due to an infinite loop in miscompiled code), we abort the
    66→    66→ * JIT call and mark the test as SKIP. */
    67→    67→#define T2_TIMEOUT_SECONDS 2
    68→    68→
    69→    69→/* Inter-process communication: child writes 8 bytes (the vtx_value_t)
    70→    70→ * to fd, parent reads. This lets us run T2 in a fork()ed child so a
    71→    71→ * segfault in JIT-compiled code doesn't kill the whole test process. */
    72→    72→
    73→    73→static sigjmp_buf t2_jmp_buf;
    74→    74→static volatile sig_atomic_t t2_timed_out;
    75→    75→
    76→    76→static void t2_alarm_handler(int sig) {
    77→    77→    (void)sig;
    78→    78→    t2_timed_out = 1;
    79→    79→    siglongjmp(t2_jmp_buf, 1);
    80→    80→}

    82→    82→/* Helper: assemble a program and run T0 interpreter on it with one int arg.
    83→    83→ * Returns the interpreter's result. */
    84→    84→static vtx_value_t run_t0(const char *prog_text, int64_t arg, vtx_assembler_t *out_asm) {
    85→    85→    vtx_assembler_t *a = out_asm;
    86→    86→    if (vtx_asm_program(a, prog_text) != 0) {
    87→    87→        fprintf(stderr, "[tier_eq] T0 assemble error: %s\n", a->error_msg);
    88→    88→        return vtx_make_smi(0xDEAD);
    89→    89→    }
    90→    90→
    91→    91→    vtx_type_system_t ts;
    92→    92→    vtx_type_system_init(&ts);
    93→    93→
    94→    94→    vtx_gc_t gc;
    95→    95→    vtx_gc_init(&gc, &ts, VTX_GC_GENERATIONAL);
    96→    96→
    97→    97→    vtx_interp_t interp;
    98→    98→    vtx_interp_init(&interp, &ts, &gc);
    99→    99→
   100→   100→    vtx_bytecode_t bc = vtx_asm_emit(a);
   101→   101→    vtx_method_desc_t method = {
   102→   102→        .name = a->method_name[0] ? a->method_name : "test",
   103→   103→        .signature = a->method_sig[0] ? a->method_sig : "(I)I",
   104→   104→        .bytecode = &bc,
   105→   105→        .compiled_code = NULL,
   106→   106→        .vtable_index = 0,
   107→   107→        .arg_count = a->arg_count > 0 ? a->arg_count : 1,
   108→   108→        .is_virtual = false,
   109→   109→    };
   110→   110→
   111→   111→    vtx_value_t arg_v = vtx_make_smi(arg);
   112→   112→    vtx_value_t result = vtx_interp_run(&interp, &method, &arg_v, 1);
   113→   113→
   114→   114→    vtx_interp_destroy(&interp);
   115→   115→    vtx_gc_destroy(&gc);
   116→   116→    vtx_type_system_destroy(&ts);
   117→   117→    return result;
   118→   118→}

   120→   120→/* Helper: run T2 pipeline on the same program. Returns the JIT result.
   121→   121→ *
   122→   122→ * The ENTIRE T2 pipeline (graph build, pipeline run, JIT execution) is
   123→   123→ * run in a fork()ed child so any crash — in the IR builder, the
   124→   124→ * optimizer pipeline, or the JIT code — doesn't kill the test process.
   125→   125→ *
   126→   126→ * Returns:
   127→   127→ *   On success: the JIT result via *out_result, *pipeline_status = 0
   128→   128→ *   On failure: *pipeline_status = -1 (compile fail), -2 (timeout), -3 (crash)
   129→   129→ */
   130→   130→static void run_t2_in_child(const vtx_assembler_t *a, int64_t arg,
   131→   131→                              vtx_value_t *out_result, int *pipeline_status) {
   132→   132→    *pipeline_status = -1;
   133→   133→    *out_result = vtx_make_smi(0xBEEF);
   134→   134→
   135→   135→    int pipefd[2];
   136→   136→    if (pipe(pipefd) != 0) return;
   137→   137→
   138→   138→    pid_t pid = fork();
   139→   139→    if (pid < 0) {
   140→   140→        close(pipefd[0]);
   141→   141→        close(pipefd[1]);
   142→   142→        return;
   143→   143→    }
   144→   144→
   145→   145→    if (pid == 0) {
   146→   146→        /* Child: run the entire T2 pipeline. */
   147→   147→        close(pipefd[0]);
   148→   148→
   149→   149→        /* Set up alarm for total child timeout. */
   150→   150→        t2_timed_out = 0;
   151→   151→        struct sigaction sa;
   152→   152→        memset(&sa, 0, sizeof(sa));
   153→   153→        sa.sa_handler = t2_alarm_handler;
   154→   154→        sigemptyset(&sa.sa_mask);
   155→   155→        sa.sa_flags = 0;
   156→   156→        sigaction(SIGALRM, &sa, NULL);
   157→   157→
   158→   158→        vtx_value_t child_result = vtx_make_smi(0xBEEF);
   159→   159→        int child_status = -1;  /* default: compile fail */
   160→   160→
   161→   161→        if (sigsetjmp(t2_jmp_buf, 1) == 0) {
   162→   162→            alarm(T2_TIMEOUT_SECONDS);
   163→   163→
   164→   164→            vtx_arena_t arena;
   165→   165→            vtx_arena_init(&arena);
   166→   166→
   167→   167→            vtx_type_system_t ts;
   168→   168→            vtx_type_system_init(&ts);
   169→   169→
   170→   170→            vtx_gc_t gc;
   171→   171→            vtx_gc_init(&gc, &ts, VTX_GC_GENERATIONAL);
   172→   172→
   173→   173→            vtx_graph_t graph;
   174→   174→            vtx_graph_init(&graph, a->arg_count > 0 ? a->arg_count : 1);
   175→   175→
   176→   176→            vtx_bytecode_t bc = vtx_asm_emit((vtx_assembler_t *)a);
   177→   177→            vtx_method_desc_t method = {
   178→   178→                .name = a->method_name[0] ? a->method_name : "test",
   179→   179→                .signature = a->method_sig[0] ? a->method_sig : "(I)I",
   180→   180→                .bytecode = &bc,
   181→   181→                .compiled_code = NULL,
   182→   182→                .vtable_index = 0,
   183→   183→                .arg_count = a->arg_count > 0 ? a->arg_count : 1,
   184→   184→                .is_virtual = false,
   185→   185→            };
   186→   186→
   187→   187→            int rc = vtx_graph_build(&graph, &bc, &method, &arena);
   188→   188→            if (rc == 0) {
   189→   189→                vtx_pipeline_config_t config = vtx_pipeline_config_t2();
   190→   190→                vtx_code_cache_t cache;
   191→   191→                vtx_code_cache_init(&cache, 1 << 20);
   192→   192→                vtx_method_registry_t registry;
   193→   193→                vtx_method_registry_init(&registry, &arena);
   194→   194→                config.code_cache = &cache;
   195→   195→                config.method_registry = &registry;
   196→   196→                config.method = &method;
   197→   197→
   198→   198→                vtx_compile_result_t result;
   199→   199→                memset(&result, 0, sizeof(result));
   200→   200→                int prc = vtx_pipeline_run(&graph, &config, &arena, &result);
   201→   201→
   202→   202→                if (prc == 0 && result.success && method.compiled_code != NULL) {
   203→   203→                    typedef vtx_value_t (*vtx_jit_entry_t)(
   204→   204→                        const vtx_method_desc_t *, void *, void *,
   205→   205→                        vtx_value_t *, uint32_t);
   206→   206→                    union { void *ptr; vtx_jit_entry_t fn; } u_e; u_e.ptr = method.compiled_code; vtx_jit_entry_t entry = u_e.fn;
   207→   207→                    vtx_value_t arg_v = vtx_make_smi(arg);
   208→   208→                    child_result = entry(&method, NULL, (void*)1, &arg_v, 1);
   209→   209→                    child_status = 0;
   210→   210→                }
   211→   211→
   212→   212→                vtx_compile_result_destroy(&result);
   213→   213→                vtx_pipeline_config_destroy(&config);
   214→   214→                vtx_code_cache_destroy(&cache);
   215→   215→                vtx_method_registry_destroy(&registry);
   216→   216→            }
   217→   217→
   218→   218→            vtx_arena_destroy(&arena);
   219→   219→            vtx_gc_destroy(&gc);
   220→   220→            vtx_type_system_destroy(&ts);
   221→   221→            alarm(0);
   222→   222→        } else {
   223→   223→            /* Timeout */
   224→   224→            child_status = -2;
   225→   225→        }
   226→   226→
   227→   227→        /* Write result + status to parent. */
   228→   228→        ssize_t w1 = write(pipefd[1], &child_result, sizeof(child_result));
   229→   229→        ssize_t w2 = write(pipefd[1], &child_status, sizeof(child_status));
   230→   230→        (void)w1; (void)w2;
   231→   231→        close(pipefd[1]);
   232→   232→        _exit(0);
   233→   233→    }
   234→   234→
   235→   235→    /* Parent: wait for child. */
   236→   236→    close(pipefd[1]);
   237→   237→    int status = 0;
   238→   238→    int poll_ms = 100;
   239→   239→    int total_ms = (T2_TIMEOUT_SECONDS + 2) * 1000;
   240→   240→    for (int elapsed = 0; elapsed < total_ms; elapsed += poll_ms) {
   241→   241→        pid_t r = waitpid(pid, &status, WNOHANG);
   242→   242→        if (r == pid) break;
   243→   243→        if (r < 0) break;
   244→   244→        usleep((useconds_t)poll_ms * 1000u);
   245→   245→    }
   246→   246→    if (waitpid(pid, &status, WNOHANG) == 0) {
   247→   247→        kill(pid, SIGKILL);
   248→   248→        waitpid(pid, &status, 0);
   249→   249→    }
   250→   250→
   251→   251→    if (WIFEXITED(status) && WEXITSTATUS(status) == 0) {
   252→   252→        /* Child exited cleanly — read result + status. */
   253→   253→        vtx_value_t child_result;
   254→   254→        int child_status;
   255→   255→        ssize_t n1 = read(pipefd[0], &child_result, sizeof(child_result));
   256→   256→        ssize_t n2 = read(pipefd[0], &child_status, sizeof(child_status));
   257→   257→        if (n1 == sizeof(child_result) && n2 == sizeof(child_status)) {
   258→   258→            *out_result = child_result;
   259→   259→            *pipeline_status = child_status;
   260→   260→        } else {
   261→   261→            *pipeline_status = -3;  /* partial read = crash */
   262→   262→        }
   263→   263→    } else {
   264→   264→        *pipeline_status = -3;  /* crash */
   265→   265→    }
   266→   266→    close(pipefd[0]);
   267→   267→}

   269→   269→/* Helper: compare two vtx_value_t for equality. */
   270→   270→static bool values_equal(vtx_value_t a, vtx_value_t b) {
   271→   271→    return a == b;  /* tagged values: equality is bit-equality */
   272→   272→}
   273→   273→
   274→   274→/* Statistics across all check_tier_equivalence calls. */
   275→   275→static uint32_t g_total = 0, g_match = 0, g_mismatch = 0, g_skip = 0, g_t2fail = 0, g_crash = 0;
   276→   276→
   277→   277→/* Helper: run a single program through both T0 and T2, return match status.
   278→   278→ *
   279→   279→ * ALWAYS returns true — individual test cases don't fail. Instead, all
   280→   280→ * mismatches/skips are tracked in g_* counters and the
   281→   281→ * tier_eq_aggregate_summary test asserts on the total mismatch count.
   282→   282→ *
   283→   283→ * This design lets the test suite complete even when T2 has many bugs
   284→   284→ * (which the audit explicitly says it will), while still surfacing every
   285→   285→ * mismatch via stdout diagnostics.
   286→   286→ */
   287→   287→static bool check_tier_equivalence(const char *test_name,
   288→   288→                                    const char *prog_text,
   289→   289→                                    int64_t arg) {
   290→   290→    vtx_assembler_t a0;
   291→   291→    vtx_asm_init(&a0);
   292→   292→
   293→   293→    vtx_value_t t0_result = run_t0(prog_text, arg, &a0);
   294→   294→    int prc = 0;
   295→   295→    vtx_value_t t2_result;
   296→   296→    run_t2_in_child(&a0, arg, &t2_result, &prc);
   297→   297→
   298→   298→    g_total++;
   299→   299→    if (prc == -1) {
   300→   300→        g_t2fail++;
   301→   301→        printf("[tier_eq] SKIP(compile) %s(arg=%lld): T0=0x%llX\n",
   302→   302→               test_name, (long long)arg, (unsigned long long)t0_result);
   303→   303→    } else if (prc == -2) {
   304→   304→        g_skip++;
   305→   305→        printf("[tier_eq] SKIP(timeout) %s(arg=%lld): T0=0x%llX (T2 hung)\n",
   306→   306→               test_name, (long long)arg, (unsigned long long)t0_result);
   307→   307→    } else if (prc == -3) {
   308→   308→        g_crash++;
   309→   309→        printf("[tier_eq] SKIP(crash)   %s(arg=%lld): T0=0x%llX (T2 segfaulted)\n",
   310→   310→               test_name, (long long)arg, (unsigned long long)t0_result);
   311→   311→    } else if (values_equal(t0_result, t2_result)) {
   312→   312→        g_match++;
   313→   313→    } else {
   314→   314→        g_mismatch++;
   315→   315→        printf("[tier_eq] MISMATCH %s(arg=%lld): T0=0x%llX T2=0x%llX\n",
   316→   316→               test_name, (long long)arg,
   317→   317→               (unsigned long long)t0_result,
   318→   318→               (unsigned long long)t2_result);
   319→   319→    }
   320→   320→
   321→   321→    vtx_asm_destroy(&a0);
   322→   322→    return true;  /* individual tests don't fail — aggregate checks totals */
   323→   323→}
